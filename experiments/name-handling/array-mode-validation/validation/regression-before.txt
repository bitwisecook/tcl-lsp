#![cfg(have_tommath)]

use std::rc::Rc;
use tcl_host_native::NativeHost;
use tcl_runtime::interp::{Code, Interp};

#[test]
fn variable_arity_and_array_name_modes_match_original_oracles() {
    let source = include_bytes!(
        "../../../experiments/name-handling/results/native-variable-array-modes/conformance.tcl"
    );
    for (environment, expected) in [
        ("tcl8.4", include_bytes!("../../../experiments/name-handling/results/native-variable-array-modes/c-8.4.20/conformance-stdout.txt").as_slice()),
        ("tcl8.5", include_bytes!("../../../experiments/name-handling/results/native-variable-array-modes/c-8.5.19/conformance-stdout.txt").as_slice()),
        ("tcl8.6", include_bytes!("../../../experiments/name-handling/results/native-variable-array-modes/c-8.6.18/conformance-stdout.txt").as_slice()),
        ("tcl9.0", include_bytes!("../../../experiments/name-handling/results/native-variable-array-modes/c-9.0.4/conformance-stdout.txt").as_slice()),
        ("tcl9.1", include_bytes!("../../../experiments/name-handling/results/native-variable-array-modes/c-9.1.0/conformance-stdout.txt").as_slice()),
        ("jim", include_bytes!("../../../experiments/name-handling/results/native-variable-array-modes/jim-5bac7c9/conformance-stdout.txt").as_slice()),
    ] {
        let mut interp = Interp::with_native_core(
            Rc::new(NativeHost::new()),
            tcl_registry::model::resolve_environment(environment).unit_profile(),
            tcl_registry::special_vars::NativeBootstrapInputs::default(),
        ).expect("matching native core");
        assert_eq!(interp.eval_str(source), Code::Ok, "{environment}: {:?}; refusal={:?}", interp.result_bytes(), interp.native_access_refusal());
        let mut observed = interp.result_bytes();
        observed.push(b'\n');
        assert_eq!(observed, expected, "{environment}");
    }
}

#[test]
fn variable_declaration_upper_arity_matches_original_oracles() {
    for (environment, oracle) in [
        ("tcl8.4", include_str!("../../../experiments/name-handling/results/native-variable-array-modes/c-8.4.20/extra-arity-stdout.txt")),
        ("tcl8.5", include_str!("../../../experiments/name-handling/results/native-variable-array-modes/c-8.5.19/extra-arity-stdout.txt")),
        ("tcl8.6", include_str!("../../../experiments/name-handling/results/native-variable-array-modes/c-8.6.18/extra-arity-stdout.txt")),
        ("tcl9.0", include_str!("../../../experiments/name-handling/results/native-variable-array-modes/c-9.0.4/extra-arity-stdout.txt")),
        ("tcl9.1", include_str!("../../../experiments/name-handling/results/native-variable-array-modes/c-9.1.0/extra-arity-stdout.txt")),
        ("jim", include_str!("../../../experiments/name-handling/results/native-variable-array-modes/jim-5bac7c9/extra-arity-stdout.txt")),
    ] {
        let mut interp = Interp::with_native_core(
            Rc::new(NativeHost::new()),
            tcl_registry::model::resolve_environment(environment).unit_profile(),
            tcl_registry::special_vars::NativeBootstrapInputs::default(),
        ).expect("matching native core");
        for (row, source) in oracle.lines().zip([
            "variable ::nh_extra_a A ::nh_extra_b",
            "variable ::nh_extra_c C ::nh_extra_d D",
        ]) {
            let wanted = tcl_syntax::list::split_list(row).expect("retained original row");
            let code = interp.eval_str(source.as_bytes());
            let expected_code = match wanted[2].as_ref() {
                "0" => Code::Ok,
                "1" => Code::Error,
                other => panic!("unexpected original completion: {other}"),
            };
            assert_eq!(code, expected_code, "{environment}: {source}; refusal={:?}", interp.native_access_refusal());
            assert_eq!(interp.result_bytes(), wanted[3].as_bytes(), "{environment}: {source}");
        }
    }
}
