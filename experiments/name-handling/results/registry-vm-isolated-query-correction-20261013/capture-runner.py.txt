import datetime,hashlib,json,os,subprocess,re
from pathlib import Path
root=Path('/workspace/name-implementation-worktrees/registry_vm')
out=root/'experiments/name-handling/results/registry-vm-isolated-query-correction-20261013'
selectors=json.loads((out/'planned-selectors.json').read_text())
paths=sorted([p for p in (root/'rust').rglob('*') if p.is_file() and '/target/' not in str(p)]+[p for p in (root/'experiments/name-handling').rglob('*') if p.is_file() and '/results/' not in str(p)]+[root/'Cargo.toml',root/'Cargo.lock',root/'rust-toolchain.toml'])
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
scope=lambda:{str(p.relative_to(root)):sha(p) for p in paths}
before=scope();(out/'source-hashes-before.json').write_text(json.dumps(before,indent=2)+'\n');rows=[]
for idx,(package,selector) in enumerate(selectors,1):
 dest=out/f'{idx:02d}';dest.mkdir(exist_ok=False)
 cmd=['cargo','test','--offline','-p',package,'--lib',selector,'--','--exact','--nocapture','--test-threads=1']
 print('START',idx,selector,flush=True)
 with (dest/'stdout.txt').open('wb') as stdout,(dest/'stderr.txt').open('wb') as stderr:
  p=subprocess.run(cmd,cwd=root,env=os.environ.copy(),stdout=stdout,stderr=stderr,check=False)
 row={'package':package,'selector':selector,'command':cmd,'exit':p.returncode,'stdout_sha256':sha(dest/'stdout.txt'),'stderr_sha256':sha(dest/'stderr.txt'),'time':datetime.datetime.now(datetime.timezone.utc).isoformat()}
 actual=(dest/'stdout.txt').read_text(errors='replace')
 executed=re.search(r'running (\d+) test',actual)
 row['actual_test_count']=int(executed.group(1)) if executed else None
 binaries=re.findall(r'Running unittests.*?\((target/[^)]+)\)',(dest/'stderr.txt').read_text(errors='replace'))
 row['test_binaries']={binary:sha(root/binary) for binary in binaries if (root/binary).is_file()}
 rows.append(row);(dest/'record.json').write_text(json.dumps(row,indent=2)+'\n');(out/'records.json').write_text(json.dumps(rows,indent=2)+'\n')
 print('END',idx,p.returncode,flush=True)
after=scope();(out/'source-hashes-after.json').write_text(json.dumps(after,indent=2)+'\n')
(out/'capture-runner.py.txt').write_bytes(Path(__file__).read_bytes())
record={'source_stable':before==after,'checkout_commit':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),'checkout':str(root),'target':os.environ['CARGO_TARGET_DIR'],'jobs':os.environ.get('CARGO_BUILD_JOBS'),'compiled_rust_inputs':'all materialized rust files including restored tracked fixture/C/build inputs; retained experiment files excluding results; root manifest/lock/toolchain','test_debug':os.environ.get('CARGO_PROFILE_TEST_DEBUG'),'rustc':subprocess.check_output(['rustc','-Vv'],text=True),'cargo':subprocess.check_output(['cargo','-V'],text=True),'selectors':len(rows),'successful_processes':sum(r['exit']==0 for r in rows),'accepted_exact_tests':sum(r['exit']==0 and r['actual_test_count']==1 for r in rows),'zero_test_attempts':sum(r['actual_test_count']==0 for r in rows),'build_blocked_attempts':sum(r['actual_test_count'] is None for r in rows),'scope':'2 pure selected-provider query owners,20 Registry metadata/which/layout selectors,6 Compiler emission/metadata/actual-original-map controls,13 genuineVM namespace/variable-query selectors,6 genuineVM original command-query selectors comparing150 retained reference query observations,2 genuineVM imported compiler/TclOO controls; accepts only exactly1 executed passing test per selector, original reporting bytes never recover native identities'}
(out/'provenance.json').write_text(json.dumps(record,indent=2)+'\n');print('SEALED',record,flush=True)
