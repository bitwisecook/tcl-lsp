"""Capture original upstream Jim naming suites with its own tcltest wrapper."""

import argparse
import datetime
import hashlib
import json
from pathlib import Path
import re
import subprocess
import tempfile


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--tree", type=Path, required=True)
    parser.add_argument("--dest", type=Path, required=True)
    parser.add_argument("--timeout", type=float, default=60)
    parser.add_argument("--suites", nargs="+", default=[
        "namespace", "proc", "proc-new", "procstatic", "procref", "rename",
        "alias", "upvar", "jiminterp", "infoframe", "xtrace", "stacktrace",
    ])
    args = parser.parse_args()
    tree, dest = args.tree.resolve(), args.dest.resolve()
    if dest.exists() and any(dest.iterdir()):
        parser.error("destination must be new or empty")
    dest.mkdir(parents=True, exist_ok=True)
    binary = tree / "jimsh"
    version = subprocess.run([str(binary), "-e", "puts [info patchlevel]"],
                             check=True, capture_output=True).stdout.decode().strip()
    revision = subprocess.check_output(["git", "-C", str(tree), "rev-parse", "HEAD"],
                                       text=True).strip()
    rows = []
    for stem in args.suites:
        source = tree / "tests" / (stem + ".test")
        out = dest / stem
        out.mkdir()
        command = [str(binary), str(source)]
        with tempfile.TemporaryDirectory(prefix="jim-naming-") as cwd:
            try:
                run = subprocess.run(command, cwd=cwd, capture_output=True,
                                     timeout=args.timeout)
                stdout, stderr, status, timed_out = run.stdout, run.stderr, run.returncode, False
            except subprocess.TimeoutExpired as error:
                stdout, stderr, status, timed_out = error.stdout or b"", error.stderr or b"", None, True
        (out / "stdout.txt").write_bytes(stdout)
        (out / "stderr.txt").write_bytes(stderr)
        counts = re.findall(rb"Total\s+(\d+)\s+Passed\s+(\d+)\s+Skipped\s+(\d+)\s+Failed\s+(\d+)", stdout)
        row = {"suite": stem, "command": command, "timeout_seconds": args.timeout,
               "process_exit_code": status, "timed_out": timed_out,
               "source_sha256": digest(source), "stdout_sha256": digest(out / "stdout.txt"),
               "stderr_sha256": digest(out / "stderr.txt"),
               "counts": dict(zip(("total", "passed", "skipped", "failed"),
                                  map(int, counts[-1]))) if counts else None}
        (out / "record.json").write_text(json.dumps(row, indent=2) + "\n")
        rows.append(row)
    provenance = {"captured_at_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
                  "provider": "Jim", "patchlevel": version, "revision": revision,
                  "binary_sha256": digest(binary),
                  "harnesses": {str(p.relative_to(tree)): digest(p) for p in
                                [tree / "tcltest.tcl", tree / "tests/testing.tcl"]},
                  "runner_sha256": digest(Path(__file__)),
                  "subject": "upstream Jim only; these captures do not execute the VM/runtime",
                  "startup": "extension-initialised upstream jimsh; original tests source their own testing.tcl and tcltest.tcl"}
    (dest / "provenance.json").write_text(json.dumps(provenance, indent=2) + "\n")
    (dest / "records.json").write_text(json.dumps(rows, indent=2) + "\n")
    for row in rows:
        print(row["suite"], row["counts"], "exit", row["process_exit_code"])


if __name__ == "__main__":
    main()
