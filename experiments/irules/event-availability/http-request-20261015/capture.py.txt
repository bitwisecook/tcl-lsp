#!/usr/bin/env python3
"""Retain the official event reference and complete fetch-process provenance."""
import argparse
import datetime
import hashlib
import json
import pathlib
import sys
import urllib.request

parser = argparse.ArgumentParser()
parser.add_argument("output", type=pathlib.Path)
args = parser.parse_args()
args.output.mkdir(parents=True, exist_ok=False)
executable = pathlib.Path(sys.executable).resolve()
source = pathlib.Path(__file__).resolve()
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
record = {
    "question": "What introduction release does the official HTTP_REQUEST reference declare?",
    "provider": "F5 BIG-IP iRules API reference documentation",
    "url": "https://clouddocs.f5.com/api/irules/HTTP_REQUEST.html",
    "invocation": sys.argv,
    "python_executable": str(executable),
    "python_version": sys.version,
    "python_sha256_before": sha(executable),
    "capture_source_sha256": sha(source),
    "started_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
    "context": "Documentation retrieval; no appliance interpreter, TMM, MCPD or iApps execution.",
}
try:
    with urllib.request.urlopen(record["url"], timeout=30) as response:
        body = response.read()
        record.update(status=response.status, final_url=response.url, headers=dict(response.headers))
        artifact = args.output / "HTTP_REQUEST.html"
        artifact.write_bytes(body)
        record.update(artifact=artifact.name, artifact_sha256=sha(artifact), bytes=len(body))
    record["exit_code"] = 0
except Exception as error:
    record.update(setup_error=repr(error), exit_code=1)
record["python_sha256_after"] = sha(executable)
record["completed_utc"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
(args.output / "capture.json").write_text(json.dumps(record, indent=2) + "\n")
print(json.dumps({key: record[key] for key in ("status", "artifact_sha256", "bytes", "exit_code") if key in record}))
sys.exit(record["exit_code"])
