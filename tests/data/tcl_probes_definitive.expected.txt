§8 setup: ::section8 has three vars
  '{a\}b}'  hex=61 5c 7d 62  value=VAL-A-bs-cb-b
  'a\}b'  hex=61 7d 62  value=VAL-AcbB
  'a{b}c'  hex=61 7b 62 7d 63  value=VAL-AbBcC

Read ${a\}b} from inside ::section8:
  rc=0 result_or_err=VAL-A-bs-cb-b

Read [set "a\\\}b"] from inside ::section8 (literal name 'a\}b'):
  rc=0 result_or_err=VAL-A-bs-cb-b

Read ${a{b}c} from inside ::section8:
  rc=0 result_or_err=VAL-AbBcC

Try EVERY $-form to read var 'a}b' (3 chars, literal close-brace):
  \$\{a\}b\}:
    rc=0 -> VAL-A-bs-cb-b
  bare $ form ('$a\\}b' -- bare stops at non-word):
    rc=1 -> {can't read "a": no such variable}
  via [set "a\\}b"]:
    rc=0 -> VAL-AcbB
