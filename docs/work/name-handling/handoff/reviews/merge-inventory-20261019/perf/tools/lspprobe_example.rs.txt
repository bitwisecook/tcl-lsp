//! Temporary ordinary LSP-path perf probe. Delete the example after use.
use std::hint::black_box;
use std::time::Instant;
use tcl_compiler::analyser::Analyser;
use tcl_lsp_core::source_graph::{RunEdge, RunEdgeKind, RunOrder};
use tcl_lsp_core::workspace_index::WorkspaceIndex;

fn main() {
    let args: Vec<_> = std::env::args().collect();
    let mode = args.get(1).map_or("document", String::as_str);
    let sizes = args.get(2).map_or("10,20,40,80", String::as_str);
    let profile = tcl_dialect::DialectProfile::find("tcl8.6").expect("stock profile");
    for n in sizes.split(',').map(|s| s.parse::<usize>().expect("size")) {
        if mode == "chain" {
            let edges: Vec<_> = (1..n).map(|i| RunEdge {
                parent: format!("file:///f{}.tcl", i - 1),
                child: format!("file:///f{i}.tcl"),
                at: 1,
                enclosing_body: None,
                kind: RunEdgeKind::Source,
            }).collect();
            for repetition in 0..3 {
                let start = Instant::now();
                let order = RunOrder::build(black_box(&edges));
                let micros = start.elapsed().as_micros();
                black_box(order);
                println!("chain n={n} repetition={repetition} build_us={micros}");
            }
            continue;
        }
        let source: String = (0..n).map(|i| format!("proc p{i:08} {{}} {{}}\np{i:08}\n")).collect();
        let start = Instant::now();
        let analysis = Analyser::new().analyse(&source, "tcl8.6");
        let analyse_us = start.elapsed().as_micros();
        assert_eq!(analysis.all_procs.len(), n, "the actual declarations must be retained");
        let uri = "file:///flat.tcl";
        for repetition in 0..3 {
            let start = Instant::now();
            let index = WorkspaceIndex::from_documents([(uri, &analysis)]);
            let index_us = start.elapsed().as_micros();
            assert_eq!(index.procs().count(), n, "index intake cannot withdraw the measured rows");
            let start = Instant::now();
            for _ in 0..100 {
                assert!(!black_box(&index).workspace_command_exists_for_call("::missing", false));
            }
            let existence_us = start.elapsed().as_micros();
            let start = Instant::now();
            let lenses = tcl_lsp_core::code_lens::code_lenses(&source, profile, Some(&analysis), None, uri);
            let lenses_us = start.elapsed().as_micros();
            assert_eq!(lenses.len(), n, "an unavailable provider is not a fast successful result");
            black_box(lenses);
            println!("document n={n} repetition={repetition} analyse_us={analyse_us} index_us={index_us} existence_100_us={existence_us} lenses_us={lenses_us}");
        }
    }
}
