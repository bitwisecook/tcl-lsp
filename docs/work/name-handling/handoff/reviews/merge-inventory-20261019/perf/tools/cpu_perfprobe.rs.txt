//! Temporary perf probe (perf agent). Delete when done.
use std::sync::atomic::Ordering;
use std::time::Instant;
use tcl_compiler::analyser::Analyser;

fn cpu_ticks() -> u64 {
    let stat = std::fs::read_to_string("/proc/self/stat").expect("Linux process CPU counters");
    let tail = stat.rsplit_once(')').expect("stat command name").1;
    let fields: Vec<_> = tail.split_whitespace().collect();
    let user: u64 = fields[11].parse().unwrap();
    let system: u64 = fields[12].parse().unwrap();
    user + system
}

fn generate(kind: &str, n: usize) -> String {
    let mut s = String::new();
    match kind {
        "calls" => {
            s.push_str("proc p {a} {return $a}\n");
            for _ in 0..n {
                s.push_str("p 1\n");
            }
        }
        "procs" => {
            for i in 0..n {
                s.push_str(&format!("proc capped_{i} {{}} {{}}\n"));
            }
        }
        "sets" => {
            for i in 0..n {
                s.push_str(&format!("set v{i} {i}\n"));
            }
        }
        "upvar" => {
            s.push_str("proc inc {name} {upvar 1 $name v; incr v}\n");
            for i in 0..n {
                s.push_str(&format!(
                    "proc c{i} {{}} {{set x{i} 1; inc x{i}; return $x{i}}}\nc{i}\n"
                ));
            }
        }
        _ => panic!("kind"),
    }
    s
}

fn analyse_mode(
    src: &str,
    dialect: &str,
    old: bool,
    input: Option<&tcl_compiler::analyser::ResolvedAnalysisInput>,
) -> (u128, tcl_compiler::analyser::AnalysisResult) {
    tcl_compiler::realm::ZZ_NO_MEMO.store(old, Ordering::Relaxed);
    tcl_lexer::ZZ_OLD.store(old, Ordering::Relaxed);
    let before_cpu = cpu_ticks();
    let t = Instant::now();
    let mut analyser = Analyser::new();
    if let Some(input) = input {
        analyser = analyser.with_resolved_input(input.clone());
    }
    let r = analyser.analyse(src, dialect);
    let ms = t.elapsed().as_millis();
    let elapsed_cpu = cpu_ticks().checked_sub(before_cpu).unwrap();
    eprintln!("CPU old={old} ticks={elapsed_cpu} bytes={}", src.len());
    tcl_compiler::realm::ZZ_NO_MEMO.store(false, Ordering::Relaxed);
    tcl_lexer::ZZ_OLD.store(false, Ordering::Relaxed);
    (ms, r)
}

fn diff(r0: &tcl_compiler::analyser::AnalysisResult, r1: &tcl_compiler::analyser::AnalysisResult) {
    let d0: Vec<_> = r0.diagnostics.iter().map(|d| format!("{d:?}")).collect();
    let d1: Vec<_> = r1.diagnostics.iter().map(|d| format!("{d:?}")).collect();
    eprintln!(
        "MISMATCH diagnostics_debug_equal={} diagnostics_equal={}",
        d0 == d1,
        r0.diagnostics == r1.diagnostics
    );
    if std::env::var_os("ZZ_DIFF").is_some() {
        // Only diagnostics: the complete AnalysisResult PartialEq remains the verdict.
        // Formatting the complete retained graph duplicates its shared subgraphs.
        macro_rules! field {
            ($name:ident) => {
                if r0.$name != r1.$name {
                    eprintln!("FIELD_DIFF {}", stringify!($name));
                }
            };
        }
        field!(analysis_context_unavailable);
        field!(resolved_input);
        field!(global_scope);
        field!(all_procs);
        field!(original_procedure_metadata);
        field!(proc_declaration_sites);
        field!(superseded_procs);
        field!(all_classes);
        field!(original_class_metadata);
        field!(original_class_configuration_metadata);
        field!(original_object_configuration_metadata);
        field!(superseded_classes);
        field!(class_body_spans);
        field!(all_variables);
        field!(all_defined_symbols);
        field!(diagnostics);
        field!(command_invocations);
        field!(package_requires);
        field!(package_provides);
        field!(package_ifneededs);
        field!(package_prefer_latest);
        field!(has_dynamic_providers);
        field!(source_targets);
        field!(path_constant_assignments);
        field!(command_aliases);
        field!(alias_offsets);
        field!(renamed_commands);
        field!(rename_offsets);
        field!(ensemble_subcommand_targets);
        field!(prefixless_ensembles);
        field!(namespace_imports);
        field!(namespace_forgets);
        field!(destroyed_commands);
        field!(namespace_exports);
        field!(namespace_paths);
        field!(namespace_path_computed);
        field!(auto_path_entries);
        field!(qualified_var_refs);
        field!(original_variable_symbols);
        field!(original_variable_write_advice);
        field!(original_variable_symbol_conflicts);
        field!(namespace_refs);
        field!(namespace_name_unknowns);
        field!(dynamic_variable_names);
        field!(namespace_overrides);
        field!(stub_commands);
        field!(stub_expr_defs);
        field!(regex_patterns);
        field!(suppressed_lines);
        field!(unknown_proc_info);
        field!(instance_classes);
        field!(created_instance_commands);
        field!(instance_command_bindings);
        field!(ambiguous_instance_names);
        field!(object_methods);
        field!(object_member_state);
        field!(unresolved_command_sites);
        field!(scoped_command_regions);
        field!(scoped_sibling_defs);
        field!(hierarchy_cache);
        field!(dialect);
        field!(body_lexer_config);
        field!(library_versions);
        if r0.retained_command_realm() != r1.retained_command_realm() {
            eprintln!("FIELD_DIFF command_realm");
        }
        if r0.original_completed_command_world() != r1.original_completed_command_world() {
            eprintln!("FIELD_DIFF original_command_world");
        }
    }
}

/// old1, new, old2: `det` is old1 == old2; `eq` is new == old2 (or old1).
fn run(src: &str, dialect: &str, compare: bool) -> (u128, u128, bool) {
    let retained_input = std::env::var_os("ZZ_RETAIN_INPUT").is_some().then(|| {
        Analyser::new()
            .prepare_analysis_input(dialect)
            .expect("complete available input")
    });
    let input = retained_input.as_ref();
    if !compare {
        let (ms, _) = analyse_mode(src, dialect, false, input);
        return (ms, 0, true);
    }
    let (_, old1) = analyse_mode(src, dialect, true, input);
    let (new_ms, new) = analyse_mode(src, dialect, false, input);
    let (old_ms, old2) = analyse_mode(src, dialect, true, input);
    let det = old1 == old2;
    let eq = new == old2 || new == old1;
    if !det {
        eprintln!("NONDETERMINISTIC old1!=old2");
    }
    if !eq {
        diff(&old2, &new);
    }
    println!("det={det}");
    (new_ms, old_ms, eq)
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let kind = args[1].as_str();
    let compare = std::env::var_os("ZZ_COMPARE").is_some();
    let dialect = std::env::var("ZZ_DIALECT").unwrap_or_else(|_| "tcl8.6".into());
    if kind == "files" {
        let mut bad = 0;
        let mut total = 0;
        for path in &args[2..] {
            let src = std::fs::read_to_string(path).expect("every corpus input must be readable");
            let res = std::panic::catch_unwind(|| run(&src, &dialect, compare));
            total += 1;
            match res {
                Ok((m, p, eq)) => {
                    if !eq {
                        bad += 1;
                    }
                    println!(
                        "{} memo_ms={m} plain_ms={p} eq={eq} {path}",
                        if eq { "OK" } else { "BAD" }
                    );
                }
                Err(_) => {
                    bad += 1;
                    println!("PANIC {path}");
                }
            }
        }
        println!("files={total} mismatches={bad}");
        assert_eq!(bad, 0, "corpus results must be identical without panics");
        return;
    }
    for n in args[2].split(',').map(|x| x.parse::<usize>().unwrap()) {
        let src = generate(kind, n);
        let (m, p, eq) = run(&src, &dialect, compare);
        println!("{kind} n={n} memo_ms={m} plain_ms={p} eq={eq}");
        assert!(eq, "synthetic AnalysisResult must be identical");
    }
}
