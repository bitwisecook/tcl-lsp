use std::sync::atomic::Ordering;
use tcl_compiler::analyser::{Analyser, AnalysisReportComparison};
fn main() {
    let path = std::env::args().nth(1).expect("original fixture");
    let out = std::path::PathBuf::from(std::env::args().nth(2).expect("diagnostic output"));
    let original = std::fs::read_to_string(path).expect("complete original input");
    let config = tcl_lexer::LexerConfig::from_grammar(tcl_registry::InvocationDialect::for_version(tcl_dialect::TclVersion::V8_6).lexer_grammar);
    let commands = tcl_compiler::segmenter::segment_commands_with_offset_and_config(&original, 0, config);
    let end13 = usize::try_from(commands[13].span.start()).expect("next command start");
    let cases = [
        ("empty-other-call", "proc empty {} {}\nproc p {} {return one}\nputs [p]\n".to_owned()),
        ("return-other-call", "proc nothing {} {return}\nproc p {} {return one}\nputs [p]\n".to_owned()),
        ("empty-set", "proc empty {} {}\nset x 1\n".to_owned()),
        ("empty-puts", "proc empty {} {}\nputs x\n".to_owned()),
        ("empty-called", "proc empty {} {}\nempty\n".to_owned()),
        ("return-puts", "proc nothing {} {return}\nputs x\n".to_owned()),
        ("first13-fib-call", format!("{}\nputs [fib 20]\n", &original[..end13])),
    ];
    tcl_compiler::realm::ZZ_NO_MEMO.store(true, Ordering::Relaxed);
    tcl_lexer::ZZ_OLD.store(true, Ordering::Relaxed);
    for (label, source) in cases {
        let report = Analyser::new().analyse_for_report(&source, "tcl8.6");
        let result = report.compare(&report);
        println!("EMPTY_VARIANT label={label} state={report:?} self={result:?}");
        if result != AnalysisReportComparison::Equal {
            std::fs::write(out.join(format!("{label}.tcl")), source).expect("save exact diagnostic input");
        }
    }
}
