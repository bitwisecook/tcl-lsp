+        }
+        let commands = frame.commands.get(frame.next..)?;
+        let mut selected = Vec::with_capacity(commands.len());
+        for command in commands {
+            if command.is_partial {
+                return None;
+            }
+            let mut matches = plan.commands.iter().filter(|original| {
+                original
+                    .words
+                    .first()
+                    .is_some_and(|word| word.span().start() == command.span.start())
+                    && original.span == command.span
+            });
+            let original = matches.next()?.clone();
+            if matches.next().is_some() {
+                return None;
+            }
+            selected.push(original);
+        }
+        plan.commands = selected;
+        Some(plan)
+    }
+}

+impl OriginalSourceConditionalContinuation {
+    /// Continue only the kernel's authentic remaining root/namespace stack.
+    /// Host publication/decode applicability remains separate and conditional.
+    pub fn source_regions(
+        &self,
+        parent: &crate::analyser::AnalysisResult,
+        child: &crate::analyser::AnalysisResult,
+    ) -> Option<OriginalSourceConditionalRegionSet> {
+        if !self.matches_parent_analysis(parent) || !self.matches_child_analysis(child) {
+            return None;
+        }
+        static MEMO: OnceLock<Mutex<Vec<Weak<RegionSetPayload>>>> = OnceLock::new();
+        let memo = MEMO.get_or_init(|| Mutex::new(Vec::new()));
+        let find = |entries: &mut Vec<Weak<RegionSetPayload>>| {
+            entries.retain(|entry| entry.strong_count() != 0);
+            entries
+                .iter()
+                .filter_map(Weak::upgrade)
+                .find(|entry| Arc::ptr_eq(&entry.continuation, &self.0))
+        };
+        if let Some(retained) = find(
+            &mut memo
+                .lock()
+                .unwrap_or_else(std::sync::PoisonError::into_inner),
+        ) {
+            return Some(OriginalSourceConditionalRegionSet(retained));
+        }
+        // Never hold the interner lock while running the shared source kernel.
+        let result = Arc::new(self.replay_region_chain()?);
+        let mut memo = memo
+            .lock()
+            .unwrap_or_else(std::sync::PoisonError::into_inner);
+        if let Some(retained) = find(&mut memo) {
+            return Some(OriginalSourceConditionalRegionSet(retained));
+        }
+        memo.push(Arc::downgrade(&result));
+        Some(OriginalSourceConditionalRegionSet(result))
+    }
+
+    fn replay_region_chain(&self) -> Option<RegionSetPayload> {
+        let context = self.0.point.input().borrowed_context_registry();
+        let mut incoming = Arc::clone(&self.0.child_outcomes);
+        let mut graph = Arc::clone(&self.0.child_graph);
+        let mut regions = Vec::<OriginalSourceConditionalRegion>::new();
+        for index in (0..self.0.plan.frames.len()).rev() {
+            let frame = &self.0.plan.frames[index];
+            if incoming.normal_completion.is_none() || !incoming.abrupt.is_empty() {
+                return Some(RegionSetPayload {
+                    continuation: Arc::clone(&self.0),
+                    regions,
+                    complete: false,
+                    normal: None,
+                });
+            }
+            let state = incoming.normal.as_deref()?.clone();
+            // Each selected child exits in the same source frame as its load.
+            // An outer frame is entered only after the previous restoration.
+            if state.variable_frame != frame.context.frame {
+                return None;
+            }
+            graph = graph.in_conditional_namespace(&frame.context.namespace_key)?;
+            let entry = SourceConditionalRegionEntry(Arc::new(RegionEntryPayload {
+                continuation: Arc::clone(&self.0),
+                predecessor: regions.last().cloned(),
+                frame_index: index,
+                incoming: Arc::clone(&incoming),
+                graph: Arc::clone(&graph),
+            }));
+            let RegionWalk {
+                region,
+                mut outcomes,
+                graph: normal_graph,
+            } = walk_region(entry, context)?;
+            regions.push(region);
+            // Completion remains conditional. Restore only this captured,
+            // positively selected namespace's real parent activation, in order.
+            if outcomes.normal_completion.is_none()
+                || outcomes.normal.is_none()
+                || !outcomes.abrupt.is_empty()
+                || normal_graph.is_none()
+            {
+                return Some(RegionSetPayload {
+                    continuation: Arc::clone(&self.0),
+                    regions,
+                    complete: false,
+                    normal: None,
+                });
+            }
+            if let Some(exit) = &frame.exit {
+                if outcomes.normal.as_ref()?.variable_frame != exit.entered {
+                    return None;
+                }
+                outcomes.restore_frame(&exit.parent_variables, &exit.parent);
+            } else if index != 0 {
+                return None;
+            }
+            incoming = Arc::new(outcomes);
+            graph = normal_graph?;
+        }
+        let mut normal = (*incoming).clone();
+        normal.restore_source_origin(Some(&self.0.point.allocation_site().source));
+        Some(RegionSetPayload {
+            continuation: Arc::clone(&self.0),
+            regions,
+            complete: true,
+            normal: Some((Arc::new(normal), graph)),
+        })
+    }
+}
+
+struct RegionWalk {
+    region: OriginalSourceConditionalRegion,
+    outcomes: SourceOutcomes,
+    graph: Option<Arc<AdviceGraph>>,
+}
+fn walk_region(
+    entry: SourceConditionalRegionEntry,
+    context: &ContextRegistry,
+) -> Option<RegionWalk> {
+    static MEMO: OnceLock<Mutex<Vec<Weak<RegionPayload>>>> = OnceLock::new();
+    let memo = MEMO.get_or_init(|| Mutex::new(Vec::new()));
+    let find = |entries: &mut Vec<Weak<RegionPayload>>| {
+        entries.retain(|candidate| candidate.strong_count() != 0);
+        entries.iter().filter_map(Weak::upgrade).find(|candidate| {
+            let previous = &candidate.entry.0;
+            Arc::ptr_eq(&previous.continuation, &entry.0.continuation)
+                && previous.predecessor == entry.0.predecessor
+                && previous.frame_index == entry.0.frame_index
+                && previous.incoming == entry.0.incoming
+        })
+    };
+    if let Some(retained) = find(
+        &mut memo
+            .lock()
+            .unwrap_or_else(std::sync::PoisonError::into_inner),
+    ) {
+        return Some(region_walk(retained));
+    }
+    let result = Arc::new(walk_region_uncached(entry.clone(), context)?);
+    let mut memo = memo
+        .lock()
+        .unwrap_or_else(std::sync::PoisonError::into_inner);
+    if let Some(retained) = find(&mut memo) {
+        return Some(region_walk(retained));
+    }
+    memo.push(Arc::downgrade(&result));
+    Some(region_walk(result))
+}
+
+fn region_walk(region: Arc<RegionPayload>) -> RegionWalk {
+    RegionWalk {
+        outcomes: (*region.outcomes).clone(),
+        graph: region.graph.clone(),
+        region: OriginalSourceConditionalRegion(region),
+    }
+}
+
+fn walk_region_uncached(
+    entry: SourceConditionalRegionEntry,
+    context: &ContextRegistry,
+) -> Option<RegionPayload> {
+    let frame = entry.frame().clone();
+    let point = entry.0.continuation.point.for_conditional_region(
+        entry.clone(),
+        (**entry.0.incoming.normal.as_ref()?).clone(),
+        frame.context.namespace_key.clone(),
+    )?;
+    let supplied = point.internal_region_entry()?;
+    let options = supplied.options();
+    let mut state = point.child_state(context.commands(), options, Some(frame.context.config))?;
+    let origin = Arc::new(SourceOriginId::from_kind(
+        SourceOriginKind::InheritedPoint {
+            parent: point.source_origin(),
+            source: entry
+                .0
+                .continuation
+                .point
+                .allocation_site()
+                .source
+                .source_image()
+                .clone(),
+        },
+    ));
+    state.current_source_origin = Some(Arc::clone(&origin));
+    let mut bindings = SourceCommandBindings {
+        lexer_config: Some(frame.context.config),
+        compilation_scope: supplied.compilation_scope,
+        analyser_source_settings: Some(point.analyser_source_settings()),
+        root_origin: Some(Arc::clone(&origin)),
+        original_analysis_owner: Some(Arc::new(
+            crate::command_binding::OriginalSourceAnalysisOwner {
+                source: Arc::clone(&origin),
+                metadata: state.baseline.metadata_context.clone(),
+                config: frame.context.config,
+            },
+        )),
+        ..SourceCommandBindings::default()
+    };
+    // Reinstall the actual outer stack and exact ordinal; no whole-file
+    // source preparation or byte-equivalent prefix reanalysis occurs.
+    bindings.conditional_source.frames = entry.0.continuation.plan.frames[..entry.0.frame_index]
+        .iter()
+        .cloned()
+        .map(|mut saved| {
+            saved.source = Arc::clone(&origin);
+            Some(saved)
+        })
+        .collect();
+    let mut current = frame.clone();
+    current.source = Arc::clone(&origin);
+    current.ordinal_start = frame.next;
+    bindings.conditional_source.resume = Some(current);
+    let outcomes = bindings.walk_source_segments(
+        &frame.image,
+        frame.base,
+        frame.commands.get(frame.next..)?,
+        &mut state,
+        &frame.context.borrowed(context.commands()),
+        None,
+    );
+    bindings.conditional_source.root_outcomes = Some(Arc::new(outcomes.clone()));
+    bindings.original_completed_root_state = None;
+    if let Some(normal) = &outcomes.normal {
+        state = (**normal).clone();
+    }
+    bindings.final_state = Arc::new(state);
+    bindings.invalidate_unpositioned_projections();
+    let normal_graph = bindings
+        .original_source_transition_seed()?
+        .normal_source_outcome_graph(context);
+    Some(RegionPayload {
+        point,
+        entry,
+        bindings: Arc::new(bindings),
+        outcomes: Arc::new(outcomes),
+        graph: normal_graph,
+    })
+}

+impl OriginalSourceConditionalRegion {
+    #[must_use]
+    pub fn matches_dependencies(
+        &self,
+        parent: &crate::analyser::AnalysisResult,
+        child_prefix: &crate::analyser::AnalysisResult,
+    ) -> bool {
+        let continuation =
+            OriginalSourceConditionalContinuation(Arc::clone(&self.0.entry.0.continuation));
+        continuation.matches_parent_analysis(parent)
+            && continuation.matches_child_analysis(child_prefix)
+    }
+    /// Exact same issuer-owned partial analysis. Host publication and ancestor
+    /// leases remain separate; equal bytes or a whole-image match cannot renew it.
+    #[must_use]
+    pub fn matches_region_analysis(&self, analysis: &crate::analyser::AnalysisResult) -> bool {
+        if analysis.original_source_region.as_ref() != Some(self)
+            || analysis.analysis_context_unavailable.is_some()
+            || analysis.resolved_input.as_ref() != Some(self.input())
+            || analysis.body_lexer_config != Some(self.input().lexer_config())
+        {
+            return false;
+        }
+        let Some(realm) = analysis.retained_command_realm() else {
+            return false;
+        };
+        let bindings = realm.source_bindings_ref();
+        realm.matches_resolved_analysis_input(self.input())
+            && realm.matches_original_source_image(self.source_image(), self.input().lexer_config())
+            && bindings
+                .original_analysis_owner()
+                .zip(self.0.bindings.original_analysis_owner())
+                .is_some_and(|(actual, expected)| Arc::ptr_eq(actual, expected))
+            && bindings.original_inherited_source_entry() == Some(&self.0.point)
+    }
+    pub(crate) fn input(&self) -> &crate::analyser::ResolvedAnalysisInput {
+        self.0.point.input()
+    }
+    pub(crate) fn source_image(&self) -> &tcl_lexer::SourceImage {
+        self.0
+            .entry
+            .0
+            .continuation
+            .point
+            .allocation_site()
+            .source
+            .source_image()
+    }
+    pub(crate) fn commands(&self) -> &[crate::segmenter::SegmentedCommand] {
+        let frame = self.0.entry.frame();
+        &frame.commands[frame.next..]
+    }
+    pub(crate) fn bindings(&self) -> &SourceCommandBindings {
+        &self.0.bindings
+    }
+    pub(crate) fn entry(&self) -> Option<crate::command_binding::SourceAnalysisEntry> {
+        self.0.point.internal_region_entry()
+    }
+    pub(crate) fn configure_analyser(&self, analyser: &mut crate::analyser::Analyser) {
+        self.0.point.analyser_source_settings().apply(analyser);
+    }
+    pub(crate) fn accepts_analyser(&self, analyser: &crate::analyser::Analyser) -> bool {
+        self.0.point.accepts_analyser_source_settings(analyser)
+    }
+}

+    pub(super) fn enter_script(
+        &mut self,
+        image: &SourceImage,
+        base: u32,
+        commands: &[crate::segmenter::SegmentedCommand],
+        state: &ModuleCommandBindings,
+        context: &SourceExecutionContext<'_>,

+        let exit = self.pending_namespace.take();

+            let matches = resume.image == *image
+                && resume.base == base
+                && resume.commands.get(resume.ordinal_start..) == Some(commands)
+                && state.current_source_origin.as_ref() == Some(&resume.source)
+                && resume.context.frame == *context.frame
+                && resume.context.namespace_key == context.namespace_identity()

+                source: Arc::clone(origin),
+                image: image.clone(),
+                base,

+                next: 0,
+                ordinal_start: 0,

+    pub(super) fn retain_command_plan(
+        &mut self,
+        index: usize,
+        command: &crate::segmenter::SegmentedCommand,
+        state: &ModuleCommandBindings,
+        context: &SourceExecutionContext<'_>,
+        prefix_complete: bool,
+    ) {
+        let Some(origin) = state.current_source_origin.as_ref() else {
+            return;
+        };
+        let site = CommandAllocationSite {
+            source: Arc::clone(origin),
+            offset: command.span.start(),
+        };
+        // Every observation participates, including unavailable/rejected ones.
+        // A later positive cannot renew a withdrawn same-site continuation.
+        if self.plans.contains_key(&site) {
+            self.plans.insert(site, None);
+            return;
+        }
+        self.plans.insert(site.clone(), None);
+        let Some(Some(current)) = self.frames.last_mut() else {
+            return;
+        };
+        current.next = current.ordinal_start + index + 1;
+        if !prefix_complete
+            || context.expression_source.is_some()
+            || context.source_purpose != current.context.source_purpose
+            || origin != &current.source

+            || state.unclosed_command_world
+            || state.source_step_observed()
+            || !state.command_observers.is_quiet()

+            return;
+        }

+            return;
+        };

+            site,
+            Some(Arc::new(SourceConditionalPlan {
+                frames: Arc::from(frames),
+            })),
+        );
+    }

+    pub(super) fn leave_script(&mut self) {
+        self.frames.pop();
+    }

+impl super::SourceCommandBindings {
+    pub(crate) fn conditional_source_continuation(
+        &self,
+        point: super::OriginalSourcePointEntry,
+        parent: &crate::analyser::AnalysisResult,
+        child: &crate::analyser::AnalysisResult,
+    ) -> Option<OriginalSourceConditionalContinuation> {
+        self.conditional_source_continuation_with_outcome(point, parent, child, None)
+    }
+
+    pub(crate) fn conditional_source_continuation_with_outcome(
+        &self,
+        point: super::OriginalSourcePointEntry,
+        parent: &crate::analyser::AnalysisResult,
+        child: &crate::analyser::AnalysisResult,
+        conditional_outcome: Option<&OriginalSourceConditionalOutcome>,
+    ) -> Option<OriginalSourceConditionalContinuation> {
+        use std::hash::{Hash, Hasher};
+        use std::sync::{Mutex, OnceLock, Weak};
+        point.matches_parent_analysis(parent).then_some(())?;
+        point.matches_child_analysis(child).then_some(())?;
+        let plan = self
+            .conditional_source
+            .plans
+            .get(point.allocation_site())?
+            .as_ref()?;
+        let dialect = point.source_file_dialect()?;
+        let grammar = dialect.source_file_grammar()?;
+        let child_bindings = child.retained_command_realm()?.source_bindings_ref();
+        let owner = child_bindings.original_analysis_owner()?;
+        let input = child.resolved_input.as_ref()?;
+        owner.matches(child_bindings, input).then_some(())?;
+        let outcomes = match conditional_outcome {
+            Some(outcome) => outcome.outcomes_for(child)?,
+            None => Arc::clone(child_bindings.conditional_source.root_outcomes.as_ref()?),
+        };
+        // This first finite issuer supports a closed normal child route only.
+        // Abrupt/unknown alternatives require their own selected route join.
+        if outcomes.normal_completion.is_none()
+            || outcomes.normal.is_none()
+            || !outcomes.abrupt.is_empty()
+        {
+            return None;
+        }
+        let graph = match conditional_outcome {
+            Some(outcome) => outcome.graph_for(child)?,
+            None => child_bindings
+                .original_source_transition_seed()?
+                .normal_source_outcome_graph(input.borrowed_context_registry())?,
+        };
+        let payload = SourceConditionalContinuationPayload {
+            point,
+            plan: Arc::clone(plan),
+            child_owner: Arc::clone(owner),
+            child_origin: Arc::clone(child_bindings.source_origin()?),
+            child_input: input.clone(),
+            child_outcomes: outcomes,
+            conditional_outcome: conditional_outcome.cloned(),
+            child_graph: graph,
+            dialect,
+            grammar,
+        };
+        type Memo = std::collections::HashMap<u64, Vec<Weak<SourceConditionalContinuationPayload>>>;
+        static MEMO: OnceLock<Mutex<Memo>> = OnceLock::new();
+        let mut hasher = std::collections::hash_map::DefaultHasher::new();
+        payload.point.hash(&mut hasher);
+        let mut memo = MEMO
+            .get_or_init(|| Mutex::new(Memo::new()))
+            .lock()
+            .unwrap_or_else(std::sync::PoisonError::into_inner);
+        let bucket = memo.entry(hasher.finish()).or_default();
+        bucket.retain(|candidate| candidate.strong_count() != 0);
+        if let Some(retained) = bucket
+            .iter()
+            .filter_map(Weak::upgrade)
+            .find(|candidate| candidate.same_request(&payload))
+        {
+            return Some(OriginalSourceConditionalContinuation(retained));
+        }
+        let payload = Arc::new(payload);
+        bucket.push(Arc::downgrade(&payload));
+        Some(OriginalSourceConditionalContinuation(payload))
+    }

+    next_key: u64,

+impl SourcePointInterner {
+    fn issue(&mut self, payload: SourcePointPayload) -> Option<OriginalSourcePointEntry> {
+        let mut hasher = std::collections::hash_map::DefaultHasher::new();
+        payload.hash(&mut hasher);
+        self.issue_in_bucket(payload, hasher.finish())
+    }
+
+    fn issue_in_bucket(
+        &mut self,
+        payload: SourcePointPayload,
+        fingerprint: u64,
+    ) -> Option<OriginalSourcePointEntry> {
+        if self.next_key.is_multiple_of(64) {
+            self.buckets.retain(|_, bucket| {
+                bucket.retain(|(payload, _)| payload.strong_count() != 0);
+                !bucket.is_empty()
+            });
+        }
+        let bucket = self.buckets.entry(fingerprint).or_default();
+        bucket.retain(|(retained, _)| retained.strong_count() != 0);
+        if let Some((payload, key)) = bucket.iter().find_map(|(candidate, key)| {
+            let candidate = candidate.upgrade()?;
+            (candidate.as_ref() == &payload).then_some((candidate, *key))
+        }) {
+            return Some(OriginalSourcePointEntry { payload, key });
+        }
+        self.next_key = self.next_key.checked_add(1)?;
+        let key = self.next_key;
+        let payload = Arc::new(payload);
+        bucket.push((Arc::downgrade(&payload), key));
+        Some(OriginalSourcePointEntry { payload, key })
+    }

+        let mut state = self.payload.snapshot.state.clone();
+        let mut baseline = (*state.baseline).clone();
+        baseline.source_point_entry = Some(self.clone());
+        baseline.invocation_realm = self.payload.snapshot.realm;
+        baseline.semantics = match baseline.logical_source_input.as_ref() {
+            Some(input) => registry.authored_source_semantics_in_context(
+                input.context_registry().context(),
+                baseline.invocation_realm,
+            ),
+            None => registry.effective_semantics_for_dialect_in_realm(
+                baseline.dialect?,
+                baseline.invocation_realm,
+            ),
+        };
+        baseline.native_entry = None;
+        baseline.native_compilation =
+            tcl_registry::native_compilation::NativeCompilationContext::default();
+        baseline.compiled_variable_provider = None;
+        baseline.refresh_fingerprint();
+        state.baseline = Arc::new(baseline);
+        Some(state)
+    }

+    pub(crate) fn namespace_scope(
+        &self,
+    ) -> Option<crate::signature_scan::scope::SignatureNamespaceScope> {
+        use crate::signature_scan::scope::SignatureNamespaceScope;
+        match self.namespace() {
+            SourceNamespaceKey::Authored(key) if self.input().has_logical_source_name_context() => {
+                Some(SignatureNamespaceScope::Symbolic(key.clone()))
+            }
+            namespace => {
+                let policy = self
+                    .payload
+                    .snapshot
+                    .state
+                    .baseline
+                    .execution_name_policy?
+                    .native_recipe()?;
+                if !matches!(
+                    policy.recipe(),
+                    tcl_syntax::naming::NativeNameProtocol::C(_)
+                ) {
+                    return None;
+                }
+                Some(SignatureNamespaceScope::C(
+                    namespace.source_artifact_path()?,
+                ))
+            }
+        }
+    }

+                return true;
+            }
+            entry = point
+                .payload
+                .snapshot
+                .state
+                .baseline
+                .source_point_entry
+                .as_ref();
+        }
+        false
+    }

+                return true;
+            }
+            entry = point
+                .payload
+                .snapshot
+                .state
+                .baseline
+                .source_point_entry
+                .as_ref();
+        }
+        false
+    }

+impl SourceCommandBindings {
+    pub(crate) fn original_inherited_source_entry(&self) -> Option<&OriginalSourcePointEntry> {
+        let point = self.final_state.baseline.source_point_entry.as_ref()?;
+        let origin = self.root_origin.as_ref()?;
+        if self.analyser_source_settings.as_ref() != Some(&point.payload.settings)
+            || self.source_metadata_owner().source_analysis_input() != Some(point.input())
+            || !matches!(origin.kind(), super::SourceOriginKind::InheritedPoint { parent, .. } if parent == &point.source_origin())
+        {
+            return None;
+        }
+        Some(point)
+    }
+
+    pub(crate) fn matches_inherited_source_origin(&self, origin: &super::SourceOriginId) -> bool {
+        self.original_inherited_source_entry().is_some()
+            && self.root_origin.as_deref() == Some(origin)
+    }
+
+    /// Exact original words at a unanimous post-argv/pre-handler point.
+    /// Compiler-only previews cannot substitute for that point inventory.
+    pub(crate) fn original_source_point_entry(
+        &self,
+        input: &ResolvedAnalysisInput,
+        original: &[NativeWord],
+        path: usize,
+    ) -> Option<OriginalSourcePointEntry> {
+        let owner = self.original_analysis_owner()?;
+        owner.matches(self, input).then_some(())?;
+        let head = original.first()?;
+        let offset = head.span().start();
+        let config = input.lexer_config();
+        if self.lexer_config != Some(config)
+            || self.source_metadata_owner().source_analysis_input() != Some(input)
+            || original.iter().any(|word| {
+                word.image() != head.image() || word.config() != config || word.group().expand
+            })
+            || original.get(path.checked_add(1)?).is_none()
+        {
+            return None;
+        }
+        let mut points = self
+            .dispatch_points_at(offset)
+            .filter(|point| !point.declaration_preview);
+        let first = points.next()?;
+        let state = &first.state;
+        if !first.original_arguments_complete_normally
+            || state.baseline.native_entry.is_some()
+            || state.baseline.hosted_execution_context.is_some()
+            || state.has_opaque_domain()
+            || state.unclosed_command_world
+            || state.source_step_observed()
+            || !state.command_observers.is_quiet()
+            || matches!(
+                state.variable_frame.layout(),
+                crate::var_resolve::VariableExecutionFrame::Unknown
+                    | crate::var_resolve::VariableExecutionFrame::Selected { .. }
+                    | crate::var_resolve::VariableExecutionFrame::ReceiverMethod { .. }
+            )
+            || points.any(|point| point != first)
+            || first
+                .original_dispatch_binding()
+                .proved_target()
+                .is_none_or(|target| !target.registry_backed)
+        {
+            return None;
+        }
+        let origin = state.current_source_origin.as_ref()?;
+        let written = first.original_written_words.as_ref()?;
+        if written.site().source != *origin || !written.matches_native_words(original) {
+            return None;
+        }
+        let payload = SourcePointPayload {
+            settings: Arc::clone(self.analyser_source_settings.as_ref()?),
+            owner: Arc::clone(owner),
+            input: input.clone(),
+            site: CommandAllocationSite {
+                source: Arc::clone(origin),
+                offset,
+            },
+            words: Arc::from(original),
+            path,
+            snapshot: Arc::new(SourceLookupSnapshot::in_realm(state.clone(), first.realm)),
+            namespace: first.namespace_key.clone(),
+            compilation_scope: self.compilation_scope,
+            region: None,
+        };
+        issue_source_point(payload)
+    }
