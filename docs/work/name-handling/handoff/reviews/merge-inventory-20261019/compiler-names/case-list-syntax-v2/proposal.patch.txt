+/// Exact list-separator edits for one current original case container. Field
+/// delimiters and bytes stay unchanged; script bodies need separate proofs.
+/// This supplies no runtime dispatch, compiler admission or Normal result.
+pub struct OriginalCaseListSyntaxGapCompaction {
+    original: NativeWord,
+    input: ResolvedAnalysisInput,
+    realm: Arc<crate::realm::CommandBindingRealm>,
+    request: crate::compilation_unit::OriginalOptimiserSourceOrigin,
+    gaps: Vec<(Span, &'static str)>,
+}
+
+impl OriginalCaseListSyntaxGapCompaction {
+    /// Complete original braced, non-expanding operand.
+    #[must_use]
+    pub fn original_operand(&self) -> &NativeWord {
+        &self.original
+    }
+
+    /// Only list-separator source spans may change.
+    #[must_use]
+    pub fn gap_edits(&self) -> &[(Span, &'static str)] {
+        &self.gaps
+    }
+
+    /// The entire source image and all grammar axes remain current.
+    #[must_use]
+    pub fn matches_source(&self, image: &SourceImage, config: LexerConfig) -> bool {
+        self.original.image() == image && self.original.config() == config
+    }
+
+    /// The complete retained input is unchanged.
+    #[must_use]
+    pub fn matches_input(&self, input: &ResolvedAnalysisInput) -> bool {
+        &self.input == input
+    }
+
+    /// Retain the same actual source request and analysis allocation.
+    #[must_use]
+    pub fn matches_analysis(&self, source: &str, analysis: &AnalysisResult) -> bool {
+        analysis.resolved_input.as_ref() == Some(&self.input)
+            && analysis.matches_original_source_image(self.original.image(), self.original.config())
+            && analysis
+                .retained_command_realm_arc()
+                .is_some_and(|current| Arc::ptr_eq(current, &self.realm))
+            && analysis.resolved_registry().is_some_and(|registry| {
+                analysis
+                    .original_optimiser_request_owner(source, registry)
+                    .is_some_and(|current| self.request.same_request_allocation(current))
+            })
+    }
+
+    /// Keep the original registry generation and complete context.
+    #[must_use]
+    pub fn matches_registry(&self, registry: &CommandRegistry) -> bool {
+        self.input.borrowed_context_registry().commands().snapshot() == registry.snapshot()
+    }
+}

+/// Independently join actual compiler syntax and current source selection
+/// before compacting a plain paired case list. Neither a declaration fallback
+/// nor a body script receipt can issue list-separator permission.
+#[must_use]
+pub fn original_case_list_syntax_gap_compaction(
+    source: &str,
+    analysis: &AnalysisResult,
+    segment: &crate::segmenter::SegmentedCommand,
+) -> Option<OriginalCaseListSyntaxGapCompaction> {
+    let input = analysis.resolved_input.as_ref()?;
+    let config = input.lexer_config();
+    let realm = analysis.retained_command_realm_arc()?;
+    let image = super::retained_document_image(source, realm, config)?;
+    let registry = analysis.resolved_registry()?;
+    let request = analysis.original_optimiser_request_owner(source, registry)?;
+    if analysis.body_lexer_config != Some(config)
+        || !analysis.matches_original_source_image(&image, config)
+        || !realm.matches_resolved_analysis_input(input)
+        || config.list_parse != tcl_dialect::ListParse::Strict
+    {
+        return None;
+    }
+    let words = super::source_registry_words(source, analysis, segment);
+    #[cfg(test)]
+    eprintln!(
+        "CASE_LIST_SYNTAX_SOURCE available={} selected={} preserves={}",
+        words.is_some(),
+        words
+            .as_ref()
+            .is_some_and(|words| matches!(words.source(), OriginalRegistrySource::Selected)),
+        words
+            .as_ref()
+            .is_some_and(OriginalRegistryWords::operands_preserve_source_lookup),
+    );
+    let words = words?;
+    if !matches!(words.source(), OriginalRegistrySource::Selected)
+        || !words.operands_preserve_source_lookup()
+    {
+        return None;
+    }
+    let mut tokens = CommandTokens::from_segmented(&image.source_map(), config, segment);
+    realm.stamp_original_tokens(&mut tokens);
+    let binding = tokens.source_binding.as_ref()?;
+    let syntax = binding.original_compiler_argument_syntax(&tokens, input, registry)?;
+    if syntax.image() != &image || syntax.config() != config {
+        return None;
+    }
+    let metadata = syntax.metadata();
+    let advice = binding.original_compilation_lookup_advice(&tokens)?;
+    if !advice.closed_authored_simulation_source_lookup(&tokens, metadata, registry)
+        || realm.original_source_registry_barrier(input.borrowed_context_registry(), &tokens)
+    {
+        return None;
+    }
+    let selected =
+        super::super::resolve_original_declared_layout(registry, metadata, &tokens, &advice)?;
+    if selected.facts.arity_accepts_frozen_arguments() != Some(true)
+        || selected.effective.words.len() != tokens.words().len()
+        || selected.effective.words.get(1..) != tokens.words().get(1..)
+        || selected.effective.origins.len() != tokens.words().len()
+        || selected.effective.origins.first()
+            != Some(&crate::registry_invocation::InvocationWordOrigin::ResolvedHead)
+        || !selected
+            .effective
+            .origins
+            .iter()
+            .enumerate()
+            .skip(1)
+            .all(|(index, origin)| {
+                *origin == crate::registry_invocation::InvocationWordOrigin::Written(index)
+            })
+    {
+        return None;
+    }
+    let argument = selected_case_argument(&words, &selected, metadata, advice.realm(), input)?;
+    let original = words.original_argument_word(argument)?.clone();
+    let native = super::super::original_native_compiler_words(
+        &image,
+        tokens.words(),
+        syntax.site().offset,
+        config,
+    )?;
+    if native.get(argument.checked_add(1)?) != Some(&original)
+        || original.group().kind != WordKind::Braced
+        || original.group().expand
+    {
+        return None;
+    }
+    let content = original.content_span().ok()?;
+    let bytes = image.bytes().get(content.as_range())?;
+    // Cooked braced contents have no exact in-place list geometry.
+    if tcl_syntax::word_rules::original_static_word_unicode_value(&original)?.as_slice() != bytes {
+        return None;
+    }
+    let gaps = list_gap_edits(bytes, content.start(), config)?;
+    Some(OriginalCaseListSyntaxGapCompaction {
+        original,
+        input: input.clone(),
+        realm: Arc::clone(realm),
+        request: request.clone(),
+        gaps,
+    })
+}

+fn selected_case_argument(
+    words: &OriginalRegistryWords,
+    selected: &super::super::ResolvedStatementInvocation,
+    metadata: super::super::InvocationMetadataContext<'_>,
+    realm: tcl_dialect::model::InvocationRealm,
+    input: &ResolvedAnalysisInput,
+) -> Option<usize> {
+    let context = input.borrowed_context_registry();
+    words.with_source_schema(context, |current| {
+        selected.with_metadata_schema(context.commands(), metadata, realm, |compiler| {
+            let current_owner = current.authored_source_descriptors();
+            let compiler_owner = compiler.authored_source_descriptors();
+            if !std::ptr::eq(current_owner.command, compiler_owner.command)
+                || !match (current_owner.subcommand, compiler_owner.subcommand) {
+                    (None, None) => true,
+                    (Some(a), Some(b)) => std::ptr::eq(a, b),
+                    _ => false,
+                }
+            {
+                return None;
+            }
+            let (case, layout) = compiler.authored_source_case_invocation()?;
+            let (_, current_layout) = current.authored_source_case_invocation()?;
+            if layout != current_layout
+                || !case.clause_flags.is_empty()
+                || !case.clause_value_flags.is_empty()
+                || case.clause_regex_flag.is_some()
+                || case.clause_end_options_flag.is_some()
+                || case.clause_force_inline_flag.is_some()
+                || case.clause_force_list_flag.is_some()
+                || case.clause_force_list_shape.is_some()
+                || case.allow_omitted_final_body
+            {
+                return None;
+            }
+            layout.clause_list_index
+        })
+    })?
+}

+fn list_fields(bytes: &[u8], config: LexerConfig) -> Option<Vec<std::ops::Range<usize>>> {
+    let mut fields = Vec::new();
+    let mut cursor = 0;
+    while let Some(element) =
+        tcl_syntax::list::find_element_bytes_with_syntax(bytes, cursor, config.list_parse).ok()?
+    {
+        let grouped = element.braced
+            || element
+                .value
+                .start
+                .checked_sub(1)
+                .and_then(|offset| bytes.get(offset))
+                == Some(&b'"');
+        let start = if grouped {
+            element.value.start.checked_sub(1)?
+        } else {
+            element.value.start
+        };
+        let end = element.value.end.checked_add(usize::from(grouped))?;
+        let gap = bytes.get(cursor..start)?;
+        if !gap.iter().copied().all(tcl_syntax::list::is_list_space)
+            || !bytes
+                .get(end..element.next)?
+                .iter()
+                .copied()
+                .all(tcl_syntax::list::is_list_space)
+        {
+            return None;
+        }
+        fields.push(start..end);
+        cursor = element.next;
+    }
+    bytes
+        .get(cursor..)?
+        .iter()
+        .copied()
+        .all(tcl_syntax::list::is_list_space)
+        .then_some(fields)
+}

+fn list_gap_edits(
+    bytes: &[u8],
+    base: u32,
+    config: LexerConfig,
+) -> Option<Vec<(Span, &'static str)>> {
+    let fields = list_fields(bytes, config)?;
+    if fields.len() % 2 != 0 {
+        return None;
+    }
+    let mut edits = Vec::new();
+    let mut candidate = Vec::new();
+    let mut cursor = 0;
+    for (index, field) in fields.iter().enumerate() {
+        let separator = if index == 0 { "" } else { " " };
+        edits.push((
+            Span::new(
+                base.checked_add(u32::try_from(cursor).ok()?)?,
+                base.checked_add(u32::try_from(field.start).ok()?)?,
+            ),
+            separator,
+        ));
+        candidate.extend_from_slice(separator.as_bytes());
+        candidate.extend_from_slice(bytes.get(field.clone())?);
+        cursor = field.end;
+    }
+    edits.push((
+        Span::new(
+            base.checked_add(u32::try_from(cursor).ok()?)?,
+            base.checked_add(u32::try_from(bytes.len()).ok()?)?,
+        ),
+        "",
+    ));
+    let reparsed = list_fields(&candidate, config)?;
+    if reparsed.len() != fields.len()
+        || fields
+            .iter()
+            .zip(&reparsed)
+            .any(|(old, new)| bytes.get(old.clone()) != candidate.get(new.clone()))
+    {
+        return None;
+    }
+    Some(edits)
+}

+    #[test]
+    fn case_list_syntax_requires_current_actual_compiler_and_source_owners() {
+        let source = "switch -regexp $x {\n  {^a} {puts 1}\n  default {puts 2}\n}\n";
+        let analysis = crate::analyser::Analyser::new().analyse(source, "tcl8.6");
+        let input = analysis.resolved_input.as_ref().unwrap();
+        let config = input.lexer_config();
+        let segments = crate::segmenter::segment_commands_with_offset_and_config(source, 0, config);
+        let segment = &segments[0];
+        let proof = original_case_list_syntax_gap_compaction(source, &analysis, segment)
+            .expect("same actual compiler and selected source case grammar");
+        assert!(!proof.gap_edits().is_empty());
+        assert!(proof.matches_analysis(source, &analysis));
+        assert!(proof.matches_input(input));
+        assert!(proof.matches_registry(analysis.resolved_registry().unwrap()));
+        assert_eq!(proof.original_operand().span(), segment.argv[3].span);
+        let foreign = crate::analyser::Analyser::new().analyse(source, "tcl8.6");
+        assert!(!proof.matches_analysis(source, &foreign));
+        let stale = source.replace("puts 1", "puts 9");
+        assert!(!proof.matches_analysis(&stale, &analysis));
+        assert!(original_case_list_syntax_gap_compaction(&stale, &analysis, segment).is_none());
+        let mut changed_config = analysis.clone();
+        changed_config.body_lexer_config = None;
+        assert!(!proof.matches_analysis(source, &changed_config));
+        assert!(
+            original_case_list_syntax_gap_compaction(source, &changed_config, segment).is_none()
+        );
+        let mut config = config;
+        config.leading_bom = match config.leading_bom {
+            tcl_lexer::LeadingBom::Content => tcl_lexer::LeadingBom::Skip,
+            tcl_lexer::LeadingBom::Skip => tcl_lexer::LeadingBom::Content,
+        };
+        let changed_input = ResolvedAnalysisInput::new(
+            input.analyser_profile(),
+            input.unit_profile(),
+            input.context_registry(),
+            config,
+        );
+        assert!(!proof.matches_input(&changed_input));
+    }

+    #[test]
+    fn paired_list_gaps_preserve_original_complete_fields() {
+        let config = LexerConfig::default();
+        let original = b"\n  {^a} {puts 1}\n  default {puts 2}\n";
+        let gaps = list_gap_edits(original, 0, config).unwrap();
+        let mut result = original.to_vec();
+        for (span, replacement) in gaps.iter().rev() {
+            result.splice(span.as_range(), replacement.bytes());
+        }
+        assert_eq!(result, b"{^a} {puts 1} default {puts 2}");
+        for source in [
+            b"{a} body extra".as_slice(),
+            b"{broken",
+            b"\"broken",
+            b"{a}junk body",
+        ] {
+            assert!(list_gap_edits(source, 0, config).is_none());
+        }
