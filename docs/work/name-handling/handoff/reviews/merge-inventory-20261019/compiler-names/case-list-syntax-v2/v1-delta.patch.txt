+    fn case_list_syntax_requires_current_actual_compiler_and_source_owners() {
+        let source = "switch -regexp $x {\n  {^a} {puts 1}\n  default {puts 2}\n}\n";
+        let analysis = crate::analyser::Analyser::new().analyse(source, "tcl8.6");
+        let input = analysis.resolved_input.as_ref().unwrap();
+        let config = input.lexer_config();
+        let segments = crate::segmenter::segment_commands_with_offset_and_config(source, 0, config);
+        let segment = &segments[0];
+        let proof = original_case_list_syntax_gap_compaction(source, &analysis, segment)
+            .expect("same actual compiler and selected source case grammar");
+        assert!(!proof.gap_edits().is_empty());
+        assert!(proof.matches_analysis(source, &analysis));
+        assert!(proof.matches_input(input));
+        assert!(proof.matches_registry(analysis.resolved_registry().unwrap()));
+        assert_eq!(proof.original_operand().span(), segment.argv[3].span);
+        let foreign = crate::analyser::Analyser::new().analyse(source, "tcl8.6");
+        assert!(!proof.matches_analysis(source, &foreign));
+        let stale = source.replace("puts 1", "puts 9");
+        assert!(!proof.matches_analysis(&stale, &analysis));
+        assert!(original_case_list_syntax_gap_compaction(&stale, &analysis, segment).is_none());
+        let mut changed_config = analysis.clone();
+        changed_config.body_lexer_config = None;
+        assert!(!proof.matches_analysis(source, &changed_config));
+        assert!(
+            original_case_list_syntax_gap_compaction(source, &changed_config, segment).is_none()
+        );
+        let mut config = config;
+        config.leading_bom = match config.leading_bom {
+            tcl_lexer::LeadingBom::Content => tcl_lexer::LeadingBom::Skip,
+            tcl_lexer::LeadingBom::Skip => tcl_lexer::LeadingBom::Content,
+        };
+        let changed_input = ResolvedAnalysisInput::new(
+            input.analyser_profile(),
+            input.unit_profile(),
+            input.context_registry(),
+            config,
+        );
+        assert!(!proof.matches_input(&changed_input));
+    }
