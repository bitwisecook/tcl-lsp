+    pub fn original_source_analysis_entry(
+        &self,
+        source: &str,
+        registry: &tcl_registry::CommandRegistry,
+        external: Option<&crate::unit_scope::CallSiteEvidence>,
+    ) -> Option<std::sync::Arc<crate::command_binding::SourceAnalysisEntry>> {
+        crate::source_graph::current_analysis(source, self)?;
+        let input = self.resolved_input.as_ref()?;
+        let realm = self.retained_command_realm()?;
+        realm.matches_resolved_analysis_input(input).then_some(())?;
+        let entry = self
+            .optimiser_source_origin
+            .as_ref()?
+            .source_analysis_entry_for_current_analysis(source, registry, input, external)?;
+        if entry
+            .source_point_entry
+            .as_ref()
+            .is_some_and(|point| !point.matches_child_analysis(self))
+        {
+            return None;
+        }
+        Some(entry)
+    }

+    fn source_entry_reborrow_keeps_the_same_inherited_request_without_a_unit_build() {
+        let point = point();
+        let input = point.input();
+        let registry = input.borrowed_context_registry().commands();
+        let source = "proc run {} {return VALUE}";
+        let child = crate::analyser::Analyser::new()
+            .analyse_with_source_point_entry(source, "tcl8.6", &point)
+            .unwrap();
+        let entry = child
+            .original_source_analysis_entry(source, registry, None)
+            .unwrap();
+        assert_eq!(entry.source_point_entry.as_ref(), Some(&point));
+        assert!(entry.native_entry.is_none());
+        assert_eq!(entry.native_compilation, Default::default());
+        assert!(
+            child
+                .original_source_analysis_entry("WRONG", registry, None)
+                .is_none()
+        );
+        let mut unavailable = child.clone();
+        unavailable.analysis_context_unavailable = Some(tcl_registry::model::OverlayMiss {
+            environment: "tcl8.6".to_owned(),
+            overlay: 1,
+        });
+        assert!(
+            unavailable
+                .original_source_analysis_entry(source, registry, None)
+                .is_none()
+        );
+        let mut wrong_config = child.clone();
+        wrong_config.body_lexer_config = None;
+        assert!(
+            wrong_config
+                .original_source_analysis_entry(source, registry, None)
+                .is_none()
+        );
+    }
