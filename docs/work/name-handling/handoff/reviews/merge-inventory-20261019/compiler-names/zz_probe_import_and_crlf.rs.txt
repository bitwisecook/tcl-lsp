    #[test]
    fn zz_probe_import_and_crlf() {
        for (dialect, source) in [
            (
                "tcl8.6",
                "namespace eval ::a {proc p {} {}; namespace export p}\nnamespace eval ::b {namespace import ::a::p; p}\n",
            ),
            (
                "tcl8.6",
                "namespace eval ::a {proc p {} {}; namespace export p}\nnamespace import ::a::p\np\n",
            ),
            (
                "tcl8.6",
                "oo::class create Widget {\r\n    method render {} {\r\n        puts \"widget $label\"\r\n    }\r\n}",
            ),
            (
                "tcl8.6",
                "namespace eval ::a { oo::class create Widget {} }\nnamespace eval ::a { Widget new }\n",
            ),
            (
                "tcl8.6",
                "oo::class create C { variable n; method get {} {return $n} }\n",
            ),
            ("tcl8.6", "proc greet {} { return hi }\nputs [greet]\nputs [greet]\n"),
            (
                "tcl8.6",
                "namespace eval a {namespace export *; proc foo {} {return A}}; namespace eval b {proc foo {} {return B}; namespace import -force ::a::*; puts [foo]}\n",
            ),
            (
                "tcl8.6",
                "namespace eval ::a { oo::class create C {}; C create rex }\nnamespace eval ::a { rex destroy }\n",
            ),
            (
                "tcl8.6",
                "namespace eval lib {oo::class create C {}}; namespace eval app {namespace path ::lib; C new}",
            ),
        ] {
            let result = std::panic::catch_unwind(|| {
                crate::analyser::Analyser::new().analyse(source, dialect)
            });
            eprintln!("ZZ_RESULT {source:?} ok={}", result.is_ok());
            if let Ok(result) = result {
                eprintln!(
                    "ZZ_WORLD {}",
                    result
                        .retained_command_realm()
                        .and_then(crate::realm::CommandBindingRealm::original_completed_command_world)
                        .is_some()
                );
                for row in &result.command_invocations {
                    eprintln!(
                        "ZZ_ROW {} at={} input={} lookup={} reference={}",
                        row.name,
                        row.range.start(),
                        row.original_name_input.is_some(),
                        row.original_lookup.is_some(),
                        format!("{:?}", row.resolved_command_reference)
                            .chars()
                            .take(240)
                            .collect::<String>()
                    );
                }
            }
        }
    }
