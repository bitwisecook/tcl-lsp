// Temporary probe from rust/tcl-compiler/src/analyser/diagnostics/fp/sh.rs (Sol-49)
fn zz_sol49_dict_for_probe() {
    for src in [
        "set d {a 1 b 2}\ndict for {k v} $d { puts \"$k=$v\" }\n",
        "proc f {x} { switch -glob $x { a* { return 1 } }\n puts $y }\n",
        "proc f {x} { while 1 { switch -glob $x {a* {break} default {break}} }; puts $y }\n",
        "set raw [gets stdin]\ninterp eval $child [list puts $raw]\n",
        "set raw [gets stdin]\nset child [interp create]\ninterp eval $child [list $raw]\n",
    ] {
        let registry = tcl_registry::model::ingress::static_context_for("tcl8.6").commands();
        let profile =
            tcl_registry::model::ingress::resolve_environment("tcl8.6").analyser_profile();
        let cu = crate::compilation_unit::CompilationUnit::build_for_profile(
            src, registry, false, profile,
        );
        eprintln!(
            "SRC {src:?} deferred_any={} codes={:?}",
            cu.ir_module.deferred_writes.any,
            codes(src, D)
        );
        let module = &cu.ir_module;
        let mut pending = vec![&module.top_level];
        pending.extend(module.procedures.values().map(|procedure| &procedure.body));
        while let Some(script) = pending.pop() {
            for statement in &script.statements {
                pending.extend(statement.child_scripts());
                let text = format!("{statement:?}");
                eprintln!("  IR {}", text.chars().take(140).collect::<String>());
                let Some(tokens) = script.retained_source_tokens_for_statement(statement) else {
                    continue;
                };
                for (offset, binding) in &tokens.nested_bindings {
                    eprintln!(
                        "    NESTED @{offset} recorded={}",
                        binding.original_recorded_command().is_some()
                    );
                }
            }
        }
        for (name, fu) in cu.procedures.iter().map(|(id, fu)| (format!("{id:?}"), fu)) {
            eprintln!("  FU {name} dynamic_writes={}", fu.dynamic_names.writes);
            let mut ids: Vec<_> = fu.cfg.blocks.keys().collect();
            ids.sort();
            for id in ids {
                let block = &fu.cfg.blocks[id];
                eprintln!(
                    "  FU {name} B{id:?} stmts={} term={}",
                    block.statements.len(),
                    format!("{:?}", block.terminator)
                        .chars()
                        .take(120)
                        .collect::<String>()
                );
            }
        }
    }
}

// Temporary probe from rust/tcl-compiler/src/registry_invocation/source_structure.rs mod tests (Sol-49)
    fn zz_sol49_hover_schema_probe() {
        for source in [
            "regexp {plainword} $line\n",
            "set line x\nregexp {plainword} $line\n",
            "lsearch -exact $list $needle\n",
        ] {
            let analysis = crate::analyser::Analyser::new().analyse(source, "tcl8.6").clone();
            let config = analysis.body_lexer_config.unwrap();
            let offset = u32::try_from(
                source.find("regexp").or_else(|| source.find("lsearch")).unwrap(),
            )
            .unwrap();
            let tail = &source[offset as usize..];
            let segment = crate::segmenter::segment_commands_with_offset_and_config(tail, offset, config)
                .into_iter()
                .next()
                .unwrap();
            eprintln!("PROBE {source:?} offset={offset}");
            let realm = analysis.retained_command_realm().unwrap();
            let binding = realm.invocation_at_source("", offset);
            eprintln!("  recorded={}", binding.original_recorded_command().is_some());
            let image = tcl_lexer::SourceImage::document(source);
            eprintln!(
                "  conditional_metadata={}",
                analysis
                    .original_conditional_registry_metadata_in_source(&image, config, offset)
                    .is_some()
            );
            let input = analysis.resolved_input.as_ref().unwrap();
            let context = input.context_registry();
            eprintln!(
                "  vendor_names={} logical={}",
                analysis.has_original_vendor_source_names(),
                realm.source_bindings_ref().original_logical_source_name_advice_input().is_some()
            );
            let registry = analysis.resolved_registry().unwrap();
            eprintln!(
                "  selected={}",
                selected_registry_words(source, &analysis, &segment, registry).is_some()
            );
            eprintln!(
                "  native_metadata={}",
                native_source_metadata(source, config, realm, &context, &segment, None).is_some()
            );
            let words = source_registry_words(source, &analysis, &segment);
            eprintln!("  words={}", words.is_some());
            if let Some(words) = words {
                eprintln!(
                    "  patterns={:?}",
                    words
                        .with_source_schema(&context, |schema| schema
                            .authored_source_pattern_arguments()
                            .map(|p| p.len()))
                        .flatten()
                );
            }
        }
    }
