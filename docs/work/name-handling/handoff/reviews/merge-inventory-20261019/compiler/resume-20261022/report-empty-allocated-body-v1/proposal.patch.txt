+    // An allocated original declaration records its body value extent, which
+    // can be empty. Only its actual current parser-issued source card admits
+    // that extent; arbitrary zero-width IR and every other role still refuse.
+    fn empty_allocated_body(
+        module: &crate::ir::Module,
+        registry: &CommandRegistry,
+        id: ProcedureId,
+        body: &ReportBodyOwner,
+    ) -> Option<()> {
+        (matches!(
+            body.role,
+            ReportBodyRole::Unit(ProcedureBodyKind::OriginalDeclaration)
+        ) && body.declaration_purpose == Some(ReportDeclarationPurpose::AllocatedSource)
+            && body.declaration.start() == body.declaration.end())
+        .then_some(())?;
+        let positioned = module
+            .retained_source_bindings
+            .as_ref()?
+            .positioned_for_module(module, registry)?;
+        let root = positioned.source_origin()?;
+        let allocation = module.original_declaration_body_units.get(&id)?;
+        let held = module.body_units.get(&id)?;
+        let source = body.source.as_ref()?;
+        let namespace = body.namespace.as_deref()?;
+        (module.procedure_ids.owns(id)
+            && held.body_kind == ProcedureBodyKind::OriginalDeclaration
+            && held.span == body.declaration
+            && held.declaration_site == body.site
+            && Some(held.body_offset) == body.body_offset
+            && held
+                .body
+                .executed_source
+                .as_ref()
+                .is_some_and(|actual| Arc::ptr_eq(actual, source))
+            && held.body.namespace_context.as_deref() == Some(namespace)
+            && body.site.as_ref() == Some(&allocation.site)
+            && allocation.matches_source_declaration_image(&module.source, allocation.site.offset)
+            && Arc::ptr_eq(&allocation.site.source, root)
+            && Arc::ptr_eq(&source.origin, root)
+            && source.text.bytes().is_empty()
+            && matches!(
+                &source.mapping,
+                crate::command_binding::ExecutedScriptMapping::Contiguous { base }
+                    if *base == body.declaration.start()
+            ))
+        .then_some(())?;
+        positioned
+            .original_declaration_body_units()
+            .into_iter()
+            .any(|card| {
+                card.allocation == *allocation
+                    && card.source == **source
+                    && Arc::ptr_eq(&card.source.origin, root)
+                    && card.namespace_key == *namespace
+            })
+            .then_some(())
     }

+    #[test]
+    fn empty_allocated_body_requires_the_current_complete_parser_card() {
+        let source = "proc retained {} {}\nretained EXTRA";
+        let left = Analyser::new().analyse_for_report(source, "tcl8.6");
+        let right = Analyser::new().analyse_for_report(source, "tcl8.6");
+        let capture = left.current_capture().expect("genuine empty original body");
+        assert_eq!(
+            left.compare(&right),
+            super::super::AnalysisReportComparison::Equal
+        );
+        let module = &capture.bodies.prepared.unit().ir_module;
+        let registry = left.analysis.resolved_registry().unwrap();
+        let (&id, _) = module
+            .original_declaration_body_units
+            .iter()
+            .next()
+            .unwrap();
+        let make = || {
+            ReportBodyInventory::procedure(
+                module,
+                id,
+                module.body_units.get(&id).unwrap(),
+                ReportBodyRole::Unit(ProcedureBodyKind::OriginalDeclaration),
+            )
+            .unwrap()
+        };
+        let body = make();
+        assert!(ReportBodyInventory::empty_allocated_body(module, registry, id, &body).is_some());
+        let mut missing = module.clone();
+        missing.original_declaration_body_units.remove(&id);
+        assert!(ReportBodyInventory::empty_allocated_body(&missing, registry, id, &body).is_none());
+        let mut changed = module.clone();
+        changed.body_units.get_mut(&id).unwrap().body_offset += 1;
+        assert!(ReportBodyInventory::empty_allocated_body(&changed, registry, id, &body).is_none());
+        let foreign = &right
+            .current_capture()
+            .unwrap()
+            .bodies
+            .prepared
+            .unit()
+            .ir_module;
+        assert!(ReportBodyInventory::empty_allocated_body(foreign, registry, id, &body).is_none());
+        let mut wrong_role = make();
+        wrong_role.role = ReportBodyRole::Carried(ProcedureBodyKind::OriginalDeclaration);
+        assert!(
+            ReportBodyInventory::empty_allocated_body(module, registry, id, &wrong_role).is_none()
+        );
+        let mut wrong_span = make();
+        wrong_span.declaration = Span::empty(0);
+        assert!(
+            ReportBodyInventory::empty_allocated_body(module, registry, id, &wrong_span).is_none()
+        );
+        let mut bodies = Vec::new();
+        assert!(ReportBodyInventory::push(&mut bodies, module, registry, id, make()).is_some());
+        assert!(ReportBodyInventory::push(&mut bodies, module, registry, id, make()).is_none());
+        assert_eq!(bodies.len(), 1);
     }
