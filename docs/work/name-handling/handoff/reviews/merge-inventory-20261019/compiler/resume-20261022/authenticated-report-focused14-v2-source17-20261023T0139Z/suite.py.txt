def source_view(stage):

        raise SystemExit('STOP: root GO must pin exact plan and source17 manifest')
    approved=go.get('approved_integrated_head')
    if not approved or subprocess.run(['git','merge-base','--is-ancestor',approved,head],check=False).returncode:
        raise SystemExit('STOP: root approved integrated HEAD is not current ancestor')
    after_approval=subprocess.check_output(['git','diff','--name-only',approved,head],text=True).splitlines()

        raise SystemExit('STOP: product changed since approved source17 HEAD')
    completed=Path(go.get('compiler49_final_receipt','/nonexistent/compiler49'))
    if not completed.is_file() or hashlib.sha256(completed.read_bytes()).hexdigest()!=go.get('compiler49_final_sha256'):
        raise SystemExit('STOP: completed compiler49 final receipt missing or unpinned')
    prior=json.loads(completed.read_text());prior_plan=completed.parent/'plan.json';prior_results=completed.parent/'results.json'
    if prior.get('results')!=36 or not prior.get('source_stable') or not prior.get('providers_stable') or not prior_plan.is_file() or hashlib.sha256(prior_plan.read_bytes()).hexdigest()!=plan['compiler49_predecessor_plan_sha256'] or not prior_results.is_file():
        raise SystemExit('STOP: not completed reviewed fixedproduct49 with stable seals')
    prior_rows=json.loads(prior_results.read_text());prior_expected=json.loads(prior_plan.read_text())['required_filters']

    for observed,expected in zip(prior_rows,prior_expected,strict=True):

            raise SystemExit('STOP: completed compiler49 exact selector/count/seal mismatch')

        raise SystemExit('STOP: fixed baseline not ancestor of approved integration')
    allowed={row['path'] for row in plan['integration_paths']}
    for row in plan['integration_paths']:
        path=Path(row['path'])

            raise SystemExit('STOP: exact17 final hash mismatch: '+str(path))
        original=subprocess.run(['git','show',base+':'+row['path']],capture_output=True)
        baseline_hash=hashlib.sha256(original.stdout).hexdigest() if original.returncode==0 else None
        if baseline_hash!=row['before_sha256']:
            raise SystemExit('STOP: source17 original baseline mismatch: '+str(path))
    exclusions=[':(exclude)docs/work/name-handling']+[':(exclude)'+p for p in sorted(allowed)]

        raise SystemExit('STOP: product delta outside exact17')
    for commit in subprocess.check_output(['git','rev-list',f'{base}..{head}'],text=True).splitlines():
        paths=subprocess.check_output(['git','diff-tree','--no-commit-id','--name-only','-r','-m',commit],text=True).splitlines()
        if any(p not in allowed and not p.startswith('docs/work/name-handling/') for p in paths):
            raise SystemExit('STOP: unreviewed source history')
    for commit in subprocess.check_output(['git','rev-list',f'{approved}..{head}'],text=True).splitlines():
        paths=subprocess.check_output(['git','diff-tree','--no-commit-id','--name-only','-r','-m',commit],text=True).splitlines()

    untracked=subprocess.check_output(['git','ls-files','--others','--exclude-standard','-z']).split(b'\0')
    if any(p and p.decode(errors='surrogateescape') not in allowed and not p.decode(errors='surrogateescape').startswith('docs/work/name-handling/') for p in untracked):

def digest(path):
    with path.open('rb') as stream:return hashlib.file_digest(stream,'sha256').hexdigest()
def source_manifest():
    records={}
    raw=subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard','-z'])
    for item in raw.split(b'\0'):
        if not item:continue
        path=Path(os.fsdecode(item))
        if (path.parts[0] in {'rust','runtime','tests','experiments','spec','specs','data','resources','.cargo'}
            or path.suffix in {'.rs','.c','.h','.tcl','.tclspec'}
            or path.name in {'Cargo.toml','Cargo.lock','rust-toolchain.toml','rustfmt.toml','clippy.toml'}):
            records[str(path)]=digest(path) if path.is_file() else None
    return records
