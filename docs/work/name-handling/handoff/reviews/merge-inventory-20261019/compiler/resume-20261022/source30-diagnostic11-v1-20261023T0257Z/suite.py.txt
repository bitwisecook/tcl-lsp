def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def write_json(path, data):
    Path(path).write_text(json.dumps(data, sort_keys=True, indent=2) + '\n')

def source_view(stage, expected_go_sha):
    head = subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()
    base = plan['source_commit']

        raise SystemExit('STOP: root GO changed during run')

    if go.get('plan_sha256') != plan_sha or go.get('source_commit') != base or go.get('source30_manifest_sha256') != plan['source30_manifest_sha256']:
        raise SystemExit('STOP: exact plan/source30 root GO mismatch')

        raise SystemExit('STOP: reviewed runner input manifest mismatch')
    for path, expected in json.loads(RUNNER_MANIFEST.read_text()).get('inputs',{}).items():
        if digest(path) != expected:
            raise SystemExit('STOP: reviewed runner input changed: '+path)
    source30 = BASE / 'source30-manifest.json'
    if digest(source30) != plan['source30_manifest_sha256'] or json.loads(source30.read_text())['paths'] != plan['integration_paths'] or len(plan['integration_paths']) != 30:
        raise SystemExit('STOP: immutable exact30 source manifest mismatch')
    # A launch/start record never establishes completion of the preceding suite.
    completed = Path(go.get('compiler49_final_receipt','/nonexistent/compiler49'))
    launch_path = Path(go.get('compiler49_launcher_receipt','/nonexistent/compiler49-launcher'))
    if not completed.is_file() or digest(completed) != go.get('compiler49_final_sha256') or not launch_path.is_file() or digest(launch_path) != go.get('compiler49_launcher_sha256'):
        raise SystemExit('STOP: completed valid49v7 final/launcher absent or unpinned')
    final = json.loads(completed.read_text())
    prior_plan = completed.parent/'plan.json'
    prior_rows_path = completed.parent/'results.json'
    if not prior_plan.is_file() or digest(prior_plan) != plan['compiler49_predecessor_plan_sha256'] or not prior_rows_path.is_file():
        raise SystemExit('STOP: actual49v7 exact plan/results absent')
    if final.get('results') != 36 or final.get('selected') != 49 or not all(final.get(k) is True for k in ['source_stable','providers_stable','artifact_stable','all36_direct_child_timers_valid']):
        raise SystemExit('STOP: previous49 incomplete or guards invalid')
    prior_rows = json.loads(prior_rows_path.read_text())
    expected_rows = json.loads(prior_plan.read_text())['required_filters']
    if len(prior_rows) != 36 or sum(r['actual_selected'] for r in prior_rows) != 49:
        raise SystemExit('STOP: preceding49 count mismatch')
    for actual, expected in zip(prior_rows,expected_rows,strict=True):
        process=actual.get('process',{})
        if actual['selector'] != expected['selector'] or actual['actual_selected'] != expected['expected_selected'] or actual['started_counts'] != [expected['expected_selected']] or len(actual['summaries']) != 1 or not all(actual.get(k) is True for k in ['source_stable','providers_stable','artifact_stable']) or actual.get('log_at_cap'):
            raise SystemExit('STOP: preceding49 exact roster or seal mismatch')
        if not all(process.get(k) is True for k in ['complete','timer_armed_before_spawn','group_empty_before_parent_reap']) or process.get('stop_reason') is not None:
            raise SystemExit('STOP: preceding49 direct-child lifecycle invalid')
    # Reconcile the actual libtest outcomes, without requiring the tests green.
    expected_overall = 0
    for actual, expected in zip(prior_rows, expected_rows, strict=True):
        if actual.get('warnings') is not False:
            raise SystemExit('STOP: preceding49 warnings are not zero')
        summary = actual['summaries'][0]
        if len(summary) != 4:
            raise SystemExit('STOP: preceding49 summary shape mismatch')
        status, passed_text, failed_text, ignored_text = summary
        passed, failed, ignored = int(passed_text), int(failed_text), int(ignored_text)
        if ignored != 0 or passed + failed != expected['expected_selected'] or passed + failed != actual['actual_selected']:
            raise SystemExit('STOP: preceding49 summary counts/ignored mismatch')
        exit_code = actual.get('exit_code')
        if not isinstance(exit_code, int) or exit_code != actual['process'].get('exit_code') or bool(exit_code) != bool(failed) or (status == 'ok') != (failed == 0):
            raise SystemExit('STOP: preceding49 row exit/status does not match actual outcomes')
        if exit_code:
            expected_overall = 1
    if final.get('overall_exit') != expected_overall:
        raise SystemExit('STOP: preceding49 final exit does not reflect actual rows')
    if list(completed.parent.glob('build-*.warnings.json')):
        raise SystemExit('STOP: preceding49 build warning receipt exists')
    raw_go_path = BASE/'compiler49-v7-root-go.raw.json'
    serialized_go_path = BASE/'compiler49-v7-root-go.receipt.json'

        raise SystemExit('STOP: preceding49 durable launcher not validly completed')

        raise SystemExit('STOP: preceding49 launcher does not match completed unlocked suite')
    if not (completed.parent/'providers-final.json').is_file() or json.loads((completed.parent/'providers-final.json').read_text()) != json.loads((completed.parent/'providers-before.json').read_text()):
        raise SystemExit('STOP: preceding49 actual final provider proof missing or changed')
    approved=go.get('approved_integrated_head')
    if not approved or subprocess.run(['git','merge-base','--is-ancestor',base,approved],check=False).returncode or subprocess.run(['git','merge-base','--is-ancestor',approved,head],check=False).returncode:
        raise SystemExit('STOP: approved22 integration ancestry mismatch')
    allowed={r['path'] for r in plan['integration_paths']}
    for row in plan['integration_paths']:
        path=Path(row['path'])
        if not path.is_file() or digest(path) != row['after_sha256']:
            raise SystemExit('STOP: exact30 final hash mismatch: '+str(path))
        original=subprocess.run(['git','show',base+':'+row['path']],capture_output=True)
        if (hashlib.sha256(original.stdout).hexdigest() if original.returncode == 0 else None) != row['before_sha256']:
            raise SystemExit('STOP: exact30 original3b1 baseline mismatch')
    exclusions=[':(exclude)docs/work/name-handling']+[':(exclude)'+p for p in sorted(allowed)]
    if subprocess.run(['git','diff','--quiet',base,'--','.',*exclusions],check=False).returncode:
        raise SystemExit('STOP: other product differs from3b1')

        for commit in subprocess.check_output(['git','rev-list',f'{anchor}..{head}'],text=True).splitlines():
            paths=subprocess.check_output(['git','diff-tree','--no-commit-id','--name-only','-r','-m',commit],text=True).splitlines()
            if any(p not in allowed_paths and not p.startswith('docs/work/name-handling/') for p in paths):
                raise SystemExit('STOP: unreviewed source history or non-doc descendant')
    for raw in subprocess.check_output(['git','ls-files','--others','--exclude-standard','-z']).split(b'\0'):
        if raw and raw.decode(errors='surrogateescape') not in allowed and not raw.decode(errors='surrogateescape').startswith('docs/work/name-handling/'):
            raise SystemExit('STOP: unreviewed untracked source')
    return head

def source_manifest():
    records = {}
    raw = subprocess.check_output(['git', 'ls-files', '--cached', '--others', '--exclude-standard', '-z'])
    for item in raw.split(b'\0'):
        if not item:
            continue
        path = Path(os.fsdecode(item))
        if (path.parts[0] in {'rust', 'runtime', 'tests', 'experiments', 'spec', 'specs', 'data', 'resources', '.cargo'}
            or path.suffix in {'.rs', '.c', '.h', '.tcl', '.tclspec'}
            or path.name in {'Cargo.toml', 'Cargo.lock', 'rust-toolchain.toml', 'rustfmt.toml', 'clippy.toml'}):
            records[str(path)] = digest(path) if path.is_file() else None
    return records

def disk_snapshot():
    return {p: shutil.disk_usage(p).free for p in ['/workspace', '/tmp']}

def require_disk(stage):
    free = disk_snapshot()
    if any(v < 2 * 1024**3 for v in free.values()):
        raise SystemExit(f'STOP: actual mounted filesystem below 2 GiB at {stage}: {free}')
    return free
