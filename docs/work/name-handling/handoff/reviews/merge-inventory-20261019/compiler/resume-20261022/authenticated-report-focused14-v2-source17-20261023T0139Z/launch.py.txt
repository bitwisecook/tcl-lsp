import datetime, hashlib, json, os, signal, subprocess, sys, time
from pathlib import Path
runner=Path('/tmp/compiler-authenticated-report-focused14-v2/run.sh')
watcher=Path('/tmp/compiler-authenticated-report-focused14-v2/runtime-watchdog.py')
plan=Path('/tmp/compiler-authenticated-report-focused14-v2/plan.json')
pins={'/tmp/compiler-authenticated-report-focused14-v2/run.sh': 'd0ef3f424312afb4c64fbe58341b353c9c19a7d1335b2c689835a79beaa10c7a', '/tmp/compiler-authenticated-report-focused14-v2/suite.py': '99acb6dade6e8ebdbe30040ff8027335666cdf65ea0abf8c8ddc76207108aeb6', '/tmp/compiler-authenticated-report-focused14-v2/guarded-cargo.py': 'a386c6028470e95469b99ce0aad8560f81c48b7a9fe075dd8cec41bfcd8ebba8', '/tmp/compiler-authenticated-report-focused14-v2/runtime-watchdog.py': '80af195fce4690d872580d175342398054d8c973553fbde09287433cbcd9046d', '/tmp/compiler-authenticated-report-focused14-v2/plan.json': '23dd4d5228b13b595cac009f3ec6b931431c2626cf497623b66ce3193ff877a3', '/tmp/compiler-authenticated-report-focused14-v2/source17-manifest.json': '0392557671118a5c916e4fb9cdfa2c919d40c57abf40344c32c2fac41b2cc3de'}
def hashes():return {p:hashlib.sha256(Path(p).read_bytes()).hexdigest() for p in pins}
assert hashes()==pins
base=Path('/tmp/compiler-authenticated-report-focused14-v2-runs'); base.mkdir(exist_ok=True); existing=set(base.iterdir())
started=datetime.datetime.now(datetime.timezone.utc).isoformat()
process=subprocess.Popen(['bash',str(runner),str(plan)],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,bufsize=1)
receipt=None
until=time.monotonic()+15
while time.monotonic()<until:
    new=set(base.iterdir())-existing
    if len(new)==1:
        receipt=new.pop();break
    if len(new)>1 or process.poll() is not None:break
    time.sleep(0.025)
if receipt is None:
    process.send_signal(signal.SIGINT)
    raise SystemExit('STOP: could not identify this suite new receipt before startup')
observer=subprocess.Popen([sys.executable,str(watcher),str(process.pid),str(receipt)])
ready=receipt/'runtime-watchdog-ready.json'
ready_deadline=time.monotonic()+10
while not ready.exists() and observer.poll() is None and time.monotonic()<ready_deadline: time.sleep(0.025)
if not ready.exists():
    process.send_signal(signal.SIGINT)
    observer.send_signal(signal.SIGINT)
    raise SystemExit('STOP: external watchdog failed readiness before tests')
(receipt/'runtime-watchdog-arm.json').write_text(json.dumps({'observer_pid':observer.pid,'attached_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'timer_seconds':180})+'\n')
launch={'started_utc':started,'suite_pid':process.pid,'observer_pid':observer.pid,'pins_before':pins,'receipt':str(receipt),'timer_seconds':180,'runtime_scope':'whole selected executable, excluding compilation','root_authorization':'Prepared compiler N13 focused successor; explicit root GO required; no execution is queued by preparation'}
(receipt/'runtime-watchdog-launch.json').write_text(json.dumps(launch,indent=2)+'\n')
print('WATCHDOG_ATTACHED',json.dumps(launch),flush=True)
for line in process.stdout:print(line,end='',flush=True)
exit_code=process.wait()
try:observer_exit=observer.wait(timeout=5)
except subprocess.TimeoutExpired:
    observer.send_signal(signal.SIGINT)
    observer_exit=observer.wait(timeout=5)
watch=json.loads((receipt/'runtime-watchdog.json').read_text()) if (receipt/'runtime-watchdog.json').exists() else {}
expected=[r['selector'] for r in json.loads(plan.read_text())['required_filters']]
observed=[]
for record in watch.get('records',[]):
    argv=record.get('cargo_argv',[])
    matches=[s for s in expected if s in argv]
    if len(matches)==1: observed.extend(matches)
watchdog_coverage_complete=(observed==expected and len(watch.get('records',[]))==len(expected))
missing_watchdog_selectors=[selector for selector in expected if selector not in observed]
invalid=(observer_exit!=0 or not watch or watch.get('watchdog_failure') or not watchdog_coverage_complete or any(r.get('timed_out') or r.get('acceptance_invalid') or not r.get('runtime_timer_attached_before_test_start') for r in watch.get('records',[])))
launch.update({'finished_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'suite_exit':exit_code,'observer_exit':observer_exit,'pins_after':hashes(),'runtime_acceptance_invalid':bool(invalid),'watchdog_coverage_complete':watchdog_coverage_complete,'watchdog_missing_selectors':missing_watchdog_selectors,'watchdog_observed_executables':len(watch.get('records',[])),'watchdog_expected_executables':len(expected)})
(receipt/'runtime-watchdog-launch.json').write_text(json.dumps(launch,indent=2)+'\n')
print('WATCHDOG_CLOSED',json.dumps(launch),flush=True)
raise SystemExit(exit_code or (2 if invalid or launch['pins_after']!=pins else 0))
