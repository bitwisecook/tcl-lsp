def strict_json(raw):
    def unique(pairs):
        result={}
        for key,value in pairs:
            if key in result:raise ValueError('duplicate JSON key '+key)
            result[key]=value
        return result
    return json.loads(raw,object_pairs_hook=unique)

def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def environment_override_keys(config, environment):
    exact = set(config['refused_environment_exact'])
    return sorted(key for key in environment if key in exact or any(
        key.startswith(pattern['prefix']) and key.endswith(pattern['suffix'])
        for pattern in config['refused_environment_patterns']))

def check_build_environment():
    config = plan['build_environment']
    overrides = environment_override_keys(config, os.environ)
    if overrides:
        raise SystemExit('STOP: unsupported build environment overrides: ' + ','.join(overrides))
    for key, expected in config['required_environment'].items():
        if os.environ.get(key) != expected:
            raise SystemExit('STOP: pinned build environment mismatch: ' + key)
    if not os.environ.get('PATH', '').startswith(config['path_prefix'] + ':'):
        raise SystemExit('STOP: pinned tool PATH prefix missing')
    if digest(config['cargo_path']) != config['cargo_sha256']:
        raise SystemExit('STOP: pinned Cargo binary hash mismatch')
    if digest(config['rustc_path']) != config['rustc_sha256']:
        raise SystemExit('STOP: pinned rustc binary hash mismatch')
    if digest(plan['typecheck_plan_path']) != config['source_typecheck_plan_sha256']:
        raise SystemExit('STOP: source typecheck environment contract changed')
    typecheck_inputs = plan['typecheck_runner_input_manifest']
    if digest(typecheck_inputs['path']) != typecheck_inputs['sha256']:
        raise SystemExit('STOP: source typecheck runner manifest changed')
    for path, expected in strict_json(Path(typecheck_inputs['path']).read_text())['inputs'].items():
        if digest(path) != expected:
            raise SystemExit('STOP: source typecheck runner input changed: ' + path)

def check_legacy_pins():
    for item in plan['predecessor_plans'].values():
        if digest(item['path']) != item['sha256']:
            raise SystemExit('STOP: original diagnostic plan changed')
    for item in plan['legacy_runner_inputs']:
        if digest(item['path']) != item['sha256']:
            raise SystemExit('STOP: original runner/provider input changed: ' + item['path'])
    for item in plan['test_inventory_sources']:
        if digest(ROOT / item['path']) != item['sha256']:
            raise SystemExit('STOP: original exact source inventory changed: ' + item['path'])
    selectors = plan['actual_tool_selectors']
    if digest(selectors['path']) != selectors['sha256']:
        raise SystemExit('STOP: actual tool selector record changed')
    selected = strict_json(Path(selectors['path']).read_text())
    names = selected['overrides']
    if {key: os.environ[key] for key in names if key in os.environ} != names:
        raise SystemExit('STOP: compiler/linker overrides differ from pinned selection')
    for name, expected in selected['programs'].items():
        if shutil.which(name) != expected:
            raise SystemExit('STOP: selected actual tool path changed: ' + name)

def write_json(path, data):
    Path(path).write_text(json.dumps(data, sort_keys=True, indent=2) + '\n')

def source_view(stage, expected_go_sha):
    check_build_environment()
    if digest(GO) != expected_go_sha:
        raise SystemExit('STOP: root GO changed')
    go = strict_json(GO.read_text())
    identity_pin = plan['actual_identity']
    if digest(identity_pin['path']) != identity_pin['sha256']:
        raise SystemExit('STOP: actual immutable identity changed')
    identity = strict_json(Path(identity_pin['path']).read_text())
    if (go.get('plan_sha256') != plan_sha
            or go.get('source_commit') != plan['source_commit']
            or go.get('approved_integrated_head') != plan['approved_integrated_head']
            or go.get('approved_integrated_tree') != plan['approved_integrated_tree']
            or go.get('actual_identity_sha256') != identity_pin['sha256']
            or go.get('source_manifest_sha256') != plan['integrated_manifest_sha256']
            or go.get('typing_receipt_sha256') != plan['typing_receipt']['sha256']):
        raise SystemExit('STOP: root GO actual identity/type/plan binding')
    if not RUNNER_MANIFEST.is_file() or go.get('runner_input_manifest_sha256') != digest(RUNNER_MANIFEST):
        raise SystemExit('STOP: root GO runner manifest pin')
    for path, expected in strict_json(RUNNER_MANIFEST.read_text())['inputs'].items():
        if digest(path) != expected:
            raise SystemExit('STOP: runner input changed '+path)
    if (identity.get('schema') != 'actual-integrated-product-identity-v1'
            or identity.get('approved_integrated_head') != plan['approved_integrated_head']
            or identity.get('approved_integrated_tree') != plan['approved_integrated_tree']
            or identity.get('source_manifest_path') != str(SOURCE46)
            or identity.get('source_manifest_sha256') != plan['integrated_manifest_sha256']
            or identity.get('candidate_manifest_sha256') != plan['candidate_manifest_sha256']
            or identity.get('candidate_patch_sha256') != plan['candidate_patch_sha256']
            or identity.get('path_count') != plan['integrated_path_count']
            or identity.get('typing') is not False or identity.get('runtime_tested') is not False):
        raise SystemExit('STOP: immutable actual input identity contract')
    if digest(SOURCE46) != plan['integrated_manifest_sha256']:
        raise SystemExit('STOP: actual source manifest changed')
    typing = strict_json(Path(plan['typing_receipt']['path']).read_text())
    if (digest(plan['typing_receipt']['path']) != plan['typing_receipt']['sha256']
            or typing.get('status') != 'PASS'
            or typing.get('guard_valid') is not True
            or typing.get('warning_free_json_stderr_stdout') is not True
            or typing.get('source_stable') is not True
            or typing.get('providers_stable') is not True
            or typing.get('cargo_exit_code') != 0
            or typing.get('plan_sha256') != plan['typecheck_plan_sha256']
            or typing.get('approved_product_head') != plan['approved_integrated_head']
            or typing.get('approved_product_tree') != plan['approved_integrated_tree']
            or typing.get('selected_packages') != ['tcl-compiler','tcl-lsp-core','tcl-lsp-db','tcl-lsp-server','tcl-vm','tcl-fuzz','tcl-spectcl','bpf-tcl-ir']):
        raise SystemExit('STOP: exact actual-source eight-package PASS0W typing prerequisite')
    head = subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()
    # Full committed product/map verification is before/after the entire group.
    # Intermediate rows retain file/tool/GO/artifact checks without a 100k scan.
    if stage not in {'startup','final'}:
        return head
    approved = plan['approved_integrated_head']
    tree = subprocess.check_output(['git','rev-parse',approved+'^{tree}'],text=True).strip()
    if tree != plan['approved_integrated_tree']:
        raise SystemExit('STOP: actual committed tree mismatch')
    if subprocess.run(['git','merge-base','--is-ancestor',approved,head],check=False).returncode:
        raise SystemExit('STOP: current HEAD is not approved product or docs descendant')
    for commit in subprocess.check_output(['git','rev-list',f'{approved}..{head}'],text=True).splitlines():
        touched=subprocess.check_output(['git','diff-tree','--no-commit-id','--name-only','-r','-m',commit],text=True).splitlines()
        if any(not path.startswith('docs/work/name-handling/') for path in touched):
            raise SystemExit('STOP: product changed after actual identity')
    if subprocess.run(['git','diff','--quiet',approved,'--','.',':(exclude)docs/work/name-handling'],check=False).returncode:
        raise SystemExit('STOP: dirty product differs from committed actual identity')
    source_map=strict_json(SOURCE46.read_text())
    rows=source_map['paths']; count=plan['integrated_path_count']
    if (source_map.get('exact_path_count')!=count or len(rows)!=count
            or len({row['path'] for row in rows})!=count
            or source_map.get('approved_product_head')!=approved
            or source_map.get('approved_product_tree')!=tree
            or source_map.get('source_commit')!=plan['source_commit']):
        raise SystemExit('STOP: actual source map exact identity/unique count')
    for row in rows:
        rel=row['path']
        if digest(ROOT/rel)!=row['after_sha256']:
            raise SystemExit('STOP: actual canonical source hash '+rel)
        actual=subprocess.check_output(['git','show',approved+':'+rel])
        if hashlib.sha256(actual).hexdigest()!=row['after_sha256']:
            raise SystemExit('STOP: actual committed source hash '+rel)
    untracked=subprocess.check_output(['git','ls-files','--others','--exclude-standard','-z']).split(b'\0')
    if any(path and not path.decode(errors='surrogateescape').startswith('docs/work/name-handling/') for path in untracked):
        raise SystemExit('STOP: untracked product outside docs evidence')
    return head

def source_manifest():
    records = {}
    raw = subprocess.check_output(['git', 'ls-files', '--cached', '--others', '--exclude-standard', '-z'])
    for item in raw.split(b'\0'):
        if not item:
            continue
        path = Path(os.fsdecode(item))
        if (path.parts[0] in {'rust', 'runtime', 'tests', 'experiments', 'spec', 'specs', 'data', 'resources', '.cargo'}
            or path.suffix in {'.rs', '.c', '.h', '.tcl', '.tclspec'}
            or path.name in {'Cargo.toml', 'Cargo.lock', 'rust-toolchain.toml', 'rustfmt.toml', 'clippy.toml'}):
            records[str(path)] = digest(path) if path.is_file() else None
    return records

def disk_snapshot():
    return {p: shutil.disk_usage(p).free for p in ['/workspace', '/tmp']}

def require_disk(stage):
    free = disk_snapshot()
    if any(v < 2 * 1024**3 for v in free.values()):
        raise SystemExit(f'STOP: actual mounted filesystem below 2 GiB at {stage}: {free}')
    return free

    records = []

        write_json(receipt / (label + '.result.json'), {'cargo_exit': result.returncode,
                    'process': process, 'build_finished': finished, 'parse_errors': parse_errors,

        raise SystemExit('STOP: guarded no-run Cargo build did not complete successfully')

            build_warnings.extend(item.get('warnings', []))
