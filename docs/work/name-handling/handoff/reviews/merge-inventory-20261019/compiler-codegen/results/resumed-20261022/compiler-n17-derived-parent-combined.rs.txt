use std::path::PathBuf;

use tcl_compiler::compilation_unit::CompilationUnit;
use tcl_registry::invocation_words::InvocationWordKind;
use tcl_registry::pack_hooks::{self, HookCall, HookWord};
use tcl_registry::value_transfer::{DeclaredEvaluation, SemanticsDeclaration};

use tcl_spectcl::discovery::{Origin, PackFile, Tier};
use tcl_spectcl::hooks;

pub fn run() {
    for (label, dialect, definition, operands) in [
        (
            "expr42",
            "tcl9.0",
            "proc vendor::f {x} {expr {$x * 2}}",
            &["21"][..],
        ),
        (
            "catxy",
            "tcl8.6",
            "proc vendor::f {a b} {string cat $a $b}",
            &["x", "y"][..],
        ),
    ] {
        for world in ["", "world_effects none"] {
            for formals in ["", "x"] {
                run_row(label, dialect, definition, operands, world, Some(formals));
            }
            // Fresh host for the direct call: compiler health/caches cannot
            // repair or poison this independent transport observation.
            run_row(label, dialect, definition, operands, world, None);
        }
    }
}

fn run_row(
    label: &str,
    dialect: &'static str,
    definition: &str,
    operands: &[&str],
    world: &str,
    formals: Option<&str>,
) {
    let source = format!(
        "speclib vendor 2.0 {{ command vendor::f {{ arity {}\n{world}\nruntime_backing tcl-body {{-pack-text {{{definition}}} -evaluate}}\n}} }}",
        operands.len(),
    );
    let packs = tcl_spectcl::pack::load_in_memory(vec![(
        PackFile {
            tier: Tier::Workspace,
            path: PathBuf::from("vm-derived-diagnosis.tclspec"),
            origin: Origin::DotDir,
            dependency_tier: None,
        },
        source,
    )]);
    let key = format!("{label} world={world:?} formals={formals:?}");
    eprintln!("VMD {key} notices={:?}", packs.notices);
    let plan = hooks::plan_for(&packs);
    let host = Rc::new(tclvm_host());
    for programs in plan.packs() {
        eprintln!(
            "VMD {key} install={:?}",
            host.install_pack_hooks(programs.clone())
        );
    }
    pack_hooks::install_host(host.clone());
    let registry = tcl_spectcl::install::registry_for_dialect_with_packs(dialect, &packs);
    let Some(spec) = registry.get("vendor::f") else {
        eprintln!("VMD {key} missing installed spec");
        pack_hooks::clear_host();
        return;
    };
    eprintln!("VMD {key} semantics={:?}", spec.semantics);
    if let SemanticsDeclaration::Declared(semantics) = spec.semantics {
        eprintln!("VMD {key} declaration={:?}", semantics.as_declared());
    }
    if let Some(formals) = formals {
        let script = format!(
            "proc ::p {{{formals}}} {{\n set known [vendor::f {}]\n return $known\n}}\n",
            operands.join(" "),
        );
        let unit = CompilationUnit::build_for_dialect(&script, &registry, false, dialect);
        print_derived_parent_gates(&key, &registry, &unit);
        let mut procedure_names: Vec<_> = unit
            .ir_module
            .procedures
            .values()
            .map(|body| body.qualified_name.as_str())
            .collect();
        procedure_names.sort_unstable();
        eprintln!(
            "VMD {key} procedures={} names={procedure_names:?}",
            procedure_names.len()
        );
        for head in ["vendor::f", "set", "return"] {
            eprintln!(
                "VMD {key} trust {head}: whole={} observed={}",
                unit.command_mutations.trusts(head),
                unit.command_mutations.observed_binding_is_the_builtin(head)
            );
        }
        for (id, body) in &unit.ir_module.procedures {
            if body.qualified_name == "::p" {
                let function = unit
                    .procedures
                    .get(id)
                    .expect("original fixture procedure analysis");
                let known = function
                    .ssa
                    .var_symbol("known")
                    .and_then(|symbol| function.sccp.values.get(&(symbol, 1)));
                let versions: Vec<_> =
                    function
                        .ssa
                        .var_symbol("known")
                        .map_or_else(Vec::new, |symbol| {
                            let mut entries: Vec<_> = function
                                .sccp
                                .values
                                .iter()
                                .filter(|((actual, _), _)| *actual == symbol)
                                .map(|((_, version), value)| (*version, value))
                                .collect();
                            entries.sort_by_key(|(version, _)| *version);
                            entries
                        });
                eprintln!("VMD {key} known-all-versions={versions:?}");
                eprintln!(
                    "VMD {key} known={known:?} routes={:?}",
                    function.sccp.explanations
                );
            }
        }
    } else if let SemanticsDeclaration::Declared(semantics) = spec.semantics {
        if let Some(declared) = semantics.as_declared() {
            if let DeclaredEvaluation::Implementation(implementation) = &declared.evaluation {
                if let Some(slot) = implementation.slot {
                    let words: Vec<_> = operands
                        .iter()
                        .map(|value| HookWord {
                            value,
                            kind: InvocationWordKind::Literal,
                        })
                        .collect();
                    let call = HookCall {
                        words: &words,
                        version: tcl_dialect::TclVersion::from_version_string(
                            dialect.strip_prefix("tcl").expect("probe Tcl profile"),
                        ),
                        in_event_body: false,
                        option: None,
                        constraints: None,
                        dialect: Some(dialect),
                        targets: &[],
                        budget: implementation.capability.budget,
                        depends: implementation.capability.depends,
                    };
                    eprintln!(
                        "VMD {key} available-before={}",
                        pack_hooks::slot_available(slot)
                    );
                    eprintln!("VMD {key} direct={:?}", pack_hooks::dispatch(slot, &call));
                    eprintln!(
                        "VMD {key} available-after={}",
                        pack_hooks::slot_available(slot)
                    );
                }
            }
        }
    }
    eprintln!(
        "VMD {key} errors={:?} crashes={:?}",
        host.error_log(),
        host.crash_records()
    );
    pack_hooks::clear_host();
}

//! OFF-TREE, UNRUN public diagnostic attachment for the unchanged derived run_row.
//!
//! Add `print_derived_parent_gates(&key, &registry, &unit);` immediately after
//! CompilationUnit::build_for_dialect in /tmp/vm-runtime-derived-probe.rs.
//! Fixtures/world/formals and acceptance assertions remain unchanged.
//! This module supplies observations only, never execution/admission authority.

fn print_derived_parent_gates(
    key: &str,
    registry: &tcl_registry::CommandRegistry,
    unit: &tcl_compiler::compilation_unit::CompilationUnit,
) {
    use tcl_compiler::ir::Statement;
    use tcl_compiler::registry_invocation::{
        InvocationMetadataContext, normal_transfer_invocation_with_metadata_context,
    };
    let input = unit.ir_module.source_metadata_input.as_ref();
    let metadata =
        input.and_then(|input| InvocationMetadataContext::for_analysis_input(registry, input));
    eprintln!(
        "N17G {key} retained-input={} exact-analysis-metadata={}",
        input.is_some(),
        metadata.is_some()
    );
    for body in unit.ir_module.procedures.values() {
        if body.qualified_name != "::p" {
            continue;
        }
        for statement in &body.body.statements {
            let Statement::AssignValue { name, tokens, .. } = statement else {
                continue;
            };
            if name != "known" {
                continue;
            }
            let Some(tokens) = tokens.as_ref() else {
                eprintln!("N17G {key} parent-known has no retained tokens");
                continue;
            };
            let normal =
                normal_transfer_invocation_with_metadata_context(registry, metadata, tokens);
            eprintln!(
                "N17G {key} parent normal-transfer={} synthetic={:?}",
                normal.is_some(),
                tokens.synthetic
            );
            let original_site = tokens
                .words()
                .get(2)
                .and_then(|word| word.sole_command_substitution())
                .map(|(_, site)| site);
            eprintln!("N17G {key} parent original-value-sole-substitution={original_site:?}");
            if let Some(binding) = tokens.source_binding.as_ref() {
                let stored = normal.as_ref().and_then(|normal| {
                    normal.stored_value_operand(&binding.variable_context, registry)
                });
                eprintln!(
                    "N17G {key} parent admission={:?} structured-recipe={:?} stored-value={:?} execution={:?} handler={:?} args-normal={} invocation-normal={}",
                    binding.native_compilation_admission_selection(),
                    binding
                        .original_structured_compilation(tokens)
                        .map(|preparation| preparation.recipe()),
                    stored.map(|(spelling, word)| (
                        spelling,
                        word.sole_command_substitution().map(|(_, site)| site)
                    )),
                    binding
                        .proved_execution_target()
                        .map(|target| target.registry_backed),
                    binding
                        .proved_handler_target()
                        .map(|target| target.registry_backed),
                    binding.original_arguments_complete_normally(tokens),
                    binding.original_invocation_completes_normally(tokens),
                );
            } else {
                eprintln!("N17G {key} parent has no retained binding");
            }
            let Some(input) = input else {
                continue;
            };
            // Public lifted_calls drops its private completeness bit. These
            // candidates are observational, never proof of a complete inventory.
            let calls =
                tcl_compiler::word_subst::lifted_calls(Some(tokens), input.lexer_config().nested());
            eprintln!(
                "N17G {key} child public-candidates={} completeness=UNOBSERVABLE checked_lifted_calls=pub(crate) retained-fold=UNOBSERVABLE(pub(crate))",
                calls.len()
            );
            for call in calls {
                let same_site = original_site.is_some_and(|site| site.span == call.span);
                eprintln!(
                    "N17G {key} child span={:?} exact-sole-site={} source-head={:?}",
                    call.span, same_site, call.command
                );
                let Some(child) = call.tokens.as_ref() else {
                    eprintln!("N17G {key} child no retained tokens");
                    continue;
                };
                let Some(binding) = child.source_binding.as_ref() else {
                    eprintln!("N17G {key} child no retained binding");
                    continue;
                };
                let child_normal =
                    normal_transfer_invocation_with_metadata_context(registry, metadata, child);
                eprintln!(
                    "N17G {key} child admission={:?} structured-recipe={:?} normal-transfer={} execution={:?} handler={:?} args-normal={} invocation-normal={} realm={:?}",
                    binding.native_compilation_admission_selection(),
                    binding
                        .original_structured_compilation(child)
                        .map(|preparation| preparation.recipe()),
                    child_normal.is_some(),
                    binding
                        .proved_execution_target()
                        .map(|target| target.registry_backed),
                    binding
                        .proved_handler_target()
                        .map(|target| target.registry_backed),
                    binding.original_arguments_complete_normally(child),
                    binding.original_invocation_completes_normally(child),
                    binding.invocation_realm(),
                );
            }
        }
    }
}
