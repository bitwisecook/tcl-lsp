fn rename_report(tag: &str, src: &str, dialect: &str, at: (u32, u32), new_name: &str) {
    let analysis = analyse(src, dialect);
    let prof = profile(dialect);
    match tcl_lsp_core::rename::rename_with_diagnosis(src, prof, at.0, at.1, new_name, &analysis, None) {
        Ok(edits) => {
            let edited = apply(src, &edits);
            save(&format!("{tag}_{dialect}.tcl"), &edited);
            println!("{tag} {dialect} edits={} -> saved {tag}_{dialect}.tcl", show_edits(src, &edits));
        }
        Err(refusal) => println!("{tag} {dialect} REFUSED {:?}", refusal.reason),
    }
}

    let method = "oo::class create K {\n    method old {} {return OLD}\n    method dest {} {return DEST}\n}\nset k [K new]\nputs \"[$k old] [$k dest]\"\n";

        rename_report("n06proc", proc_class, dialect, pos(proc_class, "proc P", 0, 5), "C");
    }
    // local declaration p -> q while a sibling file declares q
    let local = "proc p {} {return P}\nputs [p]\n";
    let sibling = "proc q {} {return Q}\n";
    let a1 = analyse(local, "tcl8.6");
    let a2 = analyse(sibling, "tcl8.6");
    let index = tcl_lsp_core::workspace_index::WorkspaceIndex::from_documents([("file:///local.tcl", &a1), ("file:///sib.tcl", &a2)]);
    let (l, c) = pos(local, "proc p", 0, 5);
    let core = tcl_lsp_core::rename::rename_with_diagnosis(local, profile("tcl8.6"), l, c, "q", &a1, None);
    println!("N06 local->sibling core={:?} workspace_has_q={}", core.as_ref().map(|e| show_edits(local, e)).map_err(|r| r.reason.clone()), index.workspace_command_exists("::q"));
    let cross = tcl_lsp_core::rename::cross_document_symbol_edits("::p", "q", &index, "file:///local.tcl");
    let ws = tcl_lsp_core::rename::workspace_symbol_rename_edits("::p", "q", &index);
    println!("N06 local->sibling cross_document_edits={} workspace_symbol_rename_edits={:?} allows_lexical_rename_advice={}", cross.len(), ws.map(|v| v.len()), index.allows_lexical_rename_advice());

#[test]
fn seam_extract_inline() {
    // inline proc across a shadowing namespace (N08 shape)
    let src = "proc p {} {return GLOBAL}\nnamespace eval n {\n    proc p {} {return LOCAL}\n    proc c {} {return [p]}\n}\nputs [n::c]\n";
    let analysis = analyse(src, "tcl8.6");
    let registry = tcl_spectcl::bundled::registry_for_dialect("tcl8.6");
    let cursor = src.find("[p]").unwrap() as u32 + 1;
    let r = tcl_lsp_core::refactor::inline_proc(src, cursor, &analysis, &registry);
    match r {
        Some(r) => {
            let mut out = src.to_owned();
            let mut edits = r.edits.clone();
            edits.sort_by(|a, b| b.start.cmp(&a.start));
            for e in edits { out.replace_range(e.start as usize..e.end as usize, &e.new_text); }
            save("seam_inline.tcl", &out);
            println!("SEAM inline_proc title={:?} saved seam_inline.tcl", r.title);
        }
        None => println!("SEAM inline_proc: none offered"),
    }
    // extract variable named `result` when `result` already exists
    let src2 = "proc f {} {\n    set result 1\n    puts [expr {2 + 3}]\n    return $result\n}\nputs [f]\n";
    let analysis = analyse(src2, "tcl8.6");
    let li = tcl_lexer::LineIndex::new_lsp(src2);
    let s = src2.find("[expr").unwrap() as u32;
    let e = s + "[expr {2 + 3}]".len() as u32;
    let r = tcl_lsp_core::refactor::extract_variable(src2, (s, e), "result", &analysis, &li);
    match r {
        Some(r) => {
            let mut out = src2.to_owned();
            let mut edits = r.edits.clone();
            edits.sort_by(|a, b| b.start.cmp(&a.start));
            for e in edits { out.replace_range(e.start as usize..e.end as usize, &e.new_text); }
            save("seam_extract.tcl", &out);
            println!("SEAM extract_variable(result) offered; saved seam_extract.tcl");
        }
        None => println!("SEAM extract_variable(result): none offered"),
    }
}

// ---------------------------------------------------------------- 4.1 scaling

fn cpu_ms<F: FnMut()>(mut f: F) -> f64 {
    let t = std::time::Instant::now();
    f();
    t.elapsed().as_secs_f64() * 1000.0
}

#[test]
fn scaling_rows() {
    for n in [250usize, 500, 1000, 2000] {
        let mut src = String::new();
        for i in 0..n { let _ = writeln!(src, "proc p{i:08} {{}} {{}}"); }
        let a = analyse(&src, "tcl8.6");
        let mut idx = None;
        let mut best_idx = f64::MAX;
        for _ in 0..3 { best_idx = best_idx.min(cpu_ms(|| { idx = Some(tcl_lsp_core::workspace_index::WorkspaceIndex::from_documents([("file:///flat.tcl", &a)])); })); }
        let idx = idx.unwrap();
        let mut best_q = f64::MAX;
        for _ in 0..3 { best_q = best_q.min(cpu_ms(|| { for _ in 0..100 { std::hint::black_box(idx.workspace_command_exists_for_call("::missing", false)); } })); }
        // procs in separate documents too
        let docs: Vec<(String, AnalysisResult)> = (0..n.min(1000)).map(|i| (format!("file:///f{i}.tcl"), analyse(&format!("proc p{i:08} {{}} {{}}\n"), "tcl8.6"))).collect();
        let mut best_multi = f64::MAX;
        for _ in 0..3 { best_multi = best_multi.min(cpu_ms(|| { std::hint::black_box(tcl_lsp_core::workspace_index::WorkspaceIndex::from_documents(docs.iter().map(|(u, a)| (u.as_str(), a)))); })); }
        println!("SCALE flat P={n} index_ms={best_idx:.3} 100_missing_exists_ms={best_q:.3} index_{}docs_ms={best_multi:.3}", docs.len());
    }
    for f in [250usize, 500, 1000, 2000] {
        let edges: Vec<tcl_lsp_core::source_graph::RunEdge> = (1..f).map(|i| tcl_lsp_core::source_graph::RunEdge {
            parent: format!("file:///c{:05}.tcl", i - 1),
            child: format!("file:///c{i:05}.tcl"),
            at: 10,
            enclosing_body: None,
            kind: tcl_lsp_core::source_graph::RunEdgeKind::Source,
        }).collect();
        let mut best = f64::MAX;
        for _ in 0..3 { best = best.min(cpu_ms(|| { std::hint::black_box(tcl_lsp_core::source_graph::RunOrder::build(&edges)); })); }
        println!("SCALE source-chain F={f} build_ms={best:.3}");
    }
    for n in [125usize, 250, 500] {
        let mut src = String::new();
        for i in 0..n { let _ = write!(src, "proc p{i:08} {{}} {{}}\np{i:08}\n"); }
        let a = analyse(&src, "tcl8.6");
        let mut best = f64::MAX;
        let mut count = 0;
        for _ in 0..3 { best = best.min(cpu_ms(|| { count = tcl_lsp_core::code_lens::code_lenses(&src, profile("tcl8.6"), Some(&a), None, "file:///lens.tcl").len(); })); }
        println!("SCALE code-lens P={n} ms={best:.3} lenses={count}");
    }
}
