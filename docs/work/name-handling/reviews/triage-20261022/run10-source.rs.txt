fn offset(source: &str, line: u32, ch: u32) -> usize {
    let mut off = 0usize;
    for (i, l) in source.split_inclusive('\n').enumerate() {
        if i as u32 == line {
            let mut units = 0u32;
            for (b, c) in l.char_indices() {
                if units >= ch {
                    return off + b;
                }
                units += c.len_utf16() as u32;
            }
            return off + l.len();
        }
        off += l.len();
    }
    off
}

fn apply(source: &str, edits: &[TextEdit]) -> String {
    let mut spans: Vec<(usize, usize, &str)> = edits
        .iter()
        .map(|e| {
            (
                offset(source, e.range.start_line, e.range.start_character),
                offset(source, e.range.end_line, e.range.end_character),
                e.new_text.as_str(),
            )
        })
        .collect();
    spans.sort_by(|a, b| b.0.cmp(&a.0));
    let mut out = source.to_owned();
    for (s, e, t) in spans {
        out.replace_range(s..e, t);
    }
    out
}

fn show_edits(source: &str, edits: &[TextEdit]) -> String {
    let mut s = String::new();
    for e in edits {
        let a = offset(source, e.range.start_line, e.range.start_character);
        let b = offset(source, e.range.end_line, e.range.end_character);
        let _ = write!(
            s,
            "[L{}:{} {:?}->{:?}] ",
            e.range.start_line, e.range.start_character, &source[a..b], e.new_text
        );
    }
    s
}

/// Locate the `nth` (0-based) occurrence of `needle` and return (line, char)
/// of its first byte plus `delta` bytes.
fn pos(source: &str, needle: &str, nth: usize, delta: usize) -> (u32, u32) {
    let mut from = 0;
    let mut found = 0;
    loop {
        let i = source[from..].find(needle).expect("needle") + from;
        if found == nth {
            let at = i + delta;
            let line = source[..at].matches('\n').count() as u32;
            let ls = source[..at].rfind('\n').map_or(0, |p| p + 1);
            let ch = source[ls..at].encode_utf16().count() as u32;
            return (line, ch);
        }
        found += 1;
        from = i + 1;
    }
}

fn n17_template(dsl: &str, helper: bool, braced: bool) -> String {
    let reference = if braced { "$arr(${a}:b)" } else { "$arr($a:b)" };
    if helper {
        format!(
            "speclib edge {dsl} {{\n    proc pick {{}} {{\n        set a x\n        set a:b BAD\n        set arr(x:b) good\n        set arr(BAD) bad\n        return {reference}\n    }}\n    command [pick] {{arity 0}}\n}}\n"
        )
    } else {
        format!(
            "speclib edge {dsl} {{\n    set a x\n    set a:b BAD\n    set arr(x:b) good\n    set arr(BAD) bad\n    command {reference} {{arity 0}}\n}}\n"
        )
    }
}

#[test]
fn n17_template_registration() {
    for dsl in ["1", "2.0"] {
        for helper in [false, true] {
            for braced in [false, true] {
                for fast in [true, false] {
                    let src = n17_template(dsl, helper, braced);
                    let options = EvalOptions {
                        static_fast_path: fast,
                        ..EvalOptions::default()
                    };
                    let pack = evaluate_pack_with(&src, &options);
                    let names: Vec<String> =
                        pack.commands.iter().map(|c| c.spec.name.to_string()).collect();
                    println!(
                        "N17T dsl={dsl} helper={helper} braced={braced} fast={fast} -> commands={names:?} notices={} load_error={:?}",
                        pack.notices.len(),
                        pack.load_error
                    );
                }
            }
        }
    }
}

fn n17_hook_source(dsl: &str, folded: &str) -> String {
    format!(
        "\nspeclib edge {dsl} {{\n    command edge::test {{\n        arity 1\n        arg 0 -role Value\n        const_fold -inputs {{words}} {{words ctx}} {{\n            set a x\n            set a:b BAD\n            set arr(x:b) good\n            set arr(BAD) bad\n            fold {folded}\n        }}\n    }}\n}}\n"
    )
}

#[test]
fn n17_live_hook() {
    for (dsl, folded) in [("1", "$arr($a:b)"), ("2.0", "$arr($a:b)"), ("2.0", "$arr(${a}:b)"), ("2.0", "good"), ("2.0", "[string length [lindex $words 0]]")] {
        let src = n17_hook_source(dsl, folded);
        let dir = scratch().join(format!("n17pack-{dsl}-{}", folded.len()));
        let _ = std::fs::remove_dir_all(&dir);
        std::fs::create_dir_all(&dir).unwrap();
        let path = dir.join("edge.tclspec");
        std::fs::write(&path, &src).unwrap();
        let packs = tcl_spectcl::pack::load(&[PackFile {
            tier: Tier::Workspace,
            path,
            origin: Origin::DotDir,
            dependency_tier: None,
        }]);
        let plan = hooks::plan_for(&packs);
        let host = Rc::new(tclvm_host());
        let mut declined = Vec::new();
        for programs in plan.packs() {
            for inst in host.install_pack_hooks(programs.clone()) {
                declined.push(format!("{:?}", inst.declined));
            }
        }
        pack_hooks::install_host(host);
        let registry = tcl_spectcl::install::registry_for_dialect_with_packs("tcl8.6", &packs);
        let opts = tcl_compiler::optimiser::manager::optimise_raw(
            "proc ::f {} {\n    set y [edge::test abcde]\n}\n",
            &registry,
            None,
        );
        let o129: Vec<String> = opts
            .iter()
            .filter(|o| o.code.as_str() == "O129")
            .map(|o| o.replacement.clone())
            .collect();
        println!(
            "N17H dsl={dsl} fold={folded} notices={} plan_empty={} declined={declined:?} O129={o129:?}",
            packs.notices.len(),
            plan.is_empty()
        );
        pack_hooks::clear_host();
    }
}

// ================================================================ part 2

fn server_rename(src: &str, dialect: &str, at: (u32, u32), new_name: &str, analysis: &AnalysisResult) -> Result<Vec<TextEdit>, String> {
    let mut index = tcl_lsp_core::workspace_index::WorkspaceIndex::new();
    index.add_document("file:///main.tcl", analysis);
    let exports = index.export_snapshot();
    let registry = tcl_registry::model::ingress::static_context_for(dialect).commands();
    tcl_lsp_core::rename::rename_in_program(
        src, profile(dialect), at.0, at.1, new_name, analysis,
        tcl_lsp_core::definition::CallResolution {
            registry: Some(registry),
            program: Some(tcl_lsp_core::definition::ProgramExports { uri: "file:///main.tcl", oracle: exports.as_ref() }),
        },
    ).map_err(|r| r.reason)
}

fn rename_report(tag: &str, src: &str, dialect: &str, at: (u32, u32), new_name: &str) {
    let analysis = analyse(src, dialect);
    println!("WORLD {tag} {dialect} completed={}", analysis.original_completed_command_world().is_some());
    match server_rename(src, dialect, at, new_name, &analysis) {
        Ok(edits) => {
            let edited = apply(src, &edits);
            save(&format!("{tag}_{dialect}.tcl"), &edited);
            println!("{tag} {dialect} edits={} -> saved {tag}_{dialect}.tcl", show_edits(src, &edits));
        }
        Err(reason) => println!("{tag} {dialect} REFUSED {reason:?}"),
    }
}

        let noputs = "proc foo {} {return A}\nfoo\n";
        rename_report("r_noputs", noputs, dialect, pos(noputs, "proc foo", 0, 5), "baz");
        let setonly = "proc foo {} {return A}\nset x [foo]\n";
        rename_report("r_setonly", setonly, dialect, pos(setonly, "proc foo", 0, 5), "baz");
        let wild = "namespace eval Foo {\n    proc bar {} {return 1}\n    namespace export bar\n}\nnamespace import ::Foo::*\nputs [bar]\n";
        let exact = "namespace eval Foo {\n    proc bar {} {return 1}\n    namespace export *\n}\nnamespace import ::Foo::bar\nputs [bar]\n";
        rename_report("r_n04wild", wild, dialect, pos(wild, "proc bar", 0, 5), "baz");
        rename_report("r_n04exact", exact, dialect, pos(exact, "proc bar", 0, 5), "baz");

        rename_report("r_n03", caf, dialect, pos(caf, "$café", 0, 2), "zz");
    }
    for dialect in ["tcl8.6", "tcl9.0", "tcl9.1"] {
        let class = "oo::class create A {method m {} {return A}}\noo::class create B {method m {} {return B}}\nputs [[A new] m]\nputs [[B new] m]\n";
        let method = "oo::class create K {\n    method old {} {return OLD}\n    method dest {} {return DEST}\n}\nset k [K new]\nputs \"[$k old] [$k dest]\"\n";

}

#[test]
fn seam_nested_passthrough_irules() {
    use tcl_compiler::compilation_unit::CompilationUnit;
    let cases = [
        ("f5-irules", "proc run {body} {uplevel 1 $body}\nwhen HTTP_REQUEST {\n    set x abc\n    if {$x contains \"a\"} {log local0. direct}\n    call run {if {$x contains \"a\"} {log local0. inl}}\n}\n"),
        ("f5-irules", "proc run {body} {uplevel 1 $body}\nproc t {} {\n    set x abc\n    if {$x contains \"a\"} {log local0. direct}\n    run {if {$x contains \"a\"} {log local0. inl}}\n}\n"),
        ("tcl8.6", "proc run {body} {uplevel 1 $body}\nproc t {} {\n    set x abc\n    if {$x eq \"a\"} {puts direct}\n    run {if {$x eq \"a\"} {puts inl}}\n}\n"),
    ];
    for (dialect, s) in cases {
        let registry = tcl_spectcl::bundled::registry_for_dialect(dialect);
        let cu = CompilationUnit::build_for_profile(s, &registry, false, profile(dialect));
        let mut roots = vec![&cu.ir_module.top_level];
        roots.extend(cu.ir_module.procedures.iter().map(|(_, p)| &p.body));
        roots.extend(cu.ir_module.body_units.iter().map(|(_, p)| &p.body));
        for (root, script) in roots.into_iter().enumerate() {
            let mut conditions = Vec::new();
            let mut blocks = 0usize;
            let mut native_calls = 0usize;
            let mut calls = Vec::new();
            tcl_compiler::ir::for_each_statement(script, &mut |stmt| {
                use tcl_compiler::ir::Statement;
                match stmt {
                    Statement::If { clauses, .. } => {
                        for clause in clauses { conditions.push(format!("{:?}", clause.condition)); }
                    }
                    Statement::Block { .. } => blocks += 1,
                    Statement::NativeCall { .. } => native_calls += 1,
                    Statement::Call { command, .. } | Statement::Barrier { command, .. } => calls.push(command.clone()),
                    _ => {}
                }
            });
            println!("SEAM nested {dialect} root={root} conditions={conditions:?} blocks={blocks} native_calls={native_calls} calls={calls:?}");
        }
    }
}

        let mut best = f64::MAX;
        let mut count = 0;
        for _ in 0..runs { best = best.min(cpu_ms(|| { count = tcl_lsp_core::code_lens::code_lenses(&src, profile("tcl8.6"), Some(&a), None, "file:///lens.tcl").len(); })); }
        println!("SCALE code-lens P={n} analyse_ms={analyse_ms:.1} ms={best:.3} lenses={count}");
    }
}


#[test]
fn n17_live_hook_effect_control() {
    for dsl in ["1", "2.0"] {
        for pure in [false, true] {
            for braced in [false, true] {
                let folded = if braced { "$arr(${a}:b)" } else { "$arr($a:b)" };
                let authored = n17_hook_source(dsl, folded);
                let authored = if pure {
                    authored.replacen("        arity 1\n", "        arity 1\n        world_effects none\n", 1)
                } else { authored };
                let tag = format!("n17_effect_{dsl}_{pure}_{braced}");
                let dir = scratch().join(&tag);
                std::fs::create_dir_all(&dir).unwrap();
                let path = dir.join("edge.tclspec");
                std::fs::write(&path, &authored).unwrap();
                let packs = tcl_spectcl::pack::load(&[PackFile { tier: Tier::Workspace, path, origin: Origin::DotDir, dependency_tier: None }]);
                let plan = hooks::plan_for(&packs);
                let host = Rc::new(tclvm_host());
                let mut declined = Vec::new();
                for programs in plan.packs() {
                    for installed in host.install_pack_hooks(programs.clone()) {
                        declined.push(format!("{:?}", installed.declined));
                    }
                }
                pack_hooks::install_host(host);
                let registry = tcl_spectcl::install::registry_for_dialect_with_packs("tcl8.6", &packs);
                let direct = registry.get("edge::test").and_then(|s| s.run_const_fold(&["abcde"], Some(tcl_dialect::TclVersion::V8_6)));
                let src = "proc ::f {} {\n    set y [edge::test abcde]\n    return $y\n}\n";
                let opts = tcl_compiler::optimiser::optimise_raw(src, &registry, Some("tcl8.6"));
                let folds: Vec<_> = opts.iter().filter(|o| o.code.as_str() == "O129").map(|o| o.replacement.clone()).collect();
                println!("N17_EFFECT dsl={dsl} pure={pure} braced={braced} notices={:?} declined={declined:?} direct={direct:?} O129={folds:?}", packs.notices);
                let edited = tcl_compiler::optimiser::apply_optimisations(src, &opts);
                let driver = "namespace eval edge {}\nproc edge::test {input} {set a x; set a:b BAD; set arr(x:b) good; set arr(BAD) bad; return $arr($a:b)}\n";
                save(&format!("{tag}.tcl"), &format!("{driver}{edited}puts [f]\n"));
                pack_hooks::clear_host();
            }
        }
    }
}

#[test]
fn n06_sibling_profile_matrix() {
    let local = "proc p {} {return P}\nputs [p]\n";
    let sibling = "proc q {} {return Q}\n";
    for dialect in ["tcl8.4", "tcl8.5", "tcl8.6", "tcl9.0", "tcl9.1", "jim"] {
        let a1 = analyse(local, dialect);
        let a2 = analyse(sibling, dialect);
        let index = tcl_lsp_core::workspace_index::WorkspaceIndex::from_documents([("file:///main.tcl", &a1), ("file:///sibling.tcl", &a2)]);
        let exports = index.export_snapshot();
        for (origin, at) in [("declaration", pos(local, "proc p", 0, 5)), ("call", pos(local, "[p]", 0, 1))] {
            let result = tcl_lsp_core::rename::rename_in_program(local, profile(dialect), at.0, at.1, "q", &a1,
                tcl_lsp_core::definition::CallResolution {
                    registry: Some(tcl_registry::model::ingress::static_context_for(dialect).commands()),
                    program: Some(tcl_lsp_core::definition::ProgramExports { uri: "file:///main.tcl", oracle: exports.as_ref() }),
                });
            match result {
                Ok(edits) => {
                    println!("N06_SIBLING {dialect} {origin} edits={}", show_edits(local, &edits));
                    save(&format!("n06_sibling_{dialect}_{origin}.tcl"), &format!("{sibling}{}puts [q]\n", apply(local, &edits)));
                }
                Err(reason) => println!("N06_SIBLING {dialect} {origin} REFUSED {:?}", reason.reason),
            }
        }
    }
}
