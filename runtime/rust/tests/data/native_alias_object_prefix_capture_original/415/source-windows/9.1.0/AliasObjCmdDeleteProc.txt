2417:  *	interpreter.
2418:  *
2419:  *----------------------------------------------------------------------
2420:  */

2422: static void
2423: AliasObjCmdDeleteProc(
2424:     void *clientData)		/* The alias record for this alias. */
2425: {
2426:     Alias *aliasPtr = (Alias *) clientData;
2427:     Target *targetPtr;
2428:     Tcl_Size i;
2429:     Tcl_Obj **objv;
2430: 
2431:     Tcl_DecrRefCount(aliasPtr->token);
2432:     objv = &aliasPtr->objPtr;
2433:     for (i = 0; i < aliasPtr->objc; i++) {
2434: 	Tcl_DecrRefCount(objv[i]);
2435:     }
2436:     Tcl_DeleteHashEntry(aliasPtr->aliasEntryPtr);
2437: 
2438:     /*
2439:      * Splice the target record out of the target interpreter's parent list.
2440:      */
2441: 
2442:     targetPtr = aliasPtr->targetPtr;
2443:     if (targetPtr->prevPtr != NULL) {
2444: 	targetPtr->prevPtr->nextPtr = targetPtr->nextPtr;
2445:     } else {
2446: 	Parent *parentPtr = &INTERP_INFO(aliasPtr->targetInterp)->parent;
2447: 
2448: 	parentPtr->targetsPtr = targetPtr->nextPtr;
2449:     }
2450:     if (targetPtr->nextPtr != NULL) {
2451: 	targetPtr->nextPtr->prevPtr = targetPtr->prevPtr;
2452:     }
2453: 
2454:     Tcl_Free(targetPtr);
2455:     Tcl_Free(aliasPtr);
2456: }
