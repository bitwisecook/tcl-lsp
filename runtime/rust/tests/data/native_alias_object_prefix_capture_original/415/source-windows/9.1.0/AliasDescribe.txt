2192: static int
2193: AliasNRCmd(
2194:     void *clientData,		/* Alias record. */
2195:     Tcl_Interp *interp,		/* Current interpreter. */
2196:     Tcl_Size objc,		/* Number of arguments. */
2197:     Tcl_Obj *const *objv)	/* Argument vector. */

2200:     Tcl_Size prefc, cmdc, i;
2201:     Tcl_Obj **prefv, **cmdv;
2202:     Tcl_Obj *listPtr;

2204:     int flags = TCL_EVAL_INVOKE;
2205: 
2206:     /*
2207:      * Append the arguments to the command prefix and invoke the command in
2208:      * the target interp's global namespace.
2209:      */
2210: 
2211:     prefc = aliasPtr->objc;
2212:     prefv = &aliasPtr->objPtr;
2213:     cmdc = prefc + objc - 1;
2214: 
2215:     /* TODO - encapsulate this into tclListObj.c */
2216:     listPtr = Tcl_NewListObj(cmdc, NULL);
2217:     ListObjGetRep(listPtr, &listRep);
2218:     cmdv = ListRepElementsBase(&listRep);
2219:     listRep.storePtr->numUsed = cmdc;
2220:     if (listRep.spanPtr) {
2221: 	listRep.spanPtr->spanStart = listRep.storePtr->firstUsed;
2222: 	listRep.spanPtr->spanLength = listRep.storePtr->numUsed;
