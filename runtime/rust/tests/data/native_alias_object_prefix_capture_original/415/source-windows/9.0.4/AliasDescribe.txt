1775: /*
1776:  *----------------------------------------------------------------------
1777:  *
1778:  * TclAliasObjCmd, TclLocalAliasObjCmd --
1779:  *
1780:  *	This is the function that services invocations of aliases in a child
1781:  *	interpreter. One such command exists for each alias. When invoked,
1782:  *	this function redirects the invocation to the target command in the
1783:  *	parent interpreter as designated by the Alias record associated with
1784:  *	this command.
1785:  *
1786:  *	TclLocalAliasObjCmd is a stripped down version used when the source
1787:  *	and target interpreters of the alias are the same. That lets a number
1788:  *	of safety precautions be avoided: the state is much more precisely
1789:  *	known.
1790:  *
1791:  * Results:
1792:  *	A standard Tcl result.
1793:  *
1794:  * Side effects:
1795:  *	Causes forwarding of the invocation; all possible side effects may
1796:  *	occur as a result of invoking the command to which the invocation is
1797:  *	forwarded.
1798:  *
1799:  *----------------------------------------------------------------------
1800:  */
1801: 
1802: static int
1803: AliasNRCmd(
1804:     void *clientData,		/* Alias record. */
1805:     Tcl_Interp *interp,		/* Current interpreter. */
1806:     int objc,			/* Number of arguments. */
1807:     Tcl_Obj *const objv[])	/* Argument vector. */
1808: {
1809:     Alias *aliasPtr = (Alias *) clientData;
1810:     Tcl_Size prefc, cmdc, i;
1811:     Tcl_Obj **prefv, **cmdv;
1812:     Tcl_Obj *listPtr;
1813:     ListRep listRep;
1814:     int flags = TCL_EVAL_INVOKE;
1815: 
1816:     /*
1817:      * Append the arguments to the command prefix and invoke the command in
1818:      * the target interp's global namespace.
1819:      */
1820: 
1821:     prefc = aliasPtr->objc;
1822:     prefv = &aliasPtr->objPtr;
1823:     cmdc = prefc + objc - 1;
1824: 
1825:     /* TODO - encapsulate this into tclListObj.c */
1826:     listPtr = Tcl_NewListObj(cmdc, NULL);
1827:     ListObjGetRep(listPtr, &listRep);
1828:     cmdv = ListRepElementsBase(&listRep);
1829:     listRep.storePtr->numUsed = cmdc;
1830:     if (listRep.spanPtr) {
1831: 	listRep.spanPtr->spanStart = listRep.storePtr->firstUsed;
1832: 	listRep.spanPtr->spanLength = listRep.storePtr->numUsed;
1833:     }
1834: 
1835:     prefv = &aliasPtr->objPtr;
1836:     memcpy(cmdv, prefv, prefc * sizeof(Tcl_Obj *));
1837:     memcpy(cmdv+prefc, objv+1, (objc-1) * sizeof(Tcl_Obj *));
1838: 
1839:     for (i=0; i<cmdc; i++) {
1840: 	Tcl_IncrRefCount(cmdv[i]);
1841:     }
1842: 
1843:     /*
1844:      * Use the ensemble rewriting machinery to ensure correct error messages:
1845:      * only the source command should show, not the full target prefix.
1846:      */
1847: 
1848:     if (TclInitRewriteEnsemble(interp, 1, prefc, objv)) {
1849: 	TclNRAddCallback(interp, TclClearRootEnsemble, NULL, NULL, NULL, NULL);
1850:     }
1851:     TclSkipTailcall(interp);
1852:     return Tcl_NREvalObj(interp, listPtr, flags);
1853: }
