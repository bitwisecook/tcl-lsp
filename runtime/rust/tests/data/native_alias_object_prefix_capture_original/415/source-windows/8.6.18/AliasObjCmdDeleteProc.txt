1954:  *	interpreter.
1955:  *
1956:  *----------------------------------------------------------------------
1957:  */
1958: 
1959: static void
1960: AliasObjCmdDeleteProc(
1961:     ClientData clientData)	/* The alias record for this alias. */
1962: {
1963:     Alias *aliasPtr = (Alias *)clientData;
1964:     Target *targetPtr;
1965:     int i;
1966:     Tcl_Obj **objv;
1967: 
1968:     Tcl_DecrRefCount(aliasPtr->token);
1969:     objv = &aliasPtr->objPtr;
1970:     for (i = 0; i < aliasPtr->objc; i++) {
1971: 	Tcl_DecrRefCount(objv[i]);
1972:     }
1973:     Tcl_DeleteHashEntry(aliasPtr->aliasEntryPtr);
1974: 
1975:     /*
1976:      * Splice the target record out of the target interpreter's parent list.
1977:      */
1978: 
1979:     targetPtr = aliasPtr->targetPtr;
1980:     if (targetPtr->prevPtr != NULL) {
1981: 	targetPtr->prevPtr->nextPtr = targetPtr->nextPtr;
1982:     } else {
1983: 	Parent *parentPtr = &((InterpInfo *) ((Interp *)
1984: 		aliasPtr->targetInterp)->interpInfo)->parent;
1985: 
1986: 	parentPtr->targetsPtr = targetPtr->nextPtr;
1987:     }
1988:     if (targetPtr->nextPtr != NULL) {
1989: 	targetPtr->nextPtr->prevPtr = targetPtr->prevPtr;
1990:     }
1991: 
1992:     ckfree(targetPtr);
1993:     ckfree(aliasPtr);
1994: }
