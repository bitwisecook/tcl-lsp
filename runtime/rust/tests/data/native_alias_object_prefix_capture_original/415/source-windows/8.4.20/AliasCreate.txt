1159:  *	for the slave interpreter.

1164: static int
1165: AliasCreate(interp, slaveInterp, masterInterp, namePtr, targetNamePtr,
1166: 	objc, objv)
1167:     Tcl_Interp *interp;		/* Interp for error reporting. */
1168:     Tcl_Interp *slaveInterp;	/* Interp where alias cmd will live or from
1169: 				 * which alias will be deleted. */
1170:     Tcl_Interp *masterInterp;	/* Interp in which target command will be
1171: 				 * invoked. */
1172:     Tcl_Obj *namePtr;		/* Name of alias cmd. */
1173:     Tcl_Obj *targetNamePtr;	/* Name of target cmd. */
1174:     int objc;			/* Additional arguments to store */
1175:     Tcl_Obj *CONST objv[];	/* with alias. */
1176: {
1177:     Alias *aliasPtr;
1178:     Tcl_HashEntry *hPtr;
1179:     Target *targetPtr;
1180:     Slave *slavePtr;
1181:     Master *masterPtr;
1182:     Tcl_Obj **prefv;
1183:     int new, i;
1184: 
1185:     aliasPtr = (Alias *) ckalloc((unsigned) (sizeof(Alias) 
1186:             + objc * sizeof(Tcl_Obj *)));
1187:     aliasPtr->namePtr		= namePtr;
1188:     Tcl_IncrRefCount(aliasPtr->namePtr);
1189:     aliasPtr->targetInterp	= masterInterp;
1190: 
1191:     aliasPtr->objc = objc + 1;
1192:     prefv = &aliasPtr->objPtr;
1193: 
1194:     *prefv = targetNamePtr;
1195:     Tcl_IncrRefCount(targetNamePtr);
1196:     for (i = 0; i < objc; i++) {
1197: 	*(++prefv) = objv[i];
1198: 	Tcl_IncrRefCount(objv[i]);
1199:     }
1200: 
1201:     Tcl_Preserve(slaveInterp);
1202:     Tcl_Preserve(masterInterp);
1203: 
1204:     aliasPtr->slaveCmd = Tcl_CreateObjCommand(slaveInterp,
1205: 	    Tcl_GetString(namePtr), AliasObjCmd, (ClientData) aliasPtr,
1206: 	    AliasObjCmdDeleteProc);
1207: 
1208:     if (TclPreventAliasLoop(interp, slaveInterp,
1209: 	    aliasPtr->slaveCmd) != TCL_OK) {
1210: 	/*
1211: 	 * Found an alias loop!	 The last call to Tcl_CreateObjCommand made
1212: 	 * the alias point to itself.  Delete the command and its alias
1213: 	 * record.  Be careful to wipe out its client data first, so the
1214: 	 * command doesn't try to delete itself.
1215: 	 */
1216: 
1217: 	Command *cmdPtr;
1218: 	
1219: 	Tcl_DecrRefCount(aliasPtr->namePtr);
1220: 	Tcl_DecrRefCount(targetNamePtr);
1221: 	for (i = 0; i < objc; i++) {
1222: 	    Tcl_DecrRefCount(objv[i]);
1223: 	}
1224: 	
1225: 	cmdPtr = (Command *) aliasPtr->slaveCmd;
1226: 	cmdPtr->clientData = NULL;
1227: 	cmdPtr->deleteProc = NULL;
1228: 	cmdPtr->deleteData = NULL;
1229: 	Tcl_DeleteCommandFromToken(slaveInterp, aliasPtr->slaveCmd);
1230: 
1231: 	ckfree((char *) aliasPtr);
1232: 
1233: 	/*
1234: 	 * The result was already set by TclPreventAliasLoop.
1235: 	 */
1236: 
1237: 	Tcl_Release(slaveInterp);
1238: 	Tcl_Release(masterInterp);
1239: 	return TCL_ERROR;
1240:     }
1241: 
1242:     /*
1243:      * Make an entry in the alias table. If it already exists delete
1244:      * the alias command. Then retry.
1245:      */
1246: 
1247:     slavePtr = &((InterpInfo *) ((Interp *) slaveInterp)->interpInfo)->slave;
1248:     while (1) {
1249: 	Alias *oldAliasPtr;
1250: 	char *string;
1251: 	
1252: 	string = Tcl_GetString(namePtr);
1253: 	hPtr = Tcl_CreateHashEntry(&slavePtr->aliasTable, string, &new);
1254: 	if (new != 0) {
1255: 	    break;
1256: 	}
1257: 
1258: 	oldAliasPtr = (Alias *) Tcl_GetHashValue(hPtr);
1259: 	Tcl_DeleteCommandFromToken(slaveInterp, oldAliasPtr->slaveCmd);
1260:     }
1261: 
1262:     aliasPtr->aliasEntryPtr = hPtr;
1263:     Tcl_SetHashValue(hPtr, (ClientData) aliasPtr);
1264:     
1265:     /*
1266:      * Create the new command. We must do it after deleting any old command,
1267:      * because the alias may be pointing at a renamed alias, as in:
1268:      *
1269:      * interp alias {} foo {} bar		# Create an alias "foo"
1270:      * rename foo zop				# Now rename the alias
1271:      * interp alias {} foo {} zop		# Now recreate "foo"...
1272:      */
1273: 
1274:     targetPtr = (Target *) ckalloc((unsigned) sizeof(Target));
1275:     targetPtr->slaveCmd = aliasPtr->slaveCmd;
1276:     targetPtr->slaveInterp = slaveInterp;
1277: 
1278:     Tcl_MutexLock(&cntMutex);
1279:     masterPtr = &((InterpInfo *) ((Interp *) masterInterp)->interpInfo)->master;
1280:     do {
1281:         hPtr = Tcl_CreateHashEntry(&masterPtr->targetTable,
1282:                 (char *) aliasCounter, &new);
1283: 	aliasCounter++;
1284:     } while (new == 0);
1285:     Tcl_MutexUnlock(&cntMutex);
1286: 
1287:     Tcl_SetHashValue(hPtr, (ClientData) targetPtr);
1288:     aliasPtr->targetEntryPtr = hPtr;
1289: 
1290:     Tcl_SetObjResult(interp, namePtr);
1291: 
1292:     Tcl_Release(slaveInterp);
1293:     Tcl_Release(masterInterp);
1294:     return TCL_OK;
1295: }
