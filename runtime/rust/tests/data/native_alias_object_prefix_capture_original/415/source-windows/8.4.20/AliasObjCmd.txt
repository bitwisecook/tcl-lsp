1447:  *	invocation is forwarded.
1448:  *
1449:  *----------------------------------------------------------------------
1450:  */
1451: 
1452: static int
1453: AliasObjCmd(clientData, interp, objc, objv)
1454:     ClientData clientData;	/* Alias record. */
1455:     Tcl_Interp *interp;		/* Current interpreter. */
1456:     int objc;			/* Number of arguments. */
1457:     Tcl_Obj *CONST objv[];	/* Argument vector. */	
1458: {
1459: #define ALIAS_CMDV_PREALLOC 10
1460:     Tcl_Interp *targetInterp;	
1461:     Alias *aliasPtr;		
1462:     int result, prefc, cmdc, i;
1463:     Tcl_Obj **prefv, **cmdv;
1464:     Tcl_Obj *cmdArr[ALIAS_CMDV_PREALLOC];
1465:     aliasPtr = (Alias *) clientData;
1466:     targetInterp = aliasPtr->targetInterp;
1467: 
1468:     /*
1469:      * Append the arguments to the command prefix and invoke the command
1470:      * in the target interp's global namespace.
1471:      */
1472:      
1473:     prefc = aliasPtr->objc;
1474:     prefv = &aliasPtr->objPtr;
1475:     cmdc = prefc + objc - 1;
1476:     if (cmdc <= ALIAS_CMDV_PREALLOC) {
1477: 	cmdv = cmdArr;
1478:     } else {
1479: 	cmdv = (Tcl_Obj **) ckalloc((unsigned) (cmdc * sizeof(Tcl_Obj *)));
1480:     }
1481: 
1482:     prefv = &aliasPtr->objPtr;
1483:     memcpy((VOID *) cmdv, (VOID *) prefv, 
1484:             (size_t) (prefc * sizeof(Tcl_Obj *)));
1485:     memcpy((VOID *) (cmdv+prefc), (VOID *) (objv+1), 
1486: 	    (size_t) ((objc-1) * sizeof(Tcl_Obj *)));
1487: 
1488:     Tcl_ResetResult(targetInterp);
1489: 
1490:     for (i=0; i<cmdc; i++) {
1491: 	Tcl_IncrRefCount(cmdv[i]);
1492:     }
1493:     if (targetInterp != interp) {
1494: 	Tcl_Preserve((ClientData) targetInterp);
1495: 	result = Tcl_EvalObjv(targetInterp, cmdc, cmdv, TCL_EVAL_INVOKE);
1496: 	TclTransferResult(targetInterp, result, interp);	
1497: 	Tcl_Release((ClientData) targetInterp);
1498:     } else {
1499: 	result = Tcl_EvalObjv(targetInterp, cmdc, cmdv, TCL_EVAL_INVOKE);
1500:     }
1501:     for (i=0; i<cmdc; i++) {
1502: 	Tcl_DecrRefCount(cmdv[i]);
1503:     }
1504: 
1505:     if (cmdv != cmdArr) {
1506: 	ckfree((char *) cmdv);
1507:     }
1508:     return result;        
1509: #undef ALIAS_CMDV_PREALLOC
1510: }
