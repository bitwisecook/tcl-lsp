1851:  *	interpreter.
1852:  *
1853:  *----------------------------------------------------------------------
1854:  */

1856: static void
1857: AliasObjCmdDeleteProc(
1858:     ClientData clientData)	/* The alias record for this alias. */
1859: {
1860:     Alias *aliasPtr = clientData;
1861:     Target *targetPtr;
1862:     int i;
1863:     Tcl_Obj **objv;
1864: 
1865:     Tcl_DecrRefCount(aliasPtr->token);
1866:     objv = &aliasPtr->objPtr;
1867:     for (i = 0; i < aliasPtr->objc; i++) {
1868: 	Tcl_DecrRefCount(objv[i]);
1869:     }
1870:     Tcl_DeleteHashEntry(aliasPtr->aliasEntryPtr);
1871: 
1872:     /*
1873:      * Splice the target record out of the target interpreter's master list.
1874:      */
1875: 
1876:     targetPtr = aliasPtr->targetPtr;
1877:     if (targetPtr->prevPtr != NULL) {
1878: 	targetPtr->prevPtr->nextPtr = targetPtr->nextPtr;
1879:     } else {
1880: 	Master *masterPtr = &((InterpInfo *) ((Interp *)
1881: 		aliasPtr->targetInterp)->interpInfo)->master;
1882: 
1883: 	masterPtr->targetsPtr = targetPtr->nextPtr;
1884:     }
1885:     if (targetPtr->nextPtr != NULL) {
1886: 	targetPtr->nextPtr->prevPtr = targetPtr->prevPtr;
1887:     }
1888: 
1889:     ckfree((char *) targetPtr);
1890:     ckfree((char *) aliasPtr);
1891: }
