4128: /**
4129:  * Add the command to the commands hash table
4130:  */
4131: static void JimCreateCommand(Jim_Interp *interp, Jim_Obj *nameObjPtr, Jim_Cmd *cmd)
4132: {
4133:     /* If the entry already exists, nameObjPtr will not be used,
4134:      * so the refCount of nameObjPtr can't be zero, relying on this function to
4135:      * release it in that case.
4136:      */
4137:     JimPanic((nameObjPtr->refCount == 0, "JimCreateCommand called with zero ref count name"));
4138: 
4139:     /* It may already exist, so we try to delete the old one.
4140:      * Note that reference count means that it won't be deleted yet if
4141:      * it exists in the call stack.
4142:      *
4143:      * BUT, if 'local' is in force, instead of deleting the existing
4144:      * proc, we stash a reference to the old proc here.
4145:      */
4146:     if (interp->local) {
4147:         Jim_HashEntry *he = Jim_FindHashEntry(&interp->commands, nameObjPtr);
4148:         if (he) {
4149:             /* Push this command over the top of the previous one */
4150:             cmd->prevCmd = Jim_GetHashEntryVal(he);
4151:             Jim_SetHashVal(&interp->commands, he, cmd);
4152:             /* Need to increment the proc epoch here so that the new command will be used */
4153:             Jim_InterpIncrProcEpoch(interp);
4154:             return;
4155:         }
4156:     }
4157: 
4158:     /* Otherwise simply replace any existing command */
4159: 
4160:     /* Note that it is not necessary to increment the 'proc epoch' because any
4161:      * existing command that is replaced will be held as a negative cache entry
4162:      * until the next time the proc epoch is incremented.
4163:      */
4164:     Jim_ReplaceHashEntry(&interp->commands, nameObjPtr, cmd);
4165: }
