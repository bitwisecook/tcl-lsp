4211: /* Create the named static variables declared for a proc body. */
4212: static int JimCreateProcedureStatics(Jim_Interp *interp, Jim_Cmd *cmdPtr, Jim_Obj *staticsListObjPtr)
4213: {
4214:     int len, i;
4215: 
4216:     len = Jim_ListLength(interp, staticsListObjPtr);
4217:     if (len == 0) {
4218:         return JIM_OK;
4219:     }
4220: 
4221:     cmdPtr->u.proc.staticVars = Jim_Alloc(sizeof(Jim_HashTable));
4222:     Jim_InitHashTable(cmdPtr->u.proc.staticVars, &JimVariablesHashTableType, interp);
4223:     for (i = 0; i < len; i++) {
4224:         Jim_Obj *initObjPtr = NULL;
4225:         Jim_Obj *nameObjPtr;
4226:         Jim_VarVal *vv = NULL;
4227:         Jim_Obj *objPtr = Jim_ListGetIndex(interp, staticsListObjPtr, i);
4228:         int subLen = Jim_ListLength(interp, objPtr);
4229:         int byref = 0;
4230: 
4231:         /* Check if it's composed of two elements. */
4232:         if (subLen != 1 && subLen != 2) {
4233:             Jim_SetResultFormatted(interp, "too many fields in static specifier \"%#s\"",
4234:                 objPtr);
4235:             return JIM_ERR;
4236:         }
4237: 
4238:         nameObjPtr = Jim_ListGetIndex(interp, objPtr, 0);
4239: 
4240:         /* How to intialise or link? */
4241:         if (subLen == 1) {
4242:             int len;
4243:             const char *pt = Jim_GetString(nameObjPtr, &len);
4244:             if (*pt == '&') {
4245:                 /* Create as a reference */
4246:                 nameObjPtr = Jim_NewStringObj(interp, pt + 1, len - 1);
4247:                 byref = 1;
4248:             }
4249:         }
4250:         Jim_IncrRefCount(nameObjPtr);
4251: 
4252:         if (subLen == 1) {
4253:             switch (SetVariableFromAny(interp, nameObjPtr)) {
4254:                 case JIM_DICT_SUGAR:
4255:                     /* XXX: This message seem unnecessarily verbose, but it matches Tcl */
4256:                     if (byref) {
4257:                         Jim_SetResultFormatted(interp, "Can't link to array element \"%#s\"", nameObjPtr);
4258:                     }
4259:                     else {
4260:                         Jim_SetResultFormatted(interp, "Can't initialise array element \"%#s\"", nameObjPtr);
4261:                     }
4262:                     Jim_DecrRefCount(interp, nameObjPtr);
4263:                     return JIM_ERR;
4264: 
4265:                 case JIM_OK:
4266:                     if (byref) {
4267:                         vv = nameObjPtr->internalRep.varValue.vv;
4268:                     }
4269:                     else {
4270:                         initObjPtr = Jim_GetVariable(interp, nameObjPtr, JIM_NONE);
4271:                         if (!initObjPtr) {
4272:                             Jim_SetResultFormatted(interp, "Could not resolve upvar \"%#s\"'s value", nameObjPtr);
4273:                             Jim_DecrRefCount(interp, nameObjPtr);
4274:                             return JIM_ERR;
4275:                         }
4276:                     }
4277:                     break;
4278: 
4279:                 case JIM_ERR:
4280:                     /* Doesn't exist */
4281:                     Jim_SetResultFormatted(interp,
4282:                         "variable for initialization of static \"%#s\" not found in the local context",
4283:                         nameObjPtr);
4284:                     Jim_DecrRefCount(interp, nameObjPtr);
4285:                     return JIM_ERR;
4286:             }
4287:         }
4288:         else {
4289:             initObjPtr = Jim_ListGetIndex(interp, objPtr, 1);
4290:         }
4291: 
4292:         if (vv == NULL) {
4293:             vv = Jim_Alloc(sizeof(*vv));
4294:             vv->objPtr = initObjPtr;
4295:             Jim_IncrRefCount(vv->objPtr);
4296:             vv->linkFramePtr = NULL;
4297:             vv->refCount = 0;
4298:         }
4299: 
4300:         if (JimSetNewVariable(cmdPtr->u.proc.staticVars, nameObjPtr, vv) != JIM_OK) {
4301:             Jim_SetResultFormatted(interp,
4302:                 "static variable name \"%#s\" duplicated in statics list", nameObjPtr);
4303:             JimIncrVarRef(vv);
4304:             JimDecrVarRef(interp, vv);
4305:             Jim_DecrRefCount(interp, nameObjPtr);
4306:             return JIM_ERR;
4307:         }
4308: 
4309:         Jim_DecrRefCount(interp, nameObjPtr);
4310:     }
4311:     return JIM_OK;
4312: }
