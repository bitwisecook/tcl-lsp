79: static int interp_cmd_alias(Jim_Interp *interp, int argc, Jim_Obj *const *argv)
80: {
81:     Jim_Interp *child = Jim_CmdPrivData(interp);
82:     Jim_Obj *aliasPrefixList;
83: 
84:     /* The prefix list will be held inside the child, but it still belongs
85:      * to the parent!
86:      */
87: 
88:     aliasPrefixList = Jim_NewListObj(interp, argv + 1, argc - 1);
89:     Jim_IncrRefCount(aliasPrefixList);
90: 
91:     Jim_RegisterCmd(child, Jim_String(argv[0]), NULL, 0, -1, JimInterpAliasProc, JimInterpDelAlias, aliasPrefixList, 0);
92:     return JIM_OK;
93: }

126: static void JimInterpCopyVariable(Jim_Interp *target, Jim_Interp *source, const char *var, const char *default_value)
127: {
128:     Jim_Obj *value = Jim_GetGlobalVariableStr(source, var, JIM_NONE);
129:     const char *str;
130: 
131:     str = value ? Jim_String(value) : default_value;
132:     if (str) {
133:         Jim_SetGlobalVariableStr(target, var, Jim_NewStringObj(target, str, -1));
134:     }

137: /**
138:  * [interp] creates a new interpreter.
139:  */
140: static int JimInterpCommand(Jim_Interp *interp, int argc, Jim_Obj *const *argv)
141: {
142:     Jim_Interp *child;
143:     char buf[34];
144:     int i;
145:     static const char * const copyvars[] = {
146:         "argv", "argc", "argv0", "jim::argv0", "jim::exe", "jim::lineedit", NULL
147:     };
148: 
149:     /* Create the interpreter command */
150:     child = Jim_CreateInterp();
151:     Jim_RegisterCoreCommands(child);
152:     Jim_InitStaticExtensions(child);
153: 
154:     /* Copy some core variables to the new interpreter */
155:     for (i = 0; copyvars[i]; i++) {
156:         JimInterpCopyVariable(child, interp, copyvars[i], NULL);
157:     }
158: 
159:     /* Allow the child interpreter to find the parent */
160:     Jim_SetAssocData(child, "interp.parent", NULL, interp);
161: 
162:     snprintf(buf, sizeof(buf), "interp.handle%ld", Jim_GetId(interp));
163:     Jim_RegisterCmd(interp, buf, "subcommand ?arg ...?", 1, -1, JimInterpSubCmdProc, JimInterpDelProc, child, 0);
164:     Jim_SetResult(interp, Jim_MakeGlobalNamespaceName(interp, Jim_NewStringObj(interp, buf, -1)));
165:     return JIM_OK;
166: }
