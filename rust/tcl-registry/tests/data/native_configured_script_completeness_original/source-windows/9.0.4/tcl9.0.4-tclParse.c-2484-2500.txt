TclObjCommandComplete(
    Tcl_Obj *objPtr)		/* Points to object holding script to
				 * check. */
{
    Tcl_Size length;
    const char *script = TclGetStringFromObj(objPtr, &length);

    return CommandComplete(script, length);
}
