use tcl_compiler::analyser::ResolvedAnalysisInput;
use tcl_compiler::command_binding::{SourceAnalysisEntry, SourceAnalysisOptions, SourceCommandBindings};
use tcl_compiler::registry_invocation::{InvocationMetadataContext, normal_transfer_invocation_with_metadata_context, resolved_tokens_invocation_with_metadata_context};
use tcl_registry::native_compilation::{NativeCompilationContext, NativeCompilationFrame, NativeCompilationMode};

fn main() {
    for engine in ["tcl8.4", "tcl8.6", "tcl9.0", "jimtcl"] {
        let environment = tcl_registry::model::ingress::resolve_known_environment(engine).unwrap();
        let context = environment.default_context_registry();
        let registry = context.commands();
        let dialect = tcl_registry::InvocationDialect::of_point(tcl_dialect::model::DialectPoint::of_dialect_name(Some(engine)).unwrap());
        let config = tcl_lexer::LexerConfig::from_grammar(dialect.lexer_grammar);
        let profile = registry.profile().unwrap();
        let input = ResolvedAnalysisInput::new(profile, profile, context.clone(), config);
        let supplied = SourceAnalysisEntry::for_supplied_source(registry, &input, config, Some(profile));
        for command in ["info exists u", "array exists u"] {
            for called in [false, true] {
                let source = format!("proc f {{}} {{{command}}}{}", if called { "; f" } else { "" });
                for (mode, options) in [
                    ("StandaloneFresh", SourceAnalysisOptions { invocation_dialect: Some(dialect), native_compilation: NativeCompilationContext { mode: NativeCompilationMode::Direct, frame: NativeCompilationFrame::ScriptCode, loop_depth: 0, catch_depth: Some(0) }, ..Default::default() }),
                    ("StandaloneUnknown", SourceAnalysisOptions { invocation_dialect: Some(dialect), unknown_entry: true, native_compilation: NativeCompilationContext { mode: NativeCompilationMode::Direct, frame: NativeCompilationFrame::ScriptCode, loop_depth: 0, catch_depth: Some(0) }, ..Default::default() }),
                    ("SuppliedFresh", supplied.options()),
                ] {
                    let bindings = SourceCommandBindings::analyse_with_options(&source, config, registry, options);
                    let offset = u32::try_from(source.find(command).unwrap()).unwrap();
                    let segment = tcl_compiler::segmenter::segment_commands_with_offset_and_config(command, offset, config).remove(0);
                    let mut tokens = tcl_compiler::ir::CommandTokens::from_segmented(&tcl_lexer::SourceMap::new(&source), config, &segment);
                    bindings.stamp_original_tokens(&mut tokens);
                    let Some(binding) = tokens.source_binding.as_ref() else {
                        println!("QUERY_PURPOSE {}", serde_json::json!({"engine": engine, "mode": mode, "called": called, "source": source, "source_binding": false}));
                        continue;
                    };
                    let metadata = if mode == "SuppliedFresh" { InvocationMetadataContext::for_analysis_input(registry, &input).unwrap() } else { InvocationMetadataContext::from(context.as_ref()) };
                    println!("QUERY_PURPOSE {}", serde_json::json!({
                        "engine": engine, "mode": mode, "called": called, "source": source,
                        "metadata_kind": if mode == "SuppliedFresh" { "SuppliedSource" } else { "AvailabilityFromOriginalContext" },
                        "native_entry": options.native_entry.is_some(), "source_point": options.source_point_entry.is_some(),
                        "authority": options.execution_name_policy().and_then(tcl_syntax::naming::ExecutionNamePolicy::native_recipe).map(tcl_syntax::naming::NamePolicyProtocol::authority).map(|v| format!("{v:?}")),
                        "runtime_reachability": format!("{:?}", binding.runtime_reachability()),
                        "target": binding.proved_execution_target().map(|target| target.command.clone()),
                        "handler": binding.proved_handler_target().map(|target| target.command.clone()),
                        "strict_query_metadata": resolved_tokens_invocation_with_metadata_context(registry, Some(metadata), &tokens).is_some(),
                        "normal_transfer": normal_transfer_invocation_with_metadata_context(registry, Some(metadata), &tokens).is_some(),
                        "normal_result": binding.original_normal_result(&tokens).map(|result| result.text()),
                        "completes_normally": binding.original_invocation_completes_normally(&tokens),
                        "inline_admitted": binding.admitted_inline_invocation().is_some(), "named_admitted": binding.admitted_named_invocation().is_some(),
                        "frame": format!("{:?}", binding.variable_context.frame_kind),
                    }));
                }
            }
        }
    }
}
