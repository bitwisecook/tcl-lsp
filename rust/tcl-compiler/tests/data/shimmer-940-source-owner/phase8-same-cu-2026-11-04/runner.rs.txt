//! Pending fixed-input source-owner probe; install only on a candidate snapshot.
//! Warning assertions are opt-in so existing gaps remain measured separately.

use serde_json::{Value, json};
use tcl_compiler::analyser::Analyser;
use tcl_compiler::compilation_unit::{CompilationUnit, UnitBuildOptions};
use tcl_compiler::ir::{Script, Statement};
use tcl_compiler::native_compilation_admission::{
    NativeCompilationAdmissionPlan, NativeCompilationAdmissionScope, script_admission_plan,
};
use tcl_compiler::shimmer::ShimmerContext;

fn plan_facts(plan: NativeCompilationAdmissionPlan<'_>) -> Value {
    match plan {
        NativeCompilationAdmissionPlan::NoRetainedObligation => {
            json!({"kind": "NoRetainedObligation"})
        }
        NativeCompilationAdmissionPlan::RefuseMissingSource => {
            json!({"kind": "RefuseMissingSource"})
        }
        NativeCompilationAdmissionPlan::HostScript(source) => json!({
            "kind": "HostScript", "source_bytes": source.text.bytes(),
            "mapping": format!("{:?}", source.mapping),
        }),
        NativeCompilationAdmissionPlan::HostProcedure(source) => json!({
            "kind": "HostProcedure", "source_bytes": source.text.bytes(),
            "mapping": format!("{:?}", source.mapping),
        }),
    }
}

fn script_facts(
    script: &Script,
    registry: &tcl_registry::CommandRegistry,
    metadata: Option<tcl_compiler::registry_invocation::InvocationMetadataContext<'_>>,
) -> Vec<Value> {
    let mut facts = Vec::new();
    let mut held = script.clone();
    tcl_compiler::ir::for_each_script_mut(&mut held, &mut |body| {
        let obligations = json!({
            "script_plan": plan_facts(script_admission_plan(body, NativeCompilationAdmissionScope::Script)),
            "procedure_plan": plan_facts(script_admission_plan(body, NativeCompilationAdmissionScope::ProcedureBody)),
            "retained_admission": body.native_compilation_admission.is_some(),
            "provider_required": body.native_compilation_admission.as_ref().map(|admission| admission.provider_required),
            "retained_failure": body.native_compilation_failure.is_some(),
            "admission_failure": body.native_compilation_admission.as_ref().map(|admission| admission.failure.is_some()),
            "admission_source_bytes": body.native_compilation_admission.as_ref().and_then(|admission| admission.source.as_ref()).map(|source| source.text.bytes()),
        });
        for statement in &body.statements {
            let span = statement.span();
            let tokens = body.retained_source_tokens_for_statement(statement);
            let native = tokens.is_some_and(|tokens| {
                tcl_compiler::registry_invocation::admitted_native_compiler_invocation(
                    registry, None, tokens,
                )
                .is_some()
            });
            let normal = tokens.is_some_and(|tokens| {
                tokens
                    .source_binding
                    .as_ref()
                    .is_some_and(|binding| binding.original_normal_result(tokens).is_some())
            });
            let normal_text = tokens.and_then(|tokens| {
                tokens
                    .source_binding
                    .as_ref()?
                    .original_normal_result(tokens)
                    .map(|result| result.text().to_owned())
            });
            let trace = tcl_compiler::registry_invocation::begin_host_structure_trace();
            let original_structure = tokens.zip(metadata).is_some_and(|(tokens, metadata)| {
                tcl_compiler::registry_invocation::original_structured_invocation_with_metadata_context(
                    registry, metadata, tokens,
                ).is_some()
            });
            let purpose_and_guards = trace.finish();
            let kind = match statement {
                Statement::Foreach { .. } => "Foreach",
                Statement::While { .. } => "While",
                Statement::If { .. } => "If",
                Statement::AssignConst { .. } => "AssignConst",
                Statement::AssignExpr { .. } => "AssignExpr",
                Statement::AssignValue { .. } => "AssignValue",
                Statement::Call { .. } => "Call",
                Statement::Barrier { .. } => "Barrier",
                Statement::NativeCall { .. } => "NativeCall",
                _ => "Other",
            };
            facts.push(json!({
                "start": span.start(), "end": span.end(), "kind": kind,
                "original_tokens": tokens.is_some(),
                "source_binding": tokens.is_some_and(|tokens| tokens.source_binding.is_some()),
                "native_compiler_admitted": native,
                "completed_normal_result": normal, "normal_text": normal_text,
                "completes_normally": tokens.is_some_and(|tokens| tokens.source_binding.as_ref().is_some_and(|binding| binding.original_invocation_completes_normally(tokens))),
                "obligations": obligations,
                "original_structure": original_structure, "purpose_and_guards": purpose_and_guards,
            }));
        }
    });
    facts
}

fn exact_unit_authority(
    unit: &CompilationUnit,
    registry: &tcl_registry::CommandRegistry,
) -> Vec<Value> {
    let metadata = tcl_compiler::registry_invocation::InvocationMetadataContext::for_module(
        registry,
        &unit.ir_module,
    );
    let mut snapshots = Vec::new();
    for function in unit.analysable_body_function_units() {
        let original = function
            .body_id
            .and_then(|id| unit.original_procedure_for_body(id));
        let retained = function
            .body_id
            .and_then(|id| original.or_else(|| unit.ir_module.body_lowering(&id)));
        let script = if function.body_id.is_none() {
            Some(&unit.ir_module.top_level)
        } else {
            retained.map(|procedure| &procedure.body)
        };
        let rows: Vec<_> = script.into_iter().flat_map(|script| {
            script_facts(script, registry, metadata)
        }).map(|row| json!({
            "start": row["start"], "end": row["end"],
            "original_tokens": row["original_tokens"], "source_binding": row["source_binding"],
            "native_compiler_admitted": row["native_compiler_admitted"],
            "completed_normal_result": row["completed_normal_result"],
            "normal_text": row["normal_text"], "completes_normally": row["completes_normally"], "obligations": row["obligations"],
        })).collect();
        snapshots.push(json!({"body_id": format!("{:?}", function.body_id), "rows": rows}));
    }
    snapshots
}

#[test]
fn phase8_same_held_unit_guards() {
    let cases = vec![
        (
            "cycle-positive",
            "set x 0; foreach i {1 2 3} {incr x; lindex $x 0}",
            1,
        ),
        (
            "invalid-seed",
            "set x abc; foreach i {1 2 3} {incr x; lindex $x 0}",
            0,
        ),
        (
            "copied-invalid-seed",
            "set a abc; set x $a; foreach i {1 2 3} {incr x; lindex $x 0}",
            0,
        ),
        (
            "zero-trips",
            "set x 0; foreach i {} {incr x; lindex $x 0}",
            0,
        ),
        (
            "one-trip",
            "set x 0; foreach i {1} {incr x; lindex $x 0}",
            0,
        ),
        (
            "unknown-list",
            "set x 0; foreach i $unknown {incr x; lindex $x 0}",
            0,
        ),
        (
            "binder-resets",
            "set x 0; foreach x {1 2 3} {incr x; lindex $x 0}",
            0,
        ),
        (
            "body-resets",
            "set x 0; foreach i {1 2 3} {set x 0; incr x; lindex $x 0}",
            0,
        ),
        (
            "in-loop-redefinition",
            "set x 0; foreach i {1 2 3} {incr x; lindex $x 0; proc incr args {}}",
            0,
        ),
        (
            "in-loop-trace",
            "proc observe args {}; set x 0; foreach i {1 2 3} {incr x; lindex $x 0; trace add variable x write observe}",
            0,
        ),
        (
            "conditional-report",
            "set ys {}; set pts {}; foreach pt {{1 2} {3 4} {5 6}} {set y [lindex $pt 1]; if {$y > 2} {lappend ys $y; lappend pts $pt}}",
            0,
        ),
        (
            "unconditional-report",
            "set ys {}; set pts {}; foreach pt {{1 2} {3 4} {5 6}} {set y [lindex $pt 1]; lappend ys $y; lappend pts $pt}",
            0,
        ),
        (
            "font-report",
            "set fontSizes {10.0 12.0 16.0 24.0}; foreach size $fontSizes {puts $size}",
            0,
        ),
    ];
    for version in tcl_dialect::TclVersion::ALL {
        for (id, source, expected) in &cases {
            let mut analyser = Analyser::new();
            let analysis = analyser.analyse(source, version.dialect_profile_name());
            let input = analyser.resolved_analysis_input();
            let settings = analyser.source_compilation_settings();
            let registry = input.borrowed_context_registry().commands();
            let profile = input.unit_profile();
            let options = UnitBuildOptions {
                registry,
                defer_top_level: false,
                config: input.lexer_config(),
                dialect: Some(profile),
                external_call_sites: None,
                declared_commands: None,
            };
            let unit = CompilationUnit::build_with_analysis_input_and_source_settings(
                source, options, None, &input, &settings,
            )
            .with_interprocedural(registry, Some(profile));
            assert!(unit.analysable_functions().next().is_some());
            assert!(
                unit.analysable_functions()
                    .all(|function| ShimmerContext::for_function(function, registry).is_some())
            );
            assert!(unit.ir_module.source_entry.native_entry.is_none());
            let bytecode = || {
                let mut backends = Vec::new();
                for traced in [false, true] {
                    let mut lowerer = tcl_compiler::lowering::Lowerer::with_config(
                        registry,
                        input.lexer_config(),
                    )
                    .with_dialect(Some(profile))
                    .with_resolved_analysis_input(input.clone())
                    .for_bytecode_backend();
                    lowerer.set_source_analysis_options(unit.ir_module.source_entry.options());
                    if traced {
                        lowerer = lowerer.trace_visible();
                    }
                    let module = lowerer.lower(source);
                    let roots: Vec<_> = module.executable_script_roots().into_iter().map(|(body, _)| {
                    script_facts(body, registry, tcl_compiler::registry_invocation::InvocationMetadataContext::for_module(registry, module))
                }).collect();
                    backends.push(json!({"traced":traced, "roots": roots}));
                }
                backends
            };
            let backend_before = bytecode();
            let original_ir = unit.ir_module.clone();
            let original_ssa: Vec<_> = unit
                .analysable_body_function_units()
                .map(|function| (function.body_id, function.ssa.clone()))
                .collect();
            let authority_before = exact_unit_authority(&unit, registry);
            let warnings = tcl_compiler::shimmer::find_thunking_warnings_for_cu(&unit, registry);
            let checks =
                tcl_compiler::compiler_checks::run_all_checks(&unit, registry, Some(profile));
            let selected: Vec<_> = warnings
                .iter()
                .filter(|warning| warning.code == tcl_core_types::DiagCode::S102)
                .collect();
            let diagnostics: Vec<_> = checks
                .iter()
                .filter(|finding| finding.code == tcl_core_types::DiagCode::S102)
                .collect();
            assert_eq!(selected.len(), *expected, "{version:?}/{id}: {warnings:?}");
            assert_eq!(diagnostics.len(), *expected, "{version:?}/{id}: {checks:?}");
            if *expected == 1 {
                assert_eq!(selected[0].variable, "x");
                assert_eq!(selected[0].span.start(), 43);
                assert_eq!(diagnostics[0].span, selected[0].span);
                assert!(diagnostics[0].replacement.is_none() && diagnostics[0].fixes.is_empty());
                assert!(
                    diagnostics[0]
                        .source_context
                        .as_ref()
                        .is_some_and(|context| !context.allows_context_action_subject())
                );
                assert!(
                    !checks
                        .iter()
                        .any(|finding| finding.code == tcl_core_types::DiagCode::S100
                            && finding.span == selected[0].span)
                );
            }
            assert!(unit.ir_module == original_ir, "held original IR changed");
            let after: Vec<_> = unit.analysable_body_function_units().collect();
            assert_eq!(original_ssa.len(), after.len());
            for ((id, ssa), function) in original_ssa.iter().zip(after) {
                assert!(
                    id == &function.body_id && ssa == &function.ssa,
                    "held SSA changed"
                );
            }
            let authority_after = exact_unit_authority(&unit, registry);
            assert_eq!(
                authority_before, authority_after,
                "admission/Normal/provider changed"
            );
            let backend_after = bytecode();
            assert_eq!(
                backend_before, backend_after,
                "Bytecode/Traced obligations changed"
            );
            println!(
                "PHASE8_SAME_CU {}",
                json!({
                    "case": id, "dialect": version.dialect_profile_name(), "source": source,
                    "expected_s102": expected,
                    "warnings": selected.iter().map(|warning| json!({"code": warning.code.to_string(), "variable": warning.variable, "start": warning.span.start(), "end": warning.span.end()})).collect::<Vec<_>>(),
                    "checks": diagnostics.iter().map(|finding| json!({"code": finding.code.to_string(), "start": finding.span.start(), "end": finding.span.end(), "fixes": finding.fixes.len(), "replacement": finding.replacement.is_some(), "action_authority": finding.source_context.as_ref().is_some_and(|context|context.allows_context_action_subject())})).collect::<Vec<_>>(),
                    "native_entry": unit.ir_module.source_entry.native_entry.is_some(),
                    "authority_before": authority_before, "authority_after": authority_after,
                    "backends_before": backend_before, "backends_after": backend_after,
                    "held_ir_equal": unit.ir_module == original_ir, "held_ssa_equal": true,
                    "analysis_codes": analysis.diagnostics.iter().map(|finding|finding.code.to_string()).collect::<Vec<_>>(),
                })
            );
        }
    }
}
