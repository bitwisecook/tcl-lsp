//! Pending fixed-input source-owner probe; install only on a candidate snapshot.
//! Warning assertions are opt-in so existing gaps remain measured separately.

use std::path::Path;

use serde_json::{Value, json};
use tcl_compiler::analyser::Analyser;
use tcl_compiler::compilation_unit::{CompilationUnit, UnitBuildOptions};
use tcl_compiler::ir::{Script, Statement};
use tcl_compiler::native_compilation_admission::{
    NativeCompilationAdmissionPlan, NativeCompilationAdmissionScope, script_admission_plan,
};
use tcl_compiler::shimmer::ShimmerContext;

fn source_for_case(root: &Path, case: &Value) -> String {
    if let Some(source) = case["source"].as_str() {
        return source.to_owned();
    }
    if let Some(fixture) = case["fixture"].as_str() {
        return std::fs::read_to_string(root.join(fixture)).unwrap();
    }
    let receipt: Value = serde_json::from_str(
        &std::fs::read_to_string(root.join(case["receipt"].as_str().unwrap())).unwrap(),
    )
    .unwrap();
    let sources: Vec<_> = receipt
        .as_array()
        .unwrap()
        .iter()
        .filter(|row| row["input"] == case["receipt_input"])
        .map(|row| row["source"].as_str().unwrap())
        .collect();
    assert!(!sources.is_empty());
    assert!(sources.iter().all(|source| *source == sources[0]));
    sources[0].to_owned()
}

fn plan_facts(plan: NativeCompilationAdmissionPlan<'_>) -> Value {
    match plan {
        NativeCompilationAdmissionPlan::NoRetainedObligation => {
            json!({"kind": "NoRetainedObligation"})
        }
        NativeCompilationAdmissionPlan::RefuseMissingSource => {
            json!({"kind": "RefuseMissingSource"})
        }
        NativeCompilationAdmissionPlan::HostScript(source) => json!({
            "kind": "HostScript", "source_bytes": source.text.bytes(),
            "mapping": format!("{:?}", source.mapping),
        }),
        NativeCompilationAdmissionPlan::HostProcedure(source) => json!({
            "kind": "HostProcedure", "source_bytes": source.text.bytes(),
            "mapping": format!("{:?}", source.mapping),
        }),
    }
}

fn script_facts(
    script: &Script,
    registry: &tcl_registry::CommandRegistry,
    metadata: Option<tcl_compiler::registry_invocation::InvocationMetadataContext<'_>>,
) -> Vec<Value> {
    let mut facts = Vec::new();
    let mut held = script.clone();
    tcl_compiler::ir::for_each_script_mut(&mut held, &mut |body| {
        let obligations = json!({
            "script_plan": plan_facts(script_admission_plan(body, NativeCompilationAdmissionScope::Script)),
            "procedure_plan": plan_facts(script_admission_plan(body, NativeCompilationAdmissionScope::ProcedureBody)),
            "retained_admission": body.native_compilation_admission.is_some(),
            "provider_required": body.native_compilation_admission.as_ref().map(|admission| admission.provider_required),
            "retained_failure": body.native_compilation_failure.is_some(),
            "admission_failure": body.native_compilation_admission.as_ref().map(|admission| admission.failure.is_some()),
            "admission_source_bytes": body.native_compilation_admission.as_ref().and_then(|admission| admission.source.as_ref()).map(|source| source.text.bytes()),
        });
        for statement in &body.statements {
            let span = statement.span();
            let tokens = body.retained_source_tokens_for_statement(statement);
            let native = tokens.is_some_and(|tokens| {
                tcl_compiler::registry_invocation::admitted_native_compiler_invocation(
                    registry, None, tokens,
                )
                .is_some()
            });
            let normal = tokens.is_some_and(|tokens| {
                tokens
                    .source_binding
                    .as_ref()
                    .is_some_and(|binding| binding.original_normal_result(tokens).is_some())
            });
            let normal_text = tokens.and_then(|tokens| {
                tokens
                    .source_binding
                    .as_ref()?
                    .original_normal_result(tokens)
                    .map(|result| result.text().to_owned())
            });
            let trace = tcl_compiler::registry_invocation::begin_host_structure_trace();
            let original_structure = tokens.zip(metadata).is_some_and(|(tokens, metadata)| {
                tcl_compiler::registry_invocation::original_structured_invocation_with_metadata_context(
                    registry, metadata, tokens,
                ).is_some()
            });
            let purpose_and_guards = trace.finish();
            let kind = match statement {
                Statement::Foreach { .. } => "Foreach",
                Statement::While { .. } => "While",
                Statement::If { .. } => "If",
                Statement::AssignConst { .. } => "AssignConst",
                Statement::AssignExpr { .. } => "AssignExpr",
                Statement::AssignValue { .. } => "AssignValue",
                Statement::Call { .. } => "Call",
                Statement::Barrier { .. } => "Barrier",
                Statement::NativeCall { .. } => "NativeCall",
                _ => "Other",
            };
            facts.push(json!({
                "start": span.start(), "end": span.end(), "kind": kind,
                "original_tokens": tokens.is_some(),
                "source_binding": tokens.is_some_and(|tokens| tokens.source_binding.is_some()),
                "native_compiler_admitted": native,
                "completed_normal_result": normal, "normal_text": normal_text,
                "completes_normally": tokens.is_some_and(|tokens| tokens.source_binding.as_ref().is_some_and(|binding| binding.original_invocation_completes_normally(tokens))),
                "obligations": obligations,
                "original_structure": original_structure, "purpose_and_guards": purpose_and_guards,
            }));
        }
    });
    facts
}

fn exact_unit_authority(
    unit: &CompilationUnit,
    registry: &tcl_registry::CommandRegistry,
) -> Vec<Value> {
    let metadata = tcl_compiler::registry_invocation::InvocationMetadataContext::for_module(
        registry,
        &unit.ir_module,
    );
    let mut snapshots = Vec::new();
    for function in unit.analysable_body_function_units() {
        let original = function
            .body_id
            .and_then(|id| unit.original_procedure_for_body(id));
        let retained = function
            .body_id
            .and_then(|id| original.or_else(|| unit.ir_module.body_lowering(&id)));
        let script = if function.body_id.is_none() {
            Some(&unit.ir_module.top_level)
        } else {
            retained.map(|procedure| &procedure.body)
        };
        let rows: Vec<_> = script.into_iter().flat_map(|script| {
            script_facts(script, registry, metadata)
        }).map(|row| json!({
            "start": row["start"], "end": row["end"],
            "original_tokens": row["original_tokens"], "source_binding": row["source_binding"],
            "native_compiler_admitted": row["native_compiler_admitted"],
            "completed_normal_result": row["completed_normal_result"],
            "normal_text": row["normal_text"], "completes_normally": row["completes_normally"], "obligations": row["obligations"],
        })).collect();
        snapshots.push(json!({"body_id": format!("{:?}", function.body_id), "rows": rows}));
    }
    snapshots
}

#[test]
fn pending_source_owner_fixed_input_matrix() {
    let root = Path::new(env!("CARGO_MANIFEST_DIR")).join("tests/data/shimmer-940-source-owner");
    let matrix: Value =
        serde_json::from_str(&std::fs::read_to_string(root.join("cases.json")).unwrap()).unwrap();
    let assert_warnings = std::env::var_os("LSP_940_ASSERT_EXPECTATIONS").is_some();
    let mut cases: Vec<Value> = matrix["cases"]
        .as_array()
        .unwrap()
        .iter()
        .filter(|case| {
            matches!(
                case["id"].as_str(),
                Some(
                    "conditional-top" | "conditional-proc" | "font-braced-top" | "font-braced-proc"
                )
            )
        })
        .cloned()
        .collect();
    for (id, source, expected) in [
        (
            "S100-dynamic-length",
            "set s abc; set n [string length $s]; lindex $n 0",
            1,
        ),
        (
            "S100-named-proc",
            "proc build {s} {set n [string length $s]; lindex $n 0}; build abc",
            1,
        ),
        (
            "S100-literal-length",
            "set n [string length abc]; lindex $n 0",
            0,
        ),
        ("S100-braced-list", "set n {1 2}; lindex $n 0", 0),
        ("S100-literal-integer", "set n 123; lindex $n 0", 0),
        (
            "S100-nested-result",
            "set s abc; set n [lindex [string length $s] 0]; lindex $n 0",
            0,
        ),
        (
            "S100-shadowed",
            "proc lindex args {return 0}; set s abc; set n [string length $s]; lindex $n 0",
            0,
        ),
        (
            "S100-renamed",
            "rename lindex saved; set s abc; set n [string length $s]; lindex $n 0",
            0,
        ),
        (
            "S100-write-traced",
            "proc observe args {}; set s abc; trace add variable n write observe; set n [string length $s]; lindex $n 0",
            0,
        ),
    ] {
        cases.push(json!({"id": id, "stage": "S100-foundation", "source": source, "expected_s100_n": if id == "S100-named-proc" { None } else { Some(expected) }, "wanted_s100_n": expected}));
    }
    for case in &cases {
        let source = source_for_case(&root, case);
        let mut analyser = Analyser::new();
        let analysis = analyser.analyse(&source, "tcl8.6");
        let input = analyser.resolved_analysis_input();
        let settings = analyser.source_compilation_settings();
        let registry = input.borrowed_context_registry().commands();
        let profile = input.unit_profile();
        let options = UnitBuildOptions {
            registry,
            defer_top_level: false,
            config: input.lexer_config(),
            dialect: Some(profile),
            external_call_sites: None,
            declared_commands: None,
        };
        let unit = CompilationUnit::build_with_analysis_input_and_source_settings(
            &source, options, None, &input, &settings,
        )
        .with_interprocedural(registry, Some(profile));
        assert!(unit.analysable_functions().next().is_some());
        assert!(
            unit.analysable_functions()
                .all(|function| { ShimmerContext::for_function(function, registry).is_some() })
        );
        assert!(unit.ir_module.source_entry.native_entry.is_none());
        assert_eq!(
            unit.ir_module
                .source_entry
                .execution_name_policy
                .and_then(tcl_syntax::naming::ExecutionNamePolicy::native_recipe)
                .map(tcl_syntax::naming::NamePolicyProtocol::authority),
            Some(tcl_syntax::naming::NamePolicyAuthority::AuthoredSimulation)
        );
        let bytecode = || {
            let mut backends = Vec::new();
            for traced in [false, true] {
                let mut lowerer =
                    tcl_compiler::lowering::Lowerer::with_config(registry, input.lexer_config())
                        .with_dialect(Some(profile))
                        .with_resolved_analysis_input(input.clone())
                        .for_bytecode_backend();
                lowerer.set_source_analysis_options(unit.ir_module.source_entry.options());
                if traced {
                    lowerer = lowerer.trace_visible();
                }
                let module = lowerer.lower(&source);
                let roots: Vec<_> = module
                    .executable_script_roots()
                    .into_iter()
                    .map(|(body, _)| {
                        script_facts(
                        body,
                        registry,
                        tcl_compiler::registry_invocation::InvocationMetadataContext::for_module(
                            registry, module,
                        ),
                    )
                    })
                    .collect();
                backends.push(json!({"traced": traced, "roots": roots}));
            }
            backends
        };
        let backend_before = bytecode();
        let original_ir = unit.ir_module.clone();
        let original_ssa: Vec<_> = unit
            .analysable_body_function_units()
            .map(|function| (function.body_id, function.ssa.clone()))
            .collect();
        let authority_before = exact_unit_authority(&unit, registry);
        let mut warnings: Vec<(String, String)> =
            tcl_compiler::shimmer::find_shimmer_warnings_for_cu(&unit, registry)
                .into_iter()
                .map(|warning| (warning.code.to_string(), warning.variable))
                .collect();
        warnings.extend(
            tcl_compiler::shimmer::find_thunking_warnings_for_cu(&unit, registry)
                .into_iter()
                .map(|warning| (warning.code.to_string(), warning.variable)),
        );
        assert!(
            unit.ir_module == original_ir,
            "warning collection changed the held original IR"
        );
        let after_functions: Vec<_> = unit.analysable_body_function_units().collect();
        assert_eq!(original_ssa.len(), after_functions.len());
        for ((body, ssa), function) in original_ssa.iter().zip(after_functions) {
            assert!(
                body == &function.body_id && ssa == &function.ssa,
                "warning collection changed a held original SSA/body owner"
            );
        }
        let authority_after = exact_unit_authority(&unit, registry);
        assert!(
            authority_before == authority_after,
            "warning collection changed native admission, Normal values or provider obligations"
        );
        let mut functions = Vec::new();
        for function in unit.analysable_body_function_units() {
            let forest = tcl_compiler::loops::build_loop_forest(
                &function.cfg,
                &function.ssa,
                &function.sccp.executable_blocks,
            );
            let mut definitions = Vec::new();
            for (block, body) in &function.ssa.blocks {
                for statement in &body.statements {
                    for (&symbol, &version) in &statement.defs {
                        definitions.push(json!({
                            "symbol": format!("{symbol:?}"), "version": version,
                            "name_label": function.ssa.var_name(symbol),
                            "type": format!("{:?}", function.types.get(&(symbol, version))),
                            "may_def": statement.may_defs.contains(&symbol),
                            "destruction_def": statement.destruction_defs.contains(&symbol),
                            "self_read": statement.uses.contains_key(&symbol),
                            "executable": function.sccp.executable_blocks.contains(block),
                            "block_label": function.cfg.block_name(*block),
                            "span": [statement.statement.span().start(), statement.statement.span().end()],
                        }));
                    }
                }
            }
            let original = function
                .body_id
                .and_then(|id| unit.original_procedure_for_body(id));
            let retained = function
                .body_id
                .and_then(|id| original.or_else(|| unit.ir_module.body_lowering(&id)));
            let script = if function.body_id.is_none() {
                Some(&unit.ir_module.top_level)
            } else {
                retained.map(|procedure| &procedure.body)
            };
            functions.push(json!({
                "body_id": format!("{:?}", function.body_id),
                "original_declaration": original.is_some(),
                "loop_forest": format!("{forest:?}"), "definitions": definitions,
                "source_statements": script.map(|body| script_facts(body, registry,
                    tcl_compiler::registry_invocation::InvocationMetadataContext::for_module(registry, &unit.ir_module))),
            }));
        }
        let backends = bytecode();
        assert_eq!(
            backend_before, backends,
            "warning collection changed Bytecode/BytecodeTraced source obligations"
        );
        println!(
            "SOURCE_OWNER_940 {}",
            json!({
                "case": case["id"], "stage": case["stage"], "source": source, "warnings": warnings, "wanted_s100_n": case["wanted_s100_n"], "asserted_s100_n": case["expected_s100_n"],
                "authority_before": authority_before, "authority_after": authority_after,
                "backends": backends,
                "diagnostics": analysis.diagnostics.iter().map(|finding| json!({
                    "code": finding.code.to_string(), "message": finding.message,
                    "start": finding.span.start(), "end": finding.span.end(),
                })).collect::<Vec<_>>(), "functions": functions,
            })
        );
        if let Some(expected) = case["expected_s100_n"].as_u64() {
            let count = warnings
                .iter()
                .filter(|(code, variable)| code == "S100" && variable == "n")
                .count();
            assert_eq!(
                u64::try_from(count).unwrap(),
                expected,
                "{}: {warnings:?}",
                case["id"]
            );
        }
        if assert_warnings {
            for expected in case["expect_present"].as_array().into_iter().flatten() {
                let pair = (
                    expected[0].as_str().unwrap().to_owned(),
                    expected[1].as_str().unwrap().to_owned(),
                );
                if pair.0.starts_with('S') {
                    assert!(
                        warnings.contains(&pair),
                        "{} missing {pair:?}: {warnings:?}",
                        case["id"]
                    );
                }
            }
            for expected in case["expect_absent"].as_array().into_iter().flatten() {
                let pair = (
                    expected[0].as_str().unwrap().to_owned(),
                    expected[1].as_str().unwrap().to_owned(),
                );
                if pair.0.starts_with('W') {
                    assert!(
                        analysis.diagnostics.iter().all(|finding| {
                            finding.code.to_string() != pair.0 || !finding.message.contains(&pair.1)
                        }),
                        "{} unexpected diagnostic {pair:?}",
                        case["id"]
                    );
                } else if pair.0.starts_with('S') {
                    assert!(
                        !warnings.contains(&pair),
                        "{} unexpected {pair:?}: {warnings:?}",
                        case["id"]
                    );
                }
            }
        }
    }
}
