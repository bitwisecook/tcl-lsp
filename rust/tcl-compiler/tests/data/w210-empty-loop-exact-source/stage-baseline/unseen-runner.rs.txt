use tcl_compiler::analyser::Analyser;
use tcl_compiler::compilation_unit::{CompilationUnit, UnitBuildOptions};
fn main() {
    for source in ["proc p {} {foreach x {} {}; puts $x}; error fail", "proc p {} {operation x; foreach x {} {}; puts $x}; error fail"] {
        let mut analyser = Analyser::new();
        let analysis = analyser.analyse(source, "tcl8.6");
        analyser.result = analysis;
        let input = analyser.resolved_analysis_input();
        let settings = analyser.source_compilation_settings();
        let registry = input.borrowed_context_registry().commands();
        let profile = input.unit_profile();
        let unit = CompilationUnit::build_with_analysis_input_and_source_settings(source, UnitBuildOptions {registry, defer_top_level: false, config:input.lexer_config(), dialect:Some(profile), external_call_sites:None, declared_commands:None},None,&input,&settings).with_interprocedural(registry,Some(profile));
        let mut unresolvable=std::collections::HashSet::new();
        let mut resolvable=std::collections::HashSet::new();
        for f in unit.analysable_functions() {
            unresolvable.extend(tcl_compiler::interprocedural::collect_positioned_opaque_callee_name_args(&f.cfg,&|_|false));
            resolvable.extend(tcl_compiler::interprocedural::collect_positioned_opaque_callee_name_args(&f.cfg,&|_|true));
            for b in f.cfg.blocks.values() {for s in &b.statements {if s.synthetic_marker()==Some(tcl_compiler::ir::SyntheticMarker::UnseenCall) {println!("unseen {:?}",s.span());}}}
        }
        println!("source={source:?} unresolved_has_x={} resolved_has_x={}",unresolvable.contains("x"),resolvable.contains("x"));
    }
}
