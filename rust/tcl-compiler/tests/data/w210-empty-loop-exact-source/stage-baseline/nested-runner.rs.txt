use tcl_compiler::analyser::Analyser;
fn main() {
    for (label, source) in [
        ("nested-unknown", "proc p {} {foreach y [operation x] {}; return $x}; error fail"),
        ("nested-writer", "proc operation {} {upvar 1 x cell; set cell 7; return {}}; proc p {} {foreach y [operation] {}; return $x}; error fail"),
    ] {
        let result = Analyser::new().analyse(source, "tcl8.6");
        let spans: Vec<_> = result.diagnostics.iter().filter(|d| d.code.as_str() == "W210").map(|d| (d.span.start(), d.span.end())).collect();
        println!("{label} {spans:?}");
    }
}
