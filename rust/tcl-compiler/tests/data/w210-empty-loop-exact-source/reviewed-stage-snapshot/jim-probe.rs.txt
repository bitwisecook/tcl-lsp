use tcl_compiler::analyser::Analyser;
fn main() {
    for (case, source) in [
        ("exact-original", "proc p {} {\n    foreach x {} {}\n    if {[info exists x]} { puts yes }\n    puts $x\n}\nforeach y {} {}\nputs $y\n"),
        ("standalone-procedure-control", "proc p {} {foreach x {} {}; puts $x}"),
        ("top-level-control", "puts $missing"),
    ] {
        let result = Analyser::new().analyse(source, "jim");
        let spans: Vec<_> = result.diagnostics.iter().filter(|finding| finding.code.as_str() == "W210").map(|finding| (finding.span.start(), finding.span.end())).collect();
        println!("{case} resolved={} context_error={} w210={spans:?}", result.resolved_input.is_some(), result.analysis_context_unavailable.is_some());
    }
}
