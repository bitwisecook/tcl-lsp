use tcl_compiler::analyser::Analyser;
use tcl_compiler::compilation_unit::{CompilationUnit, UnitBuildOptions};

fn main() {
    for dialect in ["tcl8.6", "jim"] {
        for (case, source) in [
            ("combined", "proc p {} {\n    foreach x {} {}\n    if {[info exists x]} { puts yes }\n    puts $x\n}\nforeach y {} {}\nputs $y\n"),
            ("standalone", "proc p {} {foreach x {} {}; puts $x}"),
        ] {
            let mut analyser = Analyser::new();
            let result = analyser.analyse(source, dialect);
            let advice = result.allows_retained_logical_declaration_advice();
            analyser.result = result;
            let input = analyser.resolved_analysis_input();
            let settings = analyser.source_compilation_settings();
            let registry = input.borrowed_context_registry().commands();
            let profile = input.unit_profile();
            let unit = CompilationUnit::build_with_analysis_input_and_source_settings(
                source, UnitBuildOptions { registry, defer_top_level:false, config:input.lexer_config(), dialect:Some(profile), external_call_sites:None, declared_commands:None },
                None,&input,&settings,
            ).with_interprocedural(registry,Some(profile));
            println!("INPUT {dialect} {case} logical_context={} logical_advice={} native_entry={}",input.has_logical_source_name_context(),advice,unit.ir_module.source_entry.native_entry.is_some());
            for function in unit.analysable_body_function_units() {
                let names = tcl_compiler::interprocedural::collect_positioned_opaque_callee_name_args(&function.cfg,&|_|false);
                println!("INVENTORY {dialect} {case} {} x={} y={}",function.name,names.contains("x"),names.contains("y"));
                for block in function.cfg.blocks.values() {
                    for statement in &block.statements {
                        let Some(tokens) = statement.tokens() else {continue};
                        if tokens.argv_texts.first().is_none_or(|head|head!="foreach") {continue}
                        let binding=tokens.source_binding.as_ref();
                        println!("SITE {dialect} {case} {} span={:?} stage={:?} binding={} runtime={:?} unknown={:?} absent={:?} targets={:?} registry={:?} target={} execution={} native_compiler={}",
                            function.name,statement.span(),tokens.synthetic,binding.is_some(),binding.map(|b|b.runtime_reachability()),binding.map(|b|b.unknown),binding.map(|b|b.may_be_absent),binding.map(|b|b.targets.len()),binding.map(|b|b.targets.iter().all(|t|t.registry_backed)),binding.is_some_and(|b|b.proved_target().is_some()),binding.is_some_and(|b|b.proved_execution_target().is_some()),tcl_compiler::registry_invocation::admitted_native_compiler_invocation(registry,None,tokens).is_some());
                    }
                }
            }
        }
    }
}
