use tcl_compiler::analyser::Analyser;
use tcl_compiler::compilation_unit::{CompilationUnit, UnitBuildOptions};
use tcl_compiler::cfg::Terminator;
use tcl_compiler::ssa::SsaSourceView;

fn main() {
    for source in ["proc p {c} {if {$c} {set x 1}; if {[info exists a(safe)] && $a(safe) > 0} {puts yes}}", "proc p {c} {if {$c} {set x 1; set a(safe) 1}; if {[info exists a(safe)] && $a(safe) > 0} {puts yes}}"] {
    let mut analyser = Analyser::new();
    let _ = analyser.analyse(source, "tcl9.0");
    let input = analyser.resolved_analysis_input();
    let settings = analyser.source_compilation_settings();
    let registry = input.borrowed_context_registry().commands();
    let unit = CompilationUnit::build_with_analysis_input_and_source_settings(source, UnitBuildOptions {
        registry, defer_top_level: false, config: input.lexer_config(), dialect: Some(input.unit_profile()), external_call_sites: None, declared_commands: None,
    }, None, &input, &settings);
    for fu in unit.analysable_body_function_units() {
        for (&id, block) in &fu.cfg.blocks {
            if let Some(Terminator::Branch { condition, condition_base, .. }) = &block.terminator {
                let view = SsaSourceView::at_statement(&fu.ssa, id, usize::MAX);
                let tokens = view.source_tokens();
                println!("CFG {}", serde_json::json!({
                    "source": source, "variable_spans": condition.variable_nodes().iter().map(|node| format!("{:?}", condition_base.and_then(|base| node.variable_source_span(base, input.lexer_config())))).collect::<Vec<_>>(), "block": format!("{id:?}"), "base": condition_base, "expression": format!("{condition:?}"),
                    "tokens": tokens.is_some(), "point_contexts": fu.ssa.point_contexts.is_some(),
                    "accesses": tokens.map(|tokens| tokens.variable_accesses.iter().map(|access| serde_json::json!({
                        "spelling": access.original_spelling, "start": access.source.span.start(), "end": access.source.span.end(),
                        "reference": format!("{:?}", view.read_reference(&access.source, &access.original_spelling)),
                        "cell": format!("{:?}", view.read_reference(&access.source, &access.original_spelling).and_then(|read| fu.ssa.cell_key(read.symbol))),
                    })).collect::<Vec<_>>()),
                    "x_symbol": format!("{:?}", fu.ssa.var_symbol_at_terminator(id, "x")),
                }));
            }
        }
    }
}

}
