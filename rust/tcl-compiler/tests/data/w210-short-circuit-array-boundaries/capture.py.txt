import hashlib,json,os,subprocess
from pathlib import Path
sources=[
('same-element-absent','proc p {} {if {[info exists a(safe)] && $a(safe) > 0} {puts yes}}','p',0),
('same-element-maybound-missing','proc p {c} {if {$c} {set a(safe) 1}; if {[info exists a(safe)] && $a(safe) > 0} {puts yes}}','p 0',0),
('same-element-maybound-present','proc p {c} {if {$c} {set a(safe) 1}; if {[info exists a(safe)] && $a(safe) > 0} {puts yes}}','p 1',0),
('different-element-missing-root','proc p {c} {if {$c} {set a(safe) 1}; if {[info exists a(safe)] && $a(missing) > 0} {puts yes}}','p 0',0),
('different-element-live-root','proc p {c} {if {$c} {set a(safe) 1}; if {[info exists a(safe)] && $a(missing) > 0} {puts yes}}','p 1',1),
]
rows=[]
for key in ['TCL_LSP_TCLSH84','TCL_LSP_TCLSH85','TCL_LSP_TCLSH86','TCL_LSP_TCLSH90','TCL_LSP_TCLSH91','TCL_LSP_JIMSH']:
    binary=Path(os.environ[key]);digest=hashlib.sha256(binary.read_bytes()).hexdigest()
    version=subprocess.run([str(binary)],input='puts [info patchlevel]\n',text=True,capture_output=True,check=True).stdout.strip()
    for case,source,call,expected in sources:
        script=source+'\nset status [catch {'+call+'} result]\nputs "STATUS:$status"\n'
        result=subprocess.run([str(binary)],input=script,text=True,capture_output=True,timeout=10)
        statuses=[line for line in result.stdout.splitlines() if line.startswith('STATUS:')]
        status=int(statuses[-1].split(':')[1]) if statuses else None
        row={'case':case,'source':source,'call':call,'full_input':script,'input_sha256':hashlib.sha256(script.encode()).hexdigest(),'oracle':key,'binary':str(binary),'binary_sha256':digest,'patchlevel':version,'process_exit':result.returncode,'stdout':result.stdout,'stderr':result.stderr,'catch_status':status,'expected_catch_status':expected}
        rows.append(row)
Path('/tmp/lsp-2376-array-native.json').write_text(json.dumps(rows,indent=2)+'\n')
failures=[row for row in rows if row['process_exit'] or row['catch_status']!=row['expected_catch_status']]
print(json.dumps({'rows':len(rows),'failures':len(failures),'versions':sorted(set(r['patchlevel'] for r in rows))}));assert not failures
