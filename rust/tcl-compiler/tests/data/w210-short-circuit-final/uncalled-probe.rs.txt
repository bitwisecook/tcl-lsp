use tcl_compiler::analyser::Analyser;

fn main() {
    let cases = [
        (
            "safe-and",
            "proc p {c} {if {$c} {set x 1}; if {[info exists x] && $x > 0} {puts yes}}",
            "x",
            false,
            false,
        ),
        (
            "safe-negated-or",
            "proc p {c} {if {$c} {set x 1}; if {![info exists x] || $x > 0} {puts yes}}",
            "x",
            false,
            false,
        ),
        (
            "safe-double-not",
            "proc p {c} {if {$c} {set x 1}; if {!![info exists x] && $x > 0} {puts yes}}",
            "x",
            false,
            false,
        ),
        (
            "unsafe-or",
            "proc p {c} {if {$c} {set x 1}; if {[info exists x] || $x == 0} {puts yes}}",
            "x",
            true,
            false,
        ),
        (
            "unsafe-negated-and",
            "proc p {c} {if {$c} {set x 1}; if {![info exists x] && $x == 0} {puts yes}}",
            "x",
            true,
            false,
        ),
        (
            "later-genuine-read",
            "proc p {c} {if {$c} {set x 1}; if {[info exists x] && $x > 0} {puts yes}; puts $x}",
            "x",
            true,
            true,
        ),
        (
            "standalone-missing",
            "proc p {} {puts $__standalone_missing}",
            "__standalone_missing",
            true,
            false,
        ),
    ];
    let mut failures = Vec::new();
    for (label, source, variable, expected, last) in cases {
        for dialect in ["tcl8.4", "tcl8.5", "tcl8.6", "tcl9.0", "tcl9.1"] {
            let mut analyser = Analyser::new();
            let analysis = analyser.analyse(source, dialect);
            let findings: Vec<_> = analysis
                .diagnostics
                .iter()
                .filter(|finding| finding.code.to_string() == "W210")
                .collect();
            println!(
                "ENTRYLESS {}",
                serde_json::json!({
                    "case": label, "source": source, "dialect": dialect,
                    "expected_warning": expected,
                    "findings": findings.iter().map(|finding| serde_json::json!({
                        "start": finding.span.start(), "end": finding.span.end(), "message": finding.message,
                    })).collect::<Vec<_>>(),
                })
            );
            let mut correct = findings.len() == usize::from(expected);
            if expected && findings.len() == 1 {
                correct &= source.get(findings[0].span.as_range())
                    == Some(format!("${variable}").as_str());
            }
            if last && findings.len() == 1 {
                correct &= usize::try_from(findings[0].span.start()).unwrap()
                    == source.rfind(&format!("${variable}")).unwrap();
            }
            if !correct {
                failures.push(format!("{label} under {dialect}"));
            }
        }
    }
    println!("ENTRYLESS_FAILURES {}", serde_json::json!(failures));
    if !failures.is_empty() {
        std::process::exit(1);
    }
}
