//! Existence guards cannot hide a later destruction of their exact cell.

use tcl_compiler::analyser::Analyser;
use tcl_core_types::DiagCode;

#[test]
fn exists_guards_preserve_later_unset_diagnostics_and_independent_cells() {
    let fixtures: serde_json::Value =
        serde_json::from_str(include_str!("data/w210-unset-guards.json")).unwrap();
    for case in fixtures["cases"].as_array().unwrap() {
        let source = case["source"].as_str().unwrap();
        let variable = case["variable"].as_str().unwrap();
        let expected = case["expected_w210"].as_bool().unwrap();
        let quoted = format!("'{variable}'");
        for dialect in ["tcl8.4", "tcl8.5", "tcl8.6", "tcl9.0", "tcl9.1"] {
            let analysis = Analyser::new().analyse(source, dialect);
            let found: Vec<_> = analysis
                .diagnostics
                .iter()
                .filter(|finding| {
                    finding.code == DiagCode::W210 && finding.message.contains(&quoted)
                })
                .collect();
            assert_eq!(
                !found.is_empty(),
                expected,
                "{} under {dialect}: {:?}",
                case["id"],
                analysis.diagnostics
            );
            if expected {
                assert_eq!(found.len(), 1, "{} under {dialect}", case["id"]);
                assert_eq!(
                    source.get(found[0].span.as_range()),
                    Some(format!("${variable}").as_str()),
                    "{} under {dialect}",
                    case["id"]
                );
            }
        }
    }
}
